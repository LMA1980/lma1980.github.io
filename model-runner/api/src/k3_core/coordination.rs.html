3use core::fmt;
4use core::str::FromStr;
5
6use crate::error::K3Error;

8/// Multi-core worker coordination profile for layer and operator synchronization barriers.
9#[derive(
10    Debug, Clone, Copy, PartialEq, Eq, Hash, Default, serde::Serialize, serde::Deserialize,
11)]
12#[serde(rename_all = "lowercase")]
13pub enum CoordinationProfile {
14    /// Atomic busy-polling barrier with `core::hint::spin_loop()` (lowest latency, 100% CPU utilization).
15    #[default]
16    Spin,
17    /// Kernel-assisted thread sleep yielding CPU cycles via futex / Condvar.
18    Futex,
19    /// Dynamic task scheduling via a dedicated scoped Rayon work-stealing thread pool.
20    Rayon,
21}
22
23impl CoordinationProfile {
24    /// Parses a coordination profile string ("spin", "futex", "rayon").
25    ///
26    /// # Errors
27    /// Returns [`K3Error::CapabilityMismatch`] if the profile string is unrecognized.
28    pub fn parse(name: &str) -> Result<Self, K3Error> {
29        name.parse()
30    }
31
32    /// Returns the canonical CLI string identifier for this profile.
33    #[must_use]
34    pub const fn as_str(&self) -> &'static str {
35        match self {
36            Self::Spin => "spin",
37            Self::Futex => "futex",
38            Self::Rayon => "rayon",
39        }
40    }
41
42    /// Returns an array of all available coordination profiles.
43    #[must_use]
44    pub const fn all() -> [Self; 3] {
45        [Self::Spin, Self::Futex, Self::Rayon]
46    }
47}
48
49impl fmt::Display for CoordinationProfile {
50    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
51        write!(f, "{}", self.as_str())
52    }
53}
54
55impl FromStr for CoordinationProfile {
56    type Err = K3Error;
57
58    fn from_str(s: &str) -> Result<Self, Self::Err> {
59        match s.trim().to_ascii_lowercase().as_str() {
60            "spin" => Ok(Self::Spin),
61            "futex" => Ok(Self::Futex),
62            "rayon" => Ok(Self::Rayon),
63            other => Err(K3Error::CapabilityMismatch {
64                architecture: "coordination".to_owned(),
65                reason: format!(
66                    "unknown coordination profile '{other}': expected 'spin', 'futex', or 'rayon'"
67                ),
68            }),
69        }
70    }
71}

73#[cfg(test)]
74mod tests {
75    use super::*;
76
77    #[test]
78    fn test_coordination_profile_parsing_and_display() {
79        assert_eq!(CoordinationProfile::default(), CoordinationProfile::Spin);
80        assert_eq!(CoordinationProfile::Spin.as_str(), "spin");
81        assert_eq!(CoordinationProfile::Futex.as_str(), "futex");
82        assert_eq!(CoordinationProfile::Rayon.as_str(), "rayon");
83
84        assert_eq!(CoordinationProfile::Spin.to_string(), "spin");
85        assert_eq!(CoordinationProfile::Futex.to_string(), "futex");
86        assert_eq!(CoordinationProfile::Rayon.to_string(), "rayon");
87
88        assert_eq!(
89            "spin".parse::<CoordinationProfile>().unwrap(),
90            CoordinationProfile::Spin
91        );
92        assert_eq!(
93            "SPIN".parse::<CoordinationProfile>().unwrap(),
94            CoordinationProfile::Spin
95        );
96        assert_eq!(
97            " futex ".parse::<CoordinationProfile>().unwrap(),
98            CoordinationProfile::Futex
99        );
100        assert_eq!(
101            "Rayon".parse::<CoordinationProfile>().unwrap(),
102            CoordinationProfile::Rayon
103        );
104
105        let err = "invalid".parse::<CoordinationProfile>().unwrap_err();
106        assert!(matches!(err, K3Error::CapabilityMismatch { .. }));
107        assert_eq!(err.diagnostic_code(), "capability_mismatch");
108
109        let all = CoordinationProfile::all();
110        assert_eq!(all.len(), 3);
111        assert_eq!(all[0], CoordinationProfile::Spin);
112        assert_eq!(all[1], CoordinationProfile::Futex);
113        assert_eq!(all[2], CoordinationProfile::Rayon);
114    }
115
116    #[test]
117    fn test_coordination_profile_parse_method() {
118        assert_eq!(
119            CoordinationProfile::parse("spin").unwrap(),
120            CoordinationProfile::Spin
121        );
122        assert_eq!(
123            CoordinationProfile::parse("FUTEX").unwrap(),
124            CoordinationProfile::Futex
125        );
126        assert_eq!(
127            CoordinationProfile::parse("rayon").unwrap(),
128            CoordinationProfile::Rayon
129        );
130        assert!(CoordinationProfile::parse("unknown").is_err());
131    }
