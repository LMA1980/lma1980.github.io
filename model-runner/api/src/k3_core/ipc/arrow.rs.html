1//! Apache Arrow zero-copy tensor views over shared memory arenas.
2//!
3//! Provides zero-copy wrapping of mapped shared memory into Apache Arrow
4//! buffers and primitive arrays (`UInt32Array`, `Float32Array`).
5//!
6//! Zero bulk payload bytes are copied across processes:
7//! `Buffer::from_custom_allocation` retains shared ownership of the backing
8//! arena via `Arc<dyn Allocation>`, allowing readers to inspect tokens,
9//! logits, and KV cache blocks via standard Apache Arrow columnar interfaces.
10

26/// Externally managed allocation holder for Arrow buffers referencing shared memory.
27///
28/// Keeps the underlying arena or memory mapping alive for the lifetime of
29/// the Apache Arrow `Buffer` and any dependent `Array`s.
30pub struct ShmArrowAllocation {
31    _owner: Arc<dyn Any + Send + Sync + RefUnwindSafe>,
32}
33
34impl fmt::Debug for ShmArrowAllocation {
35    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
36        f.debug_struct("ShmArrowAllocation").finish_non_exhaustive()
37    }
38}

40impl ShmArrowAllocation {
41    /// Creates a new allocation holder wrapping an arbitrary thread-safe owner.
42    #[allow(clippy::new_ret_no_self)]
43    pub fn new<T: Send + Sync + RefUnwindSafe + 'static>(owner: Arc<T>) -> Arc<dyn Allocation> {
44        Arc::new(Self { _owner: owner })
45    }
46}

48/// Creates an Apache Arrow [`Buffer`] referencing a contiguous region in shared memory.
49///
50/// # Safety
51/// - `ptr` must be non-null and point to a valid, initialized byte region of at least `byte_len` bytes.
52/// - The memory region must remain valid and immutable (or free of concurrent data races)
53///   for as long as `owner` is retained.
54/// - The caller must guarantee that reading from this memory does not violate Rust aliasing invariants.
55///
56/// # Errors
57/// Returns [`K3Error::TensorBoundsError`] if `ptr` is null.
58pub unsafe fn create_arrow_buffer_from_shm<T: Send + Sync + RefUnwindSafe + 'static>(
59    owner: Arc<T>,
60    ptr: *const u8,
61    byte_len: usize,
62) -> Result<Buffer, K3Error> {
63    let non_null = NonNull::new(ptr.cast_mut()).ok_or_else(|| K3Error::TensorBoundsError {
64        tensor_name: "arrow_shm_buffer".to_owned(),
65        offset: 0,
66        byte_len,
67        buffer_len: byte_len,
68        reason: "provided memory pointer is null".to_owned(),
69    })?;
70
71    let allocation_owner = ShmArrowAllocation::new(owner);
72
73    // SAFETY:
74    // 1. `non_null` is validated non-null above.
75    // 2. The backing allocation is kept alive by `allocation_owner` until all referring
76    //    Arrow buffers and arrays are dropped.
77    // 3. Caller guarantees the pointer references at least `byte_len` valid initialized bytes.
78    unsafe {
79        Ok(Buffer::from_custom_allocation(
80            non_null,
81            byte_len,
82            allocation_owner,
83        ))
84    }
85}

87/// Slices an existing Apache Arrow [`Buffer`] without copying bytes.
88///
89/// # Errors
90/// Returns [`K3Error::TensorBoundsError`] if `offset + byte_len` exceeds `buffer.len()`.
91pub fn slice_arrow_buffer(
92    buffer: &Buffer,
93    offset: usize,
94    byte_len: usize,
95) -> Result<Buffer, K3Error> {
96    let end_offset = offset
97        .checked_add(byte_len)
98        .ok_or_else(|| K3Error::TensorBoundsError {
99            tensor_name: "buffer_slice".to_owned(),
100            offset,
101            byte_len,
102            buffer_len: buffer.len(),
103            reason: "offset + byte_len overflowed usize".to_owned(),
104        })?;
105
106    if end_offset > buffer.len() {
107        return Err(K3Error::TensorBoundsError {
108            tensor_name: "buffer_slice".to_owned(),
109            offset,
110            byte_len,
111            buffer_len: buffer.len(),
112            reason: format!(
113                "slice range [{offset}..{end_offset}] exceeds buffer length {len}",
114                len = buffer.len()
115            ),
116        });
117    }
118
119    Ok(buffer.slice_with_length(offset, byte_len))
120}

122/// Exposes shared memory input tokens as an Apache Arrow [`UInt32Array`] without copying.
123///
124/// # Safety
125/// - `tokens_ptr` must be non-null and aligned to `align_of::<u32>()` (4 bytes).
126/// - Points to at least `token_count` valid `u32` token values in shared memory.
127/// - `owner` must keep the memory mapping alive.
128///
129/// # Errors
130/// Returns [`K3Error::TensorBoundsError`] if pointer is null, unaligned, or size overflows `usize`.
131pub unsafe fn wrap_tokens_as_uint32_array<T: Send + Sync + RefUnwindSafe + 'static>(
132    owner: Arc<T>,
133    tokens_ptr: *const u32,
134    token_count: usize,
135) -> Result<UInt32Array, K3Error> {
136    let ptr_val = tokens_ptr as usize;
137    if ptr_val % align_of::<u32>() != 0 {
138        return Err(K3Error::TensorBoundsError {
139            tensor_name: "token_input_array".to_owned(),
140            offset: ptr_val,
141            byte_len: token_count * 4,
142            buffer_len: token_count * 4,
143            reason: format!(
144                "tokens pointer 0x{ptr_val:X} is not 4-byte aligned (modulo: {m})",
145                m = ptr_val % 4
146            ),
147        });
148    }
149
150    let byte_len =
151        token_count
152            .checked_mul(size_of::<u32>())
153            .ok_or_else(|| K3Error::TensorBoundsError {
154                tensor_name: "token_input_array".to_owned(),
155                offset: 0,
156                byte_len: usize::MAX,
157                buffer_len: 0,
158                reason: "token count byte length overflowed usize".to_owned(),
159            })?;
160
161    // SAFETY: Caller guarantees pointer validity and lifetime managed by owner.
162    let buffer = unsafe { create_arrow_buffer_from_shm(owner, tokens_ptr.cast::<u8>(), byte_len)? };
163    let scalar_buffer = ScalarBuffer::<u32>::new(buffer, 0, token_count);
164    Ok(UInt32Array::new(scalar_buffer, None))
165}

167/// Exposes shared memory logits or KV cache vectors as an Apache Arrow [`Float32Array`] without copying.
168///
169/// # Safety
170/// - `data_ptr` must be non-null and aligned to `align_of::<f32>()` (4 bytes).
171/// - Points to at least `element_count` valid `f32` float values in shared memory.
172/// - `owner` must keep the memory mapping alive.
173///
174/// # Errors
175/// Returns [`K3Error::TensorBoundsError`] if pointer is null, unaligned, or size overflows `usize`.
176pub unsafe fn wrap_floats_as_float32_array<T: Send + Sync + RefUnwindSafe + 'static>(
177    owner: Arc<T>,
178    name: &str,
179    data_ptr: *const f32,
180    element_count: usize,
181) -> Result<Float32Array, K3Error> {
182    let ptr_val = data_ptr as usize;
183    if ptr_val % align_of::<f32>() != 0 {
184        return Err(K3Error::TensorBoundsError {
185            tensor_name: name.to_owned(),
186            offset: ptr_val,
187            byte_len: element_count * 4,
188            buffer_len: element_count * 4,
189            reason: format!(
190                "float pointer 0x{ptr_val:X} for tensor '{name}' is not 4-byte aligned",
191            ),
192        });
193    }
194
195    let byte_len =
196        element_count
197            .checked_mul(size_of::<f32>())
198            .ok_or_else(|| K3Error::TensorBoundsError {
199                tensor_name: name.to_owned(),
200                offset: 0,
201                byte_len: usize::MAX,
202                buffer_len: 0,
203                reason: format!("element count {element_count} byte length overflowed usize"),
204            })?;
205
206    // SAFETY: Caller guarantees pointer validity and lifetime managed by owner.
207    let buffer = unsafe { create_arrow_buffer_from_shm(owner, data_ptr.cast::<u8>(), byte_len)? };
208    let scalar_buffer = ScalarBuffer::<f32>::new(buffer, 0, element_count);
209    Ok(Float32Array::new(scalar_buffer, None))
210}

212/// High-level zero-copy Apache Arrow tensor views over a [`SharedMemoryArena`].
213#[derive(Debug, Clone)]
214pub struct ShmTensorViews {
215    arena: Arc<SharedMemoryArena>,
216}
217
218impl ShmTensorViews {
219    /// Creates a new view provider over an arena reference.
220    #[must_use]
221    pub fn new(arena: Arc<SharedMemoryArena>) -> Self {
222        Self { arena }
223    }
224
225    /// Returns the backing arena.
226    #[must_use]
227    pub fn arena(&self) -> &Arc<SharedMemoryArena> {
228        &self.arena
229    }
230
231    /// Returns a zero-copy Apache Arrow [`UInt32Array`] view over input prompt tokens.
232    ///
233    /// # Errors
234    /// Returns [`K3Error::TensorBoundsError`] if slot is out of bounds or length exceeds arena.
235    #[allow(clippy::cast_ptr_alignment)]
236    pub fn tokens_view(&self, slot_idx: usize) -> Result<UInt32Array, K3Error> {
237        let hdr = self.arena.header()?;
238        let slot_offset =
239            usize::try_from(hdr.token_slots_offset).map_err(|e| K3Error::TensorBoundsError {
240                tensor_name: "tokens_view".to_owned(),
241                offset: 0,
242                byte_len: 0,
243                buffer_len: self.arena.len(),
244                reason: format!("token_slots_offset exceeds usize: {e}"),
245            })?;
246        let slot_base = slot_offset
247            .checked_add(slot_idx.checked_mul(64 * 4 + 16).ok_or_else(|| {
248                K3Error::TensorBoundsError {
249                    tensor_name: "tokens_view".to_owned(),
250                    offset: 0,
251                    byte_len: 0,
252                    buffer_len: self.arena.len(),
253                    reason: "slot index overflowed usize".to_owned(),
254                }
255            })?)
256            .ok_or_else(|| K3Error::TensorBoundsError {
257                tensor_name: "tokens_view".to_owned(),
258                offset: 0,
259                byte_len: 0,
260                buffer_len: self.arena.len(),
261                reason: "slot index overflowed usize".to_owned(),
262            })?;
263
264        let slot_slice = self.arena.slice(slot_base, 16)?;
265        let token_hdr =
266            super::layout::TokenInputSlotHeader::ref_from_bytes(slot_slice).map_err(|e| {
267                K3Error::TensorBoundsError {
268                    tensor_name: format!("token_slot[{slot_idx}]"),
269                    offset: slot_base,
270                    byte_len: 16,
271                    buffer_len: self.arena.len(),
272                    reason: format!("failed to decode TokenInputSlotHeader: {e}"),
273                }
274            })?;
275
276        let token_count = token_hdr.token_len as usize;
277        let tokens_data_offset = slot_base + 16;
278        let tokens_bytes = self.arena.slice(tokens_data_offset, token_count * 4)?;
279
280        // SAFETY: tokens_bytes was validated within bounds and 16-byte aligned.
281        unsafe {
282            wrap_tokens_as_uint32_array(
283                Arc::clone(&self.arena),
284                tokens_bytes.as_ptr().cast::<u32>(),
285                token_count,
286            )
287        }
288    }
289
290    /// Returns a zero-copy Apache Arrow [`Float32Array`] view over output logits.
291    ///
292    /// # Errors
293    /// Returns [`K3Error::TensorBoundsError`] if slab offset or vocab length exceeds arena bounds.
294    #[allow(clippy::cast_ptr_alignment)]
295    pub fn logits_view(&self) -> Result<Float32Array, K3Error> {
296        let hdr = self.arena.header()?;
297        let slab_offset =
298            usize::try_from(hdr.output_slabs_offset).map_err(|e| K3Error::TensorBoundsError {
299                tensor_name: "output_slab".to_owned(),
300                offset: 0,
301                byte_len: 0,
302                buffer_len: self.arena.len(),
303                reason: format!("output_slabs_offset exceeds usize: {e}"),
304            })?;
305        let slab_slice = self.arena.slice(slab_offset, 64)?;
306        let slab_hdr =
307            super::layout::OutputSlabHeader::ref_from_bytes(slab_slice).map_err(|e| {
308                K3Error::TensorBoundsError {
309                    tensor_name: "output_slab".to_owned(),
310                    offset: slab_offset,
311                    byte_len: 64,
312                    buffer_len: self.arena.len(),
313                    reason: format!("failed to decode OutputSlabHeader: {e}"),
314                }
315            })?;
316
317        let vocab_size = slab_hdr.vocab_size as usize;
318        let logits_offset = slab_offset + 64;
319        let logits_bytes = self.arena.slice(logits_offset, vocab_size * 4)?;
320
321        // SAFETY: logits_bytes is 64-byte aligned and checked within bounds.
322        unsafe {
323            wrap_floats_as_float32_array(
324                Arc::clone(&self.arena),
325                "output_logits",
326                logits_bytes.as_ptr().cast::<f32>(),
327                vocab_size,
328            )
329        }
330    }
331}

337    #[test]
338    fn test_arrow_zero_copy_buffer_creation_and_slicing() {
339        let data = vec![10_u32, 20, 30, 40, 50];
340        let owner = Arc::new(data);
341
342        // SAFETY: pointer points to valid 20 bytes of initialized u32 data.
343        let buf = unsafe {
344            create_arrow_buffer_from_shm(Arc::clone(&owner), owner.as_ptr().cast::<u8>(), 20)
345        }
346        .expect("arrow buffer creation");
347
348        assert_eq!(buf.len(), 20);
349        assert_eq!(buf.as_ptr(), owner.as_ptr().cast::<u8>());
350
351        let sliced = slice_arrow_buffer(&buf, 4, 12).expect("slice");
352        assert_eq!(sliced.len(), 12);
353        // SAFETY: 4 bytes into the buffer is &owner[1].
354        assert_eq!(sliced.as_ptr(), unsafe {
355            owner.as_ptr().add(1).cast::<u8>()
356        });
357
358        let oob = slice_arrow_buffer(&buf, 10, 20);
359        assert!(matches!(oob, Err(K3Error::TensorBoundsError { .. })));
360    }

400    #[cfg_attr(miri, ignore)]
401    fn test_shm_tensor_views_end_to_end() {
402        use zerocopy::IntoBytes;
403
404        let mut arena = SharedMemoryArena::create("views_test", 65536).expect("arena");
405        {
406            let hdr = arena.header_mut().expect("header_mut");
407            *hdr = super::super::layout::ArenaHeader::default();
408            hdr.token_slots_offset = 1024;
409            hdr.output_slabs_offset = 4096;
410        }
411
412        // Setup token slot
413        {
414            let slot_hdr = super::super::layout::TokenInputSlotHeader {
415                slot_seq_id: 1,
416                token_len: 3,
417                start_position: 0,
418            };
419            arena
420                .slice_mut(1024, 16)
421                .expect("slot_hdr")
422                .copy_from_slice(slot_hdr.as_bytes());
423            let tokens_slice = arena.slice_mut(1040, 12).expect("tokens_slice");
424            let tokens = [42_u32, 43, 44];
425            tokens_slice.copy_from_slice(tokens.as_bytes());
426        }
427
428        // Setup output slab
429        {
430            let slab_hdr = super::super::layout::OutputSlabHeader {
431                request_id: 1,
432                vocab_size: 4,
433                position: 0,
434                status: super::super::layout::SLAB_STATUS_WRITTEN,
435                _reserved: [0; 44],
436            };
437            arena
438                .slice_mut(4096, 64)
439                .expect("slab_hdr")
440                .copy_from_slice(slab_hdr.as_bytes());
441            let logits_slice = arena.slice_mut(4160, 16).expect("logits_slice");
442            let logits = [1.0_f32, 2.0, 3.0, 4.0];
443            logits_slice.copy_from_slice(logits.as_bytes());
444        }
445
446        let views = ShmTensorViews::new(Arc::new(arena));
447        let token_array = views.tokens_view(0).expect("token_array");
448        assert_eq!(token_array.len(), 3);
449        assert_eq!(token_array.value(0), 42);
450        assert_eq!(token_array.value(2), 44);
451
452        let logits_array = views.logits_view().expect("logits_array");
453        assert_eq!(logits_array.len(), 4);
454        assert!((logits_array.value(1) - 2.0).abs() < f32::EPSILON);
455    }
