1//! In-place shared memory arena layouts for the `SpacemiT` K3 runtime.
2//!
3//! All structures are `#[repr(C)]` with zero implicit compiler padding,
4//! implementing zerocopy 0.8 traits (`FromBytes`, `IntoBytes`, `KnownLayout`, `Immutable`).
5//!
6//! Cacheline-aligned boundaries (64 bytes) ensure optimal memory bus transactions
7//! and vector memory operations on `SpacemiT` K3 X100 (VLEN=256) and A100 (VLEN=1024) cores.
8
9use core::mem::{align_of, size_of};
10
11use zerocopy::{FromBytes, Immutable, IntoBytes, KnownLayout};
12
13/// Magic identifier for K3 shared memory arena (`0x4B334B56` = b"K3KV").
14pub const ARENA_MAGIC: u32 = 0x4B33_4B56;
15
16/// Magic identifier byte array `*b"K3KV"`.
17pub const ARENA_MAGIC_BYTES: [u8; 4] = *b"K3KV";
18
19/// Version of the arena layout schema.
20pub const ARENA_VERSION: u32 = 1;
21
22/// Fixed number of tokens per KV cache block.
23pub const BLOCK_SIZE_TOKENS: u32 = 16;
24
25/// Standard hardware cacheline alignment (64 bytes).
26pub const CACHELINE_ALIGNMENT: usize = 64;
27
28// Block tag constants per ADR-018 & ADR-042
29pub const TAG_FREE: u8 = 0;
30pub const TAG_VOLATILE: u8 = 1;
31pub const TAG_PERSIST: u8 = 2;
32pub const TAG_PURGE: u8 = 3;
33
34/// Symbolic enumeration of block retention tags.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
36#[repr(u8)]
37pub enum KvBlockTag {
38    /// Available for allocation.
39    Free = TAG_FREE,
40    /// Ephemeral context (discarded on eviction or session completion).
41    Volatile = TAG_VOLATILE,
42    /// Persistent checkpoint (retained across worker restarts and prompt sharing).
43    Persist = TAG_PERSIST,
44    /// Explicitly marked for eviction and zeroing.
45    Purge = TAG_PURGE,
46}
47
48impl KvBlockTag {
49    /// Converts a raw `u8` to a [`KvBlockTag`].
50    #[must_use]
51    pub const fn from_u8(val: u8) -> Option<Self> {
52        match val {
53            TAG_FREE => Some(Self::Free),
54            TAG_VOLATILE => Some(Self::Volatile),
55            TAG_PERSIST => Some(Self::Persist),
56            TAG_PURGE => Some(Self::Purge),
57            _ => None,
58        }
59    }
60
61    /// Returns the raw `u8` representation.
62    #[must_use]
63    pub const fn as_u8(self) -> u8 {
64        self as u8
65    }
66}

68/// Fixed 128-byte header aligned to 64-byte cacheline boundaries at offset 0 of the arena.
69///
70/// Contains placement geometry, sizing bounds, and byte offsets to all arena sub-regions.
71#[repr(C, align(64))]
72#[allow(clippy::pub_underscore_fields)]
73#[derive(Debug, Clone, Copy, PartialEq, Eq, FromBytes, IntoBytes, KnownLayout, Immutable)]
74pub struct ArenaHeader {
75    /// Magic identifier (`0x4B334B56` / `b"K3KV"`).
76    pub magic: u32,
77    /// Layout schema version (currently 1).
78    pub version: u32,
79    /// Total capacity of the mapped shared memory arena in bytes.
80    pub total_bytes: u64,
81    /// Total number of 16-token KV blocks allocated in this arena.
82    pub block_count: u32,
83    /// Number of tokens per block (fixed at 16).
84    pub tokens_per_block: u32,
85    /// Maximum configured context length (`n_ctx`).
86    pub n_ctx: u32,
87    /// Attention head dimension (e.g. 64, 128).
88    pub head_dim: u32,
89    /// Number of transformer layers in the model.
90    pub num_layers: u32,
91    /// Number of key-value attention heads (for GQA/MQA models).
92    pub head_count_kv: u32,
93    /// Model vocabulary size (dimension of output logits).
94    pub vocab_size: u32,
95    /// Number of active token slots (typically 1 for single-stream worker).
96    pub active_slots: u32,
97    /// Byte offset from arena base to the [`KvBlockMeta`] table.
98    pub meta_table_offset: u64,
99    /// Byte offset from arena base to the input token slot region.
100    pub token_slots_offset: u64,
101    /// Byte offset from arena base to the KV cache block vector data region.
102    pub kv_blocks_offset: u64,
103    /// Byte offset from arena base to the output logit slab region.
104    pub output_slabs_offset: u64,
105    /// Monotonically incrementing generation counter for state tracking.
106    pub generation_counter: u64,
107    /// Explicit padding to ensure exact 128-byte size with zero compiler gaps.
108    pub _reserved: [u8; 40],
109}
110
111impl Default for ArenaHeader {
112    fn default() -> Self {
113        Self {
114            magic: ARENA_MAGIC,
115            version: ARENA_VERSION,
116            total_bytes: 0,
117            block_count: 4,
118            tokens_per_block: BLOCK_SIZE_TOKENS,
119            n_ctx: 64,
120            head_dim: 64,
121            num_layers: 2,
122            head_count_kv: 2,
123            vocab_size: 32000,
124            active_slots: 1,
125            meta_table_offset: 128,
126            token_slots_offset: 128 + 4 * 64,
127            kv_blocks_offset: 1024,
128            output_slabs_offset: 4096,
129            generation_counter: 0,
130            _reserved: [0; 40],
131        }
132    }
133}

135impl ArenaHeader {
136    /// Validates magic and version consistency.
137    ///
138    /// # Errors
139    /// Returns an error message if magic, version, or token block size are invalid.
140    pub fn validate(&self) -> Result<(), &'static str> {
141        if self.magic != ARENA_MAGIC {
142            return Err("invalid arena magic signature");
143        }
144        if self.version != ARENA_VERSION {
145            return Err("unsupported arena layout version");
146        }
147        if self.tokens_per_block != BLOCK_SIZE_TOKENS {
148            return Err("tokens_per_block must be 16");
149        }
150        Ok(())
151    }
152
153    /// Returns the magic number as 4 big-endian bytes.
154    #[must_use]
155    pub const fn magic_bytes(&self) -> [u8; 4] {
156        self.magic.to_be_bytes()
157    }
158}

160/// 64-byte aligned metadata record for a 16-token KV cache block.
161///
162/// Designed to fit exactly within a single 64-byte cacheline.
163/// Zero implicit padding bytes ensure seamless zerocopy casting.
164#[repr(C, align(64))]
165#[allow(clippy::pub_underscore_fields)]
166#[derive(Debug, Clone, Copy, PartialEq, Eq, FromBytes, IntoBytes, KnownLayout, Immutable)]
167pub struct KvBlockMeta {
168    /// Logical block index within the arena (`0 .. block_count`).
169    pub block_id: u32,
170    /// Retention policy tag (0=free, 1=volatile, 2=persist, 3=purge).
171    pub tag: u8,
172    /// Explicit padding byte aligning `refcount` to 2-byte boundary.
173    pub _pad0: u8,
174    /// Reference count for shared prefix caching.
175    pub refcount: u16,
176    /// Number of tokens committed to this block (`0 ..= 16`).
177    pub committed_token_count: u32,
178    /// Explicit padding aligning `sampler_rng_state` to 8-byte boundary.
179    pub _pad1: u32,
180    /// Checkpointed 256-bit PRNG state (e.g. for resume from checkpoint without re-prefill).
181    pub sampler_rng_state: [u64; 4],
182    /// Reserved trailing padding completing the 64-byte cacheline.
183    pub padding: [u8; 16],
184}
185
186impl Default for KvBlockMeta {
187    fn default() -> Self {
188        Self {
189            block_id: 0,
190            tag: TAG_VOLATILE,
191            _pad0: 0,
192            refcount: 1,
193            committed_token_count: 0,
194            _pad1: 0,
195            sampler_rng_state: [42, 0, 0, 0],
196            padding: [0; 16],
197        }
198    }
199}

201impl KvBlockMeta {
202    /// Creates a new [`KvBlockMeta`] with the specified block ID and initial committed token count.
203    #[must_use]
204    pub const fn new(block_id: u32, committed_token_count: u32) -> Self {
205        Self {
206            block_id,
207            tag: TAG_VOLATILE,
208            _pad0: 0,
209            refcount: 1,
210            committed_token_count,
211            _pad1: 0,
212            sampler_rng_state: [42, 0, 0, 0],
213            padding: [0; 16],
214        }
215    }
216
217    /// Returns the parsed [`KvBlockTag`].
218    #[must_use]
219    pub fn tag_enum(&self) -> Option<KvBlockTag> {
220        KvBlockTag::from_u8(self.tag)
221    }
222
223    /// Sets the block retention tag.
224    pub fn set_tag(&mut self, tag: KvBlockTag) {
225        self.tag = tag.as_u8();
226    }
227
228    /// Returns the number of tokens committed to this block.
229    #[must_use]
230    pub const fn committed_tokens(&self) -> u32 {
231        self.committed_token_count
232    }
233
234    /// Sets the number of committed tokens.
235    pub fn set_committed_tokens(&mut self, count: u32) {
236        self.committed_token_count = count;
237    }
238
239    /// Returns the current reference count.
240    #[must_use]
241    #[inline]
242    pub const fn ref_count(&self) -> u16 {
243        self.refcount
244    }
245
246    /// Increments the reference count using saturating addition.
247    ///
248    /// Returns the updated reference count.
249    #[inline]
250    pub fn inc_ref(&mut self) -> u16 {
251        self.refcount = self.refcount.saturating_add(1);
252        self.refcount
253    }
254
255    /// Decrements the reference count using saturating subtraction.
256    ///
257    /// Returns the updated reference count.
258    #[inline]
259    pub fn dec_ref(&mut self) -> u16 {
260        self.refcount = self.refcount.saturating_sub(1);
261        self.refcount
262    }
263
264    /// Returns `true` if this block is shared by multiple sessions (`refcount > 1`).
265    #[must_use]
266    #[inline]
267    pub const fn is_shared(&self) -> bool {
268        self.refcount > 1
269    }
270
271    /// Returns `true` if mutating this block requires a Copy-On-Write (COW) clone (`refcount > 1`).
272    #[must_use]
273    #[inline]
274    pub const fn needs_cow(&self) -> bool {
275        self.refcount > 1
276    }
277
278    /// Returns `true` if this block is unreferenced and eligible for LRU eviction (`refcount == 0 && tag == TAG_VOLATILE`).
279    #[must_use]
280    #[inline]
281    pub const fn is_evictable(&self) -> bool {
282        self.refcount == 0 && self.tag == TAG_VOLATILE
283    }
284}

286/// Header preceding an input token slot in shared memory.
287///
288/// Followed immediately by `token_len * 4` bytes of `u32` token IDs.
289#[repr(C, align(8))]
290#[derive(
291    Default, Debug, Clone, Copy, PartialEq, Eq, FromBytes, IntoBytes, KnownLayout, Immutable,
292)]
293pub struct TokenInputSlotHeader {
294    /// Unique sequence identifier or request ID for this input.
295    pub slot_seq_id: u64,
296    /// Number of active tokens present in the trailing tokens array.
297    pub token_len: u32,
298    /// Starting sequence position for decode.
299    pub start_position: u32,
300}

302/// Slab status indicators for synchronization between worker and supervisor.
303pub const SLAB_STATUS_IDLE: u32 = 0;
304pub const SLAB_STATUS_WRITTEN: u32 = 1;
305pub const SLAB_STATUS_READ: u32 = 2;
306
307/// 64-byte aligned header preceding the output logits slab in shared memory.
308///
309/// Logit float values (`[f32; vocab_size]`) start immediately at offset 64,
310/// ensuring full cacheline alignment for vector stores (`vse32.v`).
311#[repr(C, align(64))]
312#[allow(clippy::pub_underscore_fields)]
313#[derive(Debug, Clone, Copy, PartialEq, Eq, FromBytes, IntoBytes, KnownLayout, Immutable)]
314pub struct OutputSlabHeader {
315    /// Associated forward pass request identifier.
316    pub request_id: u64,
317    /// Vocabulary dimension (number of `f32` logit elements).
318    pub vocab_size: u32,
319    /// Sequence position corresponding to these logits.
320    pub position: u32,
321    /// Status flag (0=idle, 1=written, 2=read).
322    pub status: u32,
323    /// Reserved trailing padding completing the 64-byte header.
324    pub _reserved: [u8; 44],
325}
326
327impl Default for OutputSlabHeader {
328    fn default() -> Self {
329        Self {
330            request_id: 0,
331            vocab_size: 0,
332            position: 0,
333            status: SLAB_STATUS_IDLE,
334            _reserved: [0; 44],
335        }
336    }
337}

447    #[test]
448    fn test_kv_block_meta_refcounting_lifecycle() {
449        let mut meta = KvBlockMeta::default();
450        // default has refcount=1, tag=TAG_VOLATILE
451        assert_eq!(meta.refcount, 1);
452        assert!(!meta.is_shared());
453        assert!(!meta.needs_cow());
454        assert!(!meta.is_evictable());
455
456        // Increment refcount
457        assert_eq!(meta.inc_ref(), 2);
458        assert!(meta.is_shared());
459        assert!(meta.needs_cow());
460        assert!(!meta.is_evictable());
461
462        // Decrement refcount back to 1
463        assert_eq!(meta.dec_ref(), 1);
464        assert!(!meta.is_shared());
465        assert!(!meta.needs_cow());
466        assert!(!meta.is_evictable());
467
468        // Decrement to 0
469        assert_eq!(meta.dec_ref(), 0);
470        assert!(!meta.is_shared());
471        assert!(!meta.needs_cow());
472        assert!(meta.is_evictable()); // refcount == 0 && tag == TAG_VOLATILE
473
474        // Saturating sub underflow safety
475        assert_eq!(meta.dec_ref(), 0);
476        assert_eq!(meta.refcount, 0);
477
478        // Saturating add overflow safety
479        meta.refcount = u16::MAX;
480        assert_eq!(meta.inc_ref(), u16::MAX);
481        assert_eq!(meta.refcount, u16::MAX);
482
483        // If tag is not TAG_VOLATILE, not evictable even with refcount 0
484        meta.refcount = 0;
485        meta.set_tag(KvBlockTag::Persist);
486        assert!(!meta.is_evictable());
487
488        meta.set_tag(KvBlockTag::Free);
489        assert!(!meta.is_evictable());
490    }
