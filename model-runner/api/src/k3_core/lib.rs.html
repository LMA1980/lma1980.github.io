50    impl ModelWeights for DummyWeights<'_> {
51        fn tensor(&self, name: &str) -> Result<TensorView<'_>, K3Error> {
52            match name {
53                "blk.0.attn_q.weight" => TensorView::from_bytes(
54                    "blk.0.attn_q.weight",
55                    QuantType::Q4_K_M,
56                    &[256, 2],
57                    0,
58                    self.buffer,
59                ),
60                "output_norm.weight" => TensorView::from_bytes(
61                    "output_norm.weight",
62                    QuantType::F32,
63                    &[16],
64                    288,
65                    self.buffer,
66                ),
67                _ => Err(K3Error::TensorBoundsError {
68                    tensor_name: name.to_owned(),
69                    offset: 0,
70                    byte_len: 0,
71                    buffer_len: self.buffer.len(),
72                    reason: "tensor not found".to_owned(),
73                }),
74            }
75        }
76
77        fn tensor_count(&self) -> usize {
78            2
79        }
80
81        fn total_weight_bytes(&self) -> usize {
82            self.buffer.len()
83        }
84    }

87    struct DummyRecipe;
88
89    impl ArchRecipe for DummyRecipe {
90        fn architecture_id(&self) -> &'static str {
91            "synthetic"
92        }
93
94        fn supports_quant(&self, quant: QuantType) -> bool {
95            matches!(quant, QuantType::F32 | QuantType::F16 | QuantType::Q4_K_M)
96        }
97
98        fn prefill_step(
99            &self,
100            ctx: &mut ExecutionContext<'_>,
101            weights: &dyn ModelWeights,
102        ) -> Result<(), K3Error> {
103            let q = weights.tensor("blk.0.attn_q.weight")?;
104            let norm = weights.tensor("output_norm.weight")?;
105            assert_eq!(q.byte_len(), 288);
106            assert_eq!(norm.byte_len(), 64);
107
108            let prompt_len = ctx.prompt_tokens().len();
109            ctx.advance_position(prompt_len)?;
110            if let Some(first_logit) = ctx.logits_mut().first_mut() {
111                *first_logit = 1.0;
112            }
113            Ok(())
114        }
115
116        fn decode_step(
117            &self,
118            ctx: &mut ExecutionContext<'_>,
119            _weights: &dyn ModelWeights,
120        ) -> Result<(), K3Error> {
121            let token = ctx
122                .decode_token()
123                .ok_or_else(|| K3Error::CapabilityMismatch {
124                    architecture: self.architecture_id().to_owned(),
125                    reason: "decode_step called without a staged decode token".to_owned(),
126                })?;
127            ctx.advance_position(1)?;
128            if let Some(slot) = ctx.logits_mut().get_mut(token as usize) {
129                *slot = 2.0;
130            }
131            Ok(())
132        }
133    }

136    fn quant_type_storage_bytes_and_gguf_mapping() {
137        assert_eq!(QuantType::from_gguf_type_id(0), Some(QuantType::F32));
138        assert_eq!(QuantType::from_gguf_type_id(1), Some(QuantType::F16));
139        assert_eq!(QuantType::from_gguf_type_id(2), Some(QuantType::Q4_0));
140        assert_eq!(QuantType::from_gguf_type_id(8), Some(QuantType::Q8_0));
141        assert_eq!(QuantType::from_gguf_type_id(12), Some(QuantType::Q4_K_M));
142        assert_eq!(QuantType::from_gguf_type_id(14), Some(QuantType::Q6_K));
143        assert_eq!(QuantType::from_gguf_type_id(30), Some(QuantType::BF16));
144        assert_eq!(QuantType::from_gguf_type_id(999), None);
145
146        assert_eq!(QuantType::F32.storage_bytes("w", 16).unwrap(), 64);
147        assert_eq!(QuantType::F16.storage_bytes("w", 16).unwrap(), 32);
148        assert_eq!(QuantType::BF16.storage_bytes("w", 16).unwrap(), 32);
149        assert_eq!(QuantType::Q4_0.storage_bytes("w", 64).unwrap(), 36);
150        assert_eq!(QuantType::Q8_0.storage_bytes("w", 64).unwrap(), 68);
151        assert_eq!(QuantType::Q4_K_M.storage_bytes("w", 512).unwrap(), 288);
152        assert_eq!(QuantType::Q6_K.storage_bytes("w", 256).unwrap(), 210);
153
154        // Unaligned block size must fail
155        let unaligned = QuantType::Q4_K_M.storage_bytes("blk.0.wq", 300);
156        assert!(matches!(unaligned, Err(K3Error::TensorBoundsError { .. })));
157    }

159    #[test]
160    fn tensor_view_bounds_calculation_verifies_valid_and_invalid_ranges() {
161        let backing = vec![0xAB_u8; 352];
162
163        // Valid Q4_K_M tensor: 256 * 2 = 512 elements -> 2 blocks * 144 = 288 bytes at offset 0
164        let view_q4 = TensorView::from_bytes(
165            "blk.0.attn_q.weight",
166            QuantType::Q4_K_M,
167            &[256, 2],
168            0,
169            &backing,
170        )
171        .expect("valid Q4_K_M bounds");
172        assert_eq!(view_q4.name(), "blk.0.attn_q.weight");
173        assert_eq!(view_q4.quant(), QuantType::Q4_K_M);
174        assert_eq!(view_q4.shape(), &[256, 2]);
175        assert_eq!(view_q4.num_elements(), 512);
176        assert_eq!(view_q4.offset(), 0);
177        assert_eq!(view_q4.byte_len(), 288);
178        assert_eq!(view_q4.as_bytes().as_ptr(), backing.as_ptr());
179
180        // Valid F32 tensor: 16 elements -> 64 bytes at offset 288 (288 + 64 == 352)
181        let view_f32 =
182            TensorView::from_bytes("output_norm.weight", QuantType::F32, &[16], 288, &backing)
183                .expect("valid F32 bounds at end of buffer");
184        assert_eq!(view_f32.offset(), 288);
185        assert_eq!(view_f32.byte_len(), 64);
186
187        // Out-of-bounds by 1 byte
188        let oob =
189            TensorView::from_bytes("output_norm.weight", QuantType::F32, &[16], 289, &backing);
190        assert_eq!(
191            oob,
192            Err(K3Error::TensorBoundsError {
193                tensor_name: "output_norm.weight".to_owned(),
194                offset: 289,
195                byte_len: 64,
196                buffer_len: 352,
197                reason: "tensor range [289..353] exceeds backing buffer length 352".to_owned(),
198            })
199        );
200
201        // Empty shape rejected
202        let empty_shape = TensorView::from_bytes("empty", QuantType::F32, &[], 0, &backing);
203        assert!(matches!(
204            empty_shape,
205            Err(K3Error::TensorBoundsError { .. })
206        ));
207
208        // Zero dimension rejected
209        let zero_dim = TensorView::from_bytes("zero", QuantType::F32, &[256, 0], 0, &backing);
210        assert!(matches!(zero_dim, Err(K3Error::TensorBoundsError { .. })));
211
212        // Offset + byte_len overflow rejected
213        let overflow_offset =
214            TensorView::from_bytes("overflow", QuantType::F32, &[16], usize::MAX - 10, &backing);
215        assert!(matches!(
216            overflow_offset,
217            Err(K3Error::TensorBoundsError { .. })
218        ));
219    }

281    #[test]
282    fn single_prompt_execution_context_and_arch_recipe_lifecycle() {
283        let backing = vec![0u8; 352];
284        let weights = DummyWeights { buffer: &backing };
285        let recipe = DummyRecipe;
286
287        assert_eq!(recipe.architecture_id(), "synthetic");
288        assert!(recipe.supports_quant(QuantType::Q4_K_M));
289        assert!(!recipe.supports_quant(QuantType::Q8_0));
290        assert!(weights.contains_tensor("blk.0.attn_q.weight"));
291        assert!(!weights.contains_tensor("nonexistent.weight"));
292        assert_eq!(weights.tensor_count(), 2);
293        assert_eq!(weights.total_weight_bytes(), 352);
294
295        let prompt = [10_u32, 20, 30];
296        let mut logits = [0.0_f32; 8];
297        let mut kv_cache = [0.0_f32; 64];
298
299        let mut ctx = ExecutionContext::new(&prompt, 4, &mut logits, &mut kv_cache)
300            .expect("prompt fits in n_ctx=4");
301        recipe
302            .prefill_step(&mut ctx, &weights)
303            .expect("prefill succeeds");
304        assert_eq!(ctx.position(), 3);
305        assert!((ctx.logits()[0] - 1.0).abs() < f32::EPSILON);
306
307        // Decode 1 token (reaching n_ctx = 4)
308        ctx.prepare_decode(5).expect("4th token fits in n_ctx=4");
309        recipe
310            .decode_step(&mut ctx, &weights)
311            .expect("decode succeeds");
312        assert_eq!(ctx.position(), 4);
313        assert!((ctx.logits()[5] - 2.0).abs() < f32::EPSILON);
314
315        // Attempting a 5th token must fail with ContextLengthExceeded (ADR-027)
316        let overflow = ctx.prepare_decode(6);
317        assert_eq!(
318            overflow,
319            Err(K3Error::ContextLengthExceeded {
320                requested_tokens: 5,
321                max_context_tokens: 4,
322            })
323        );
324    }
