44/// RAII guard that automatically removes the Unix domain socket file upon drop.
45#[derive(Debug)]
46pub struct SocketFileGuard {
47    path: PathBuf,
48    active: bool,
49}
50
51impl SocketFileGuard {
52    /// Creates a new guard for the given socket file path.
53    #[must_use]
54    pub fn new(path: PathBuf) -> Self {
55        Self { path, active: true }
56    }
57
58    /// Disarms the guard so the socket file is retained on drop.
59    pub fn disarm(&mut self) {
60        self.active = false;
61    }
62}
63
64impl Drop for SocketFileGuard {
65    fn drop(&mut self) {
66        if self.active && self.path.exists() {
67            let _ = fs::remove_file(&self.path);
68        }
69    }
70}

72/// Prepares and binds a `tokio::net::UnixListener` after verifying and cleaning stale socket files.
73///
74/// # Errors
75/// Returns [`io::Error`] if an active daemon is already listening or binding fails.
76pub fn prepare_and_bind_socket(socket_path: &Path) -> io::Result<tokio::net::UnixListener> {
77    if let Some(parent) = socket_path.parent() {
78        if !parent.as_os_str().is_empty() && !parent.exists() {
79            fs::create_dir_all(parent)?;
80        }
81    }
82
83    if socket_path.exists() {
84        // Safe state verification (AGENTS.md Rule 1): probe if another daemon is alive
85        match UnixStream::connect(socket_path) {
86            Ok(_) => {
87                return Err(io::Error::new(
88                    io::ErrorKind::AddrInUse,
89                    format!(
90                        "Another k3-kv daemon is actively listening on {}",
91                        socket_path.display()
92                    ),
93                ));
94            }
95            Err(e) if e.kind() == io::ErrorKind::ConnectionRefused => {
96                // Stale socket file from previous unclean exit -> unlink it
97                let _ = fs::remove_file(socket_path);
98            }
99            Err(e) if e.kind() == io::ErrorKind::NotFound => {
100                // Socket file disappeared concurrently -> proceed
101            }
102            Err(_) => {
103                // Broken or unconnectable node -> attempt unlink
104                let _ = fs::remove_file(socket_path);
105            }
106        }
107    }
108
109    let listener = tokio::net::UnixListener::bind(socket_path)?;
110
111    // Set secure socket permissions (0600: read/write by owner only)
112    #[cfg(unix)]
113    {
114        let perms = fs::Permissions::from_mode(0o600);
115        let _ = fs::set_permissions(socket_path, perms);
116    }
117
118    Ok(listener)
119}

121/// Standalone Unix Domain Socket server managing KV cache arenas.
122#[derive(Debug)]
123pub struct KvServer {
124    config: ServerConfig,
125    holder: Arc<KvHolder>,
126    shutdown_notify: Arc<tokio::sync::Notify>,
127    is_shutdown: Arc<AtomicBool>,
128    active_clients: Arc<Mutex<HashMap<u64, UnixStream>>>,
129    next_client_id: Arc<AtomicU64>,
130}
131
132impl KvServer {
133    /// Creates a new [`KvServer`] with the specified configuration.
134    #[must_use]
135    pub fn new(config: ServerConfig) -> Self {
136        let holder = Arc::new(KvHolder::with_memlock_ceiling(config.max_memory_bytes));
137        Self::with_holder(config, holder)
138    }
139
140    /// Creates a new [`KvServer`] wrapping an existing [`KvHolder`].
141    #[must_use]
142    pub fn with_holder(config: ServerConfig, holder: Arc<KvHolder>) -> Self {
143        Self {
144            config,
145            holder,
146            shutdown_notify: Arc::new(tokio::sync::Notify::new()),
147            is_shutdown: Arc::new(AtomicBool::new(false)),
148            active_clients: Arc::new(Mutex::new(HashMap::new())),
149            next_client_id: Arc::new(AtomicU64::new(1)),
150        }
151    }
152
153    /// Creates a new [`KvServer`] bound to a given path with the specified holder.
154    ///
155    /// # Errors
156    /// Returns [`io::Error`] if path verification fails.
157    pub fn bind(socket_path: impl AsRef<Path>, holder: KvHolder) -> io::Result<Self> {
158        let path = socket_path.as_ref().to_path_buf();
159        let max_mem = holder.memlock_ceiling_bytes();
160        Ok(Self::with_holder(
161            ServerConfig {
162                socket_path: path,
163                max_memory_bytes: max_mem,
164            },
165            Arc::new(holder),
166        ))
167    }
168
169    /// Returns a reference to the inner memory holder.
170    #[must_use]
171    pub fn holder(&self) -> &Arc<KvHolder> {
172        &self.holder
173    }
174
175    /// Returns the configured socket path.
176    #[must_use]
177    pub fn socket_path(&self) -> &Path {
178        &self.config.socket_path
179    }
180
181    /// Returns a shutdown notification handle.
182    #[must_use]
183    pub fn shutdown_handle(&self) -> Arc<tokio::sync::Notify> {
184        Arc::clone(&self.shutdown_notify)
185    }
186
187    /// Returns the number of currently active client connections.
188    #[must_use]
189    pub fn active_client_count(&self) -> usize {
190        self.active_clients.lock().map_or(0, |c| c.len())
191    }
192
193    /// Triggers graceful server shutdown, stopping the accept loop and terminating active client streams.
194    pub fn shutdown(&self) {
195        self.is_shutdown.store(true, Ordering::SeqCst);
196        self.shutdown_notify.notify_waiters();
197        self.shutdown_notify.notify_one();
198        if let Ok(mut clients) = self.active_clients.lock() {
199            for stream in clients.values() {
200                let _ = stream.shutdown(std::net::Shutdown::Both);
201            }
202            clients.clear();
203        }
204    }
205
206    /// Runs the server accept loop until shutdown notification is received.
207    ///
208    /// # Errors
209    /// Returns [`io::Error`] on fatal socket or listener failure.
210    pub async fn run(&self) -> io::Result<()> {
211        let listener = prepare_and_bind_socket(&self.config.socket_path)?;
212        let _guard = SocketFileGuard::new(self.config.socket_path.clone());
213
214        while !self.is_shutdown.load(Ordering::SeqCst) {
215            tokio::select! {
216                accept_res = listener.accept() => {
217                    match accept_res {
218                        Ok((stream, _addr)) => {
219                            if self.is_shutdown.load(Ordering::SeqCst) {
220                                break;
221                            }
222                            let holder = Arc::clone(&self.holder);
223                            match stream.into_std() {
224                                Ok(std_stream) => {
225                                    if let Err(e) = std_stream.set_nonblocking(false) {
226                                        eprintln!("[k3-kv] Failed to set stream blocking mode: {e}");
227                                        continue;
228                                    }
229
230                                    let client_id = self.next_client_id.fetch_add(1, Ordering::Relaxed);
231                                    let active_clients = Arc::clone(&self.active_clients);
232
233                                    // Track connection for shutdown signaling
234                                    if let Ok(cloned) = std_stream.try_clone() {
235                                        if let Ok(mut clients) = active_clients.lock() {
236                                            if self.is_shutdown.load(Ordering::SeqCst) {
237                                                let _ = cloned.shutdown(std::net::Shutdown::Both);
238                                                break;
239                                            }
240                                            clients.insert(client_id, cloned);
241                                        }
242                                    }
243
244                                    tokio::task::spawn_blocking(move || {
245                                        struct ClientGuard {
246                                            id: u64,
247                                            clients: Arc<Mutex<HashMap<u64, UnixStream>>>,
248                                        }
249                                        impl Drop for ClientGuard {
250                                            fn drop(&mut self) {
251                                                if let Ok(mut map) = self.clients.lock() {
252                                                    map.remove(&self.id);
253                                                }
254                                            }
255                                        }
256                                        let _guard = ClientGuard {
257                                            id: client_id,
258                                            clients: active_clients,
259                                        };
260                                        let _ = handle_connection(&std_stream, &holder);
261                                    });
262                                }
263                                Err(e) => {
264                                    eprintln!("[k3-kv] Failed to convert tokio stream to std: {e}");
265                                }
266                            }
267                        }
268                        Err(e) => {
269                            eprintln!("[k3-kv] Listener accept error: {e}");
270                        }
271                    }
272                }
273                () = self.shutdown_notify.notified() => {
274                    break;
275                }
276            }
277        }
278
279        // Cleanly terminate active client connections so blocking worker threads unblock immediately
280        self.shutdown();
281        Ok(())
282    }
283}

285/// Internal dispatch errors for protocol rejection.
286#[derive(Debug)]
287pub enum DispatchError {
288    /// Worker command or unsupported functionality.
289    CapabilityMismatch(String),
290    /// Frame corruption or decoding error.
291    ProtocolError(String),
292}
293
294/// Handles all requests from a single connected client session sequentially.
295///
296/// # Errors
297/// Returns [`io::Error`] on fatal I/O failure.
298pub fn handle_connection(stream: &UnixStream, holder: &Arc<KvHolder>) -> io::Result<()> {
299    let fory = global_fory();
300
301    loop {
302        let (header, payload, opt_fd) = match recv_frame(stream.as_raw_fd()) {
303            Ok(res) => res,
304            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
305                // Client closed socket cleanly
306                break;
307            }
308            Err(e) => {
309                // Invalid magic, payload > 16 MiB, or socket read failure
310                return Err(e);
311            }
312        };
313
314        // Drop any unexpected client-supplied FD safely
315        drop(opt_fd);
316
317        let dispatch_res = dispatch_single_request(header.type_id, &payload, holder, fory);
318        match dispatch_res {
319            Ok((resp_msg, pass_fd)) => {
320                send_msg_frame(stream, &resp_msg, pass_fd)?;
321            }
322            Err(DispatchError::CapabilityMismatch(msg)) => {
323                let err_resp = IpcMessage::AllocateArenaResponse(AllocateArenaResponse {
324                    arena_id: 0,
325                    arena_bytes: 0,
326                    status: StatusCode::CapabilityMismatch,
327                    error_message: Some(msg),
328                });
329                send_msg_frame(stream, &err_resp, None)?;
330            }
331            Err(DispatchError::ProtocolError(msg)) => {
332                let err_resp = IpcMessage::AllocateArenaResponse(AllocateArenaResponse {
333                    arena_id: 0,
334                    arena_bytes: 0,
335                    status: StatusCode::ProtocolError,
336                    error_message: Some(msg),
337                });
338                send_msg_frame(stream, &err_resp, None)?;
339            }
340        }
341    }
342
343    Ok(())
344}

346/// Routes an individual incoming message to the appropriate `KvHolder` action.
347///
348/// # Errors
349/// Returns [`DispatchError::ProtocolError`] if message deserialization fails, or
350/// [`DispatchError::CapabilityMismatch`] for worker commands or unknown type IDs.
351pub fn dispatch_single_request(
352    type_id: u16,
353    payload: &[u8],
354    holder: &Arc<KvHolder>,
355    fory: &fory::Fory,
356) -> Result<(IpcMessage, Option<std::os::fd::RawFd>), DispatchError> {
357    match type_id {
358        // 1. AllocateArenaRequest (Type ID: 101)
359        101 => {
360            let req: AllocateArenaRequest = fory.deserialize(payload).map_err(|e| {
361                DispatchError::ProtocolError(format!("Corrupt AllocateArenaRequest: {e}"))
362            })?;
363
364            let resp = match holder.allocate_arena(
365                &req.model_id,
366                req.arena_bytes,
367                req.n_ctx,
368                req.max_blocks,
369            ) {
370                Ok(res) => AllocateArenaResponse {
371                    arena_id: res.arena_id,
372                    arena_bytes: res.total_bytes,
373                    status: StatusCode::Success,
374                    error_message: None,
375                },
376                Err(err) => AllocateArenaResponse {
377                    arena_id: 0,
378                    arena_bytes: 0,
379                    status: err.status_code(),
380                    error_message: Some(err.to_string()),
381                },
382            };
383
384            Ok((IpcMessage::AllocateArenaResponse(resp), None))
385        }
386
387        // 2. GetArenaFdRequest (Type ID: 103)
388        103 => {
389            let req: GetArenaFdRequest = fory.deserialize(payload).map_err(|e| {
390                DispatchError::ProtocolError(format!("Corrupt GetArenaFdRequest: {e}"))
391            })?;
392
393            if let Some(arena) = holder.get_arena(req.arena_id) {
394                let resp = GetArenaFdResponse {
395                    arena_id: req.arena_id,
396                    status: StatusCode::Success,
397                    error_message: None,
398                };
399                let raw_fd = arena.as_raw_fd();
400                Ok((IpcMessage::GetArenaFdResponse(resp), Some(raw_fd)))
401            } else {
402                let resp = GetArenaFdResponse {
403                    arena_id: req.arena_id,
404                    status: StatusCode::ArenaNotFound,
405                    error_message: Some(format!("Arena {} not found", req.arena_id)),
406                };
407                Ok((IpcMessage::GetArenaFdResponse(resp), None))
408            }
409        }
410
411        // 3. DeallocateArenaRequest (Type ID: 105)
412        105 => {
413            let req: DeallocateArenaRequest = fory.deserialize(payload).map_err(|e| {
414                DispatchError::ProtocolError(format!("Corrupt DeallocateArenaRequest: {e}"))
415            })?;
416
417            let resp = match holder.deallocate(req.arena_id) {
418                Ok(_) => DeallocateArenaResponse {
419                    arena_id: req.arena_id,
420                    status: StatusCode::Success,
421                    error_message: None,
422                },
423                Err(err) => DeallocateArenaResponse {
424                    arena_id: req.arena_id,
425                    status: err.status_code(),
426                    error_message: Some(err.to_string()),
427                },
428            };
429
430            Ok((IpcMessage::DeallocateArenaResponse(resp), None))
431        }
432
433        // 4. PingRequest (Type ID: 107)
434        107 => {
435            let req: PingRequest = fory
436                .deserialize(payload)
437                .map_err(|e| DispatchError::ProtocolError(format!("Corrupt PingRequest: {e}")))?;
438
439            let resp = PingResponse {
440                timestamp_ns: req.timestamp_ns,
441            };
442
443            Ok((IpcMessage::PingResponse(resp), None))
444        }
445
446        // 5. Compute Worker Commands sent to KV holder -> CapabilityMismatch (Exit code 10)
447        201..=206 => Err(DispatchError::CapabilityMismatch(format!(
448            "k3-kv is a context memory holder daemon and does not execute worker commands (type {type_id})"
449        ))),
450
451        // 6. Unknown Type ID
452        unknown => Err(DispatchError::CapabilityMismatch(format!(
453            "Unrecognized IPC message type ID: {unknown}"
454        ))),
455    }
456}

748    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
749    async fn test_concurrent_client_connections() {
750        let tmp_dir = std::env::temp_dir().join(format!("k3_kv_con_{}", std::process::id()));
751        fs::create_dir_all(&tmp_dir).unwrap();
752        let sock_path = tmp_dir.join("k3-kv.sock");
753
754        let config = ServerConfig {
755            socket_path: sock_path.clone(),
756            max_memory_bytes: 1024 * 1024 * 1024,
757        };
758        let server = Arc::new(KvServer::new(config));
759        let server_clone = Arc::clone(&server);
760
761        let handle = tokio::spawn(async move {
762            server_clone.run().await.unwrap();
763        });
764
765        // Wait until server is accepting
766        for _ in 0..50 {
767            if UnixStream::connect(&sock_path).is_ok() {
768                break;
769            }
770            tokio::time::sleep(std::time::Duration::from_millis(20)).await;
771        }
772
773        // Spawn 6 concurrent clients
774        let mut tasks = Vec::new();
775        for i in 0..6 {
776            let path = sock_path.clone();
777            tasks.push(tokio::task::spawn_blocking(move || {
778                let client = UnixStream::connect(&path).unwrap();
779                let req = IpcMessage::AllocateArenaRequest(AllocateArenaRequest {
780                    model_id: format!("model-{i}"),
781                    arena_bytes: 65536,
782                    n_ctx: 64,
783                    max_blocks: 4,
784                });
785                send_msg_frame(&client, &req, None).unwrap();
786                let (resp, _) = recv_msg_frame(&client).unwrap();
787                let arena_id = match resp {
788                    IpcMessage::AllocateArenaResponse(r) => {
789                        assert_eq!(r.status, StatusCode::Success);
790                        r.arena_id
791                    }
792                    _ => panic!(),
793                };
794
795                let get_fd = IpcMessage::GetArenaFdRequest(GetArenaFdRequest { arena_id });
796                send_msg_frame(&client, &get_fd, None).unwrap();
797                let (_, opt_fd) = recv_msg_frame(&client).unwrap();
798                assert!(opt_fd.is_some());
799
800                let dealloc =
801                    IpcMessage::DeallocateArenaRequest(DeallocateArenaRequest { arena_id });
802                send_msg_frame(&client, &dealloc, None).unwrap();
803                let (resp_d, _) = recv_msg_frame(&client).unwrap();
804                match resp_d {
805                    IpcMessage::DeallocateArenaResponse(r) => {
806                        assert_eq!(r.status, StatusCode::Success);
807                    }
808                    _ => panic!(),
809                }
810                drop(client);
811            }));
812        }
813
814        for t in tasks {
815            t.await.unwrap();
816        }
817
818        server.shutdown();
819        handle.await.unwrap();
820        let _ = fs::remove_dir_all(&tmp_dir);
821    }

823    #[tokio::test(flavor = "multi_thread", worker_threads = 2)]
824    async fn test_server_shutdown_unblocks_connected_client_task() {
825        use std::io::Read;
826
827        let tmp_dir = std::env::temp_dir().join(format!("k3_kv_unblock_{}", std::process::id()));
828        fs::create_dir_all(&tmp_dir).unwrap();
829        let sock_path = tmp_dir.join("k3-kv.sock");
830
831        let config = ServerConfig {
832            socket_path: sock_path.clone(),
833            max_memory_bytes: 1024 * 1024 * 1024,
834        };
835        let server = Arc::new(KvServer::new(config));
836        let server_clone = Arc::clone(&server);
837
838        let handle = tokio::spawn(async move {
839            server_clone.run().await.unwrap();
840        });
841
842        // Wait until server is accepting
843        let mut client_opt = None;
844        for _ in 0..50 {
845            if sock_path.exists() {
846                if let Ok(client) = UnixStream::connect(&sock_path) {
847                    client_opt = Some(client);
848                    break;
849                }
850            }
851            tokio::time::sleep(std::time::Duration::from_millis(20)).await;
852        }
853        let client = client_opt.expect("client must connect");
854
855        // Send a ping to verify connection is active and handled by worker thread
856        let ping = IpcMessage::PingRequest(PingRequest { timestamp_ns: 777 });
857        send_msg_frame(&client, &ping, None).unwrap();
858        let (resp, _) = recv_msg_frame(&client).unwrap();
859        assert_eq!(
860            resp,
861            IpcMessage::PingResponse(PingResponse { timestamp_ns: 777 })
862        );
863
864        assert_eq!(server.active_client_count(), 1);
865
866        // Keep client open and call server.shutdown()
867        server.shutdown();
868
869        // Server run() loop and blocking tasks must terminate promptly without hanging
870        tokio::time::timeout(std::time::Duration::from_secs(2), handle)
871            .await
872            .expect("server handle must exit within 2s on shutdown")
873            .unwrap();
874
875        // Reading from client should now report EOF or error because server shut it down
876        let mut buf = [0u8; 16];
877        let mut client_ref = &client;
878        let read_res = client_ref.read(&mut buf);
879        match read_res {
880            Ok(0) => {} // EOF
881            Err(e) if e.kind() == io::ErrorKind::ConnectionReset => {}
882            Err(e) if e.kind() == io::ErrorKind::BrokenPipe => {}
883            Err(e) if e.kind() == io::ErrorKind::NotConnected => {}
884            other => panic!("Expected EOF or connection error, got {other:?}"),
885        }
886
887        drop(client);
888        let _ = fs::remove_dir_all(&tmp_dir);
889    }
