1//! Secret Resolution and Credential Isolation (Feature 30, ADR-041, AGENTS.md §3).
2//!
3//! Enforces fail-closed resolution for secrets specified either as literal tokens
4//! or file paths (e.g., mode 0600 `/etc/k3/credentials/<source>.env`).
5//! Sensitive credential memory is zeroized via `.wipe()` using authentic volatile
6//! memory writes (`std::ptr::write_volatile`) upon completion.
7
8use std::path::Path;
9use std::sync::atomic::{compiler_fence, Ordering};
10
11use crate::ingestion::adapter::{IngestionError, SourceCredentials};
12
13/// Volatiles zeroizes a byte slice using `std::ptr::write_volatile` and a
14/// sequential consistency compiler fence to defeat LLVM Dead Store Elimination (DSE).
15#[inline]
16pub fn volatile_zeroize_bytes(buf: &mut [u8]) {
17    let len = buf.len();
18    let ptr = buf.as_mut_ptr();
19    for i in 0..len {
20        // SAFETY: `ptr` points to `len` contiguous initialized bytes of `buf`.
21        // `i < len` ensures `ptr.add(i)` is within allocated buffer bounds and properly aligned.
22        unsafe {
23            std::ptr::write_volatile(ptr.add(i), 0u8);
24        }
25    }
26    compiler_fence(Ordering::SeqCst);
27}

29/// Volatiles zeroizes the heap allocation of a `String` up to its allocated capacity
30/// using `std::ptr::write_volatile`, then clears its length.
31///
32/// Defeats compiler optimization (LLVM Dead Store Elimination) when discarding sensitive
33/// tokens or passwords before deallocation.
34#[inline]
35pub fn volatile_zeroize_string(s: &mut String) {
36    // SAFETY: We safely access the underlying vector to overwrite all allocated bytes.
37    // 1. `as_mut_vec()` provides mutable access to the backing `Vec<u8>`.
38    // 2. `as_mut_ptr()` yields a pointer to the allocated allocation.
39    // 3. For any non-empty allocation (`cap > 0`), the pointer is valid for writes
40    //    up to `cap` bytes, and `u8` requires 1-byte alignment (always aligned).
41    // 4. Overwriting bytes with `0u8` (ASCII NUL) leaves valid UTF-8 characters.
42    // 5. Volatile writes cannot be reordered or eliminated by compiler DSE optimizations.
43    unsafe {
44        let vec = s.as_mut_vec();
45        let ptr = vec.as_mut_ptr();
46        let cap = vec.capacity();
47        if cap > 0 && !ptr.is_null() {
48            for i in 0..cap {
49                std::ptr::write_volatile(ptr.add(i), 0u8);
50            }
51        }
52    }
53    compiler_fence(Ordering::SeqCst);
54    s.clear();
55}

57/// Resolves a secret token from either a file path or a literal value.
58///
59/// # Invariants (AGENTS.md §3)
60/// - If `val` appears to be a file path (starts with `/`, `./`, `../`, equals `.` or `..`,
61///   or contains path separators `/` or `\\`), it is treated strictly as a file path.
62/// - If `val` points to an existing directory (even without path separators), it is treated
63///   as an unreadable credential path and **FAILS CLOSED** returning `None`.
64/// - If the file cannot be read, does not exist, is a directory, or is empty,
65///   this function **FAILS CLOSED** and returns `None`.
66/// - **NEVER** falls back to using the file path string itself as the secret token.
67#[must_use]
68pub fn resolve_secret_value(val: &str) -> Option<String> {
69    let trimmed = val.trim();
70    if trimmed.is_empty() {
71        return None;
72    }
73
74    let is_path = trimmed == "."
75        || trimmed == ".."
76        || trimmed.starts_with('/')
77        || trimmed.starts_with("./")
78        || trimmed.starts_with("../")
79        || trimmed.contains('/')
80        || trimmed.contains('\\');
81
82    let path = Path::new(trimmed);
83    if path.is_dir() {
84        // Any path that resolves to an existing directory must FAIL CLOSED!
85        return None;
86    }
87
88    if path.is_file() {
89        match std::fs::read_to_string(path) {
90            Ok(content) => {
91                let secret = content.trim().to_string();
92                if secret.is_empty() {
93                    None
94                } else {
95                    Some(secret)
96                }
97            }
98            Err(_) => None, // FAIL CLOSED!
99        }
100    } else if is_path {
101        // Path does not exist, is a directory, or is not a readable file -> FAIL CLOSED!
102        None
103    } else {
104        // Literal non-path string
105        Some(trimmed.to_string())
106    }
107}

109impl SourceCredentials {
110    /// Constructs `SourceCredentials` from a raw secret string or path, failing closed.
111    ///
112    /// # Invariants (AGENTS.md §3)
113    /// If `val` appears to be a path, attempts to read and trim the file. If reading fails
114    /// (file does not exist, is empty, or permission denied), returns
115    /// [`IngestionError::CredentialResolutionFailed`]. It never returns the path as a token.
116    ///
117    /// # Errors
118    /// Returns [`IngestionError::CredentialResolutionFailed`] if path resolution fails.
119    pub fn from_secret_string(val: &str) -> Result<Self, IngestionError> {
120        let trimmed = val.trim();
121        if trimmed.is_empty() {
122            return Ok(Self::Anonymous);
123        }
124
125        match resolve_secret_value(trimmed) {
126            Some(token) => Ok(Self::Token(token)),
127            None => Err(IngestionError::CredentialResolutionFailed(format!(
128                "failed to resolve secret from '{trimmed}': unreadable, empty, or missing file; failing closed"
129            ))),
130        }
131    }
132
133    /// Constructs `SourceCredentials` from an environment variable, failing closed.
134    ///
135    /// # Errors
136    /// Returns [`IngestionError::CredentialResolutionFailed`] if the variable is set but resolution fails.
137    pub fn from_env(var_name: &str) -> Result<Self, IngestionError> {
138        match std::env::var(var_name) {
139            Ok(val) => Self::from_secret_string(&val),
140            Err(_) => Ok(Self::Anonymous),
141        }
142    }
143
144    /// Securely zeroizes in-memory token and secret buffers using `std::ptr::write_volatile`
145    /// to defeat LLVM Dead Store Elimination (DSE), and resets to [`SourceCredentials::Anonymous`].
146    pub fn wipe(&mut self) {
147        match self {
148            Self::Token(s) | Self::BearerToken(s) => {
149                volatile_zeroize_string(s);
150            }
151            Self::BasicAuth {
152                username,
153                password_or_token,
154            } => {
155                volatile_zeroize_string(username);
156                volatile_zeroize_string(password_or_token);
157            }
158            _ => {}
159        }
160        *self = Self::Anonymous;
161    }
162}
