1//! Derived Weight Layout Caching & Stale Recipe Detection (US-21, ADR-037, ADR-038).
2//!
3//! Caches hardware-repacked (Cluster 1 A100 `8x16` `i8` IME tiles) or format-converted weights
4//! as derived blobs keyed by `(source_digest, recipe_version, target_layout)`. Enforces
5//! operator-controlled retention policies, memory admission accounting, and auto-rebuilding
6//! of stale derivations upon runtime upgrades.
7

21/// Operator-configurable retention policy for original source weights after a derived layout is generated (US-21 Scenario 2).
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, Default)]
23#[serde(rename_all = "snake_case")]
24pub enum RetentionPolicy {
25    /// Retains original source weights indefinitely in storage alongside derived layouts (default).
26    #[default]
27    Keep,
28    /// Immediately marks original source weights for reclamation once the derived layout is promoted.
29    PurgeImmediate,
30    /// Retains original source weights for an operator-configured grace period before marking for reclamation.
31    PurgeAfterDelay {
32        /// Grace period in seconds (e.g. 86400 for 24 hours).
33        delay_secs: u64,
34    },
35}
36
37impl RetentionPolicy {
38    /// Returns the canonical policy string identifier.
39    #[must_use]
40    pub const fn as_str(&self) -> &'static str {
41        match self {
42            Self::Keep => "keep",
43            Self::PurgeImmediate => "purge_immediate",
44            Self::PurgeAfterDelay { .. } => "purge_after_delay",
45        }
46    }
47}

49/// Outcome of derived weight resolution.
50#[derive(Debug, Clone, PartialEq, Eq)]
51pub enum DerivedResolution {
52    /// The derived weight layout was already cached on disk and in `DuckDB` catalog (US-21 Scenario 1).
53    Cached {
54        /// SHA-256 digest of derived layout blob.
55        derived_digest: String,
56        /// Filesystem path to cached derived blob.
57        derived_path: PathBuf,
58    },
59    /// A new derived layout was freshly converted and registered.
60    Created {
61        /// SHA-256 digest of derived layout blob.
62        derived_digest: String,
63        /// Filesystem path to newly created derived blob.
64        derived_path: PathBuf,
65    },
66}
67
68/// Outcome of evaluating potentially stale derived layouts upon breaking runtime upgrades (US-21 Scenario 3).
69#[derive(Debug, Clone, PartialEq, Eq)]
70pub enum StaleRecipeOutcome {
71    /// Layout was already valid and up-to-date with active recipe version.
72    UpToDate { derived_digest: String },
73    /// Layout was stale, but original source weights were present and layout was auto-rebuilt.
74    Rebuilt {
75        new_derived_digest: String,
76        new_derived_path: PathBuf,
77    },
78    /// Layout was stale and original source weights were purged; requires operator advisory.
79    SourcePurgedAdvisory {
80        source_digest: String,
81        stale_recipe_version: String,
82        active_recipe_version: String,
83        advisory_message: String,
84    },
85}

87/// Enforces memory admission accounting for simultaneously loaded weight layouts (US-21, ADR-037).
88///
89/// Refuses load with [`K3Error::InsufficientLockableMemory`] if total required locked bytes
90/// exceed available RAM or `RLIMIT_MEMLOCK` bound.
91///
92/// # Errors
93/// Returns [`K3Error::InsufficientLockableMemory`] on ceiling violation.
94pub fn check_derived_memory_admission(
95    current_locked_bytes: u64,
96    additional_layout_bytes: u64,
97    kv_arena_bytes: u64,
98    available_ram_bytes: u64,
99    rlimit_memlock_bytes: u64,
100) -> Result<(), K3Error> {
101    let total_required = current_locked_bytes
102        .saturating_add(additional_layout_bytes)
103        .saturating_add(kv_arena_bytes);
104
105    if (rlimit_memlock_bytes > 0 && total_required > rlimit_memlock_bytes)
106        || (available_ram_bytes > 0 && total_required > available_ram_bytes)
107    {
108        return Err(K3Error::InsufficientLockableMemory {
109            required_bytes: total_required,
110            weight_bytes: current_locked_bytes.saturating_add(additional_layout_bytes),
111            kv_bytes: kv_arena_bytes,
112            available_bytes: available_ram_bytes,
113            rlimit_memlock_bytes,
114        });
115    }
116
117    Ok(())
118}

120/// Manager for derived weight layout caching, retention lifecycle, and stale recipe detection.
121#[derive(Debug, Clone)]
122pub struct DerivedBlobManager {
123    store: Arc<ContentAddressedStore>,
124    catalog: Arc<dyn CasCatalog>,
125}
126
127impl DerivedBlobManager {
128    /// Constructs a new derived blob manager bound to a content-addressed store.
129    #[must_use]
130    pub fn new(store: Arc<ContentAddressedStore>) -> Self {
131        let catalog = store.catalog().clone();
132        Self { store, catalog }
133    }
134
135    /// Resolves or builds a hardware-repacked or format-converted derived blob (US-21 Scenario 1 & 2).
136    ///
137    /// Keyed by `(source_digest, recipe_version, target_layout)`.
138    ///
139    /// If already cached and valid, directly returns [`DerivedResolution::Cached`].
140    /// Otherwise, invokes `converter` to populate a staging file, promotes it to `blobs/sha256/<derived_digest>`,
141    /// registers in `DuckDB`, and applies the operator's configured `retention` policy to the source weights.
142    ///
143    /// # Errors
144    /// Returns [`K3Error`] on conversion failure, catalog error, or missing source weights.
145    #[allow(clippy::too_many_lines)]
146    pub fn get_or_create_derived_layout<F>(
147        &self,
148        source_digest: &str,
149        recipe_version: &str,
150        target_layout: &str,
151        retention: RetentionPolicy,
152        converter: F,
153    ) -> Result<DerivedResolution, K3Error>
154    where
155        F: FnOnce(&Path, &Path) -> Result<u64, K3Error>,
156    {
157        super::cas::validate_sha256_digest(source_digest)?;
158        let source_clean = source_digest.to_owned();
159
160        // 1. Check if valid derived blob already exists in DuckDB catalog and filesystem (Scenario 1)
161        if let Ok(Some(existing)) =
162            self.catalog
163                .get_derived_blob(source_digest, recipe_version, target_layout)
164        {
165            if existing.status == "valid" {
166                let derived_path = self.store.blobs_dir().join(&existing.derived_digest);
167                if derived_path.exists() {
168                    // Update access timestamp
169                    let now = SystemTime::now()
170                        .duration_since(UNIX_EPOCH)
171                        .unwrap_or_default()
172                        .as_secs();
173                    let _ = self
174                        .catalog
175                        .update_blob_access(&existing.derived_digest, now);
176
177                    return Ok(DerivedResolution::Cached {
178                        derived_digest: existing.derived_digest,
179                        derived_path,
180                    });
181                }
182            }
183        }
184
185        // 2. Validate that original source weights blob exists
186        super::cas::validate_sha256_digest(source_digest)?;
187        let blobs_dir = self.store.blobs_dir();
188        let source_path = blobs_dir.join(source_digest);
189        if !source_path.exists() {
190            return Err(K3Error::CapabilityMismatch {
191                architecture: "derived_weights".to_owned(),
192                reason: format!(
193                    "original source weights blob '{}' not found in store; cannot derive layout",
194                    source_path.display()
195                ),
196            });
197        }
198        let canon_blobs = fs::canonicalize(&blobs_dir).unwrap_or(blobs_dir);
199        let canon_source = fs::canonicalize(&source_path).unwrap_or_else(|_| source_path.clone());
200        if !canon_source.starts_with(&canon_blobs) {
201            return Err(K3Error::CapabilityMismatch {
202                architecture: "derived_weights".to_owned(),
203                reason: format!("path traversal detected in source_digest '{source_digest}'"),
204            });
205        }
206
207        // 3. Create staging file for conversion
208        let (staging_path, mut staging_file) = self.store.create_staging_file()?;
209
210        // 4. Run conversion
211        let derived_bytes = match converter(&source_path, &staging_path) {
212            Ok(bytes) => bytes,
213            Err(e) => {
214                let _ = fs::remove_file(&staging_path);
215                return Err(e);
216            }
217        };
218
219        // Flush staging file
220        staging_file
221            .flush()
222            .map_err(|e| K3Error::CapabilityMismatch {
223                architecture: "derived_weights".to_owned(),
224                reason: format!("failed to flush staging derived weights: {e}"),
225            })?;
226        drop(staging_file);
227
228        // 5. Compute SHA-256 digest of converted derived weights
229        let derived_digest = Sha256::digest_file(&staging_path).map_err(|e| {
230            let _ = fs::remove_file(&staging_path);
231            K3Error::CapabilityMismatch {
232                architecture: "derived_weights".to_owned(),
233                reason: format!("failed to hash derived staging weights: {e}"),
234            }
235        })?;
236
237        // 6. Atomically promote staging file to blobs/sha256/<derived_digest>
238        let final_derived_path = self.store.blobs_dir().join(&derived_digest);
239        if final_derived_path.exists() {
240            let _ = fs::remove_file(&staging_path);
241        } else {
242            fs::rename(&staging_path, &final_derived_path).map_err(|e| {
243                let _ = fs::remove_file(&staging_path);
244                K3Error::CapabilityMismatch {
245                    architecture: "derived_weights".to_owned(),
246                    reason: format!(
247                        "failed to atomically promote derived weights to '{}': {e}",
248                        final_derived_path.display()
249                    ),
250                }
251            })?;
252
253            #[cfg(unix)]
254            {
255                use std::os::unix::fs::PermissionsExt;
256                let _ = fs::set_permissions(&final_derived_path, fs::Permissions::from_mode(0o444));
257            }
258        }
259
260        // 7. Register derived blob in DuckDB catalog
261        let now = SystemTime::now()
262            .duration_since(UNIX_EPOCH)
263            .unwrap_or_default()
264            .as_secs();
265
266        let blob_rec = ModelBlobRecord {
267            digest: derived_digest.clone(),
268            size_bytes: derived_bytes,
269            blob_type: "derived_weights".to_owned(),
270            created_at: now,
271            last_accessed_at: now,
272            ref_count: 1,
273        };
274        self.catalog
275            .save_blob(&blob_rec)
276            .map_err(|e| K3Error::CapabilityMismatch {
277                architecture: "derived_catalog".to_owned(),
278                reason: format!("failed to save derived blob record: {e}"),
279            })?;
280
281        let derived_rec = DerivedBlobRecord {
282            derived_digest: derived_digest.clone(),
283            source_digest: source_clean.clone(),
284            recipe_version: recipe_version.to_owned(),
285            target_layout: target_layout.to_owned(),
286            status: "valid".to_owned(),
287            created_at: now,
288        };
289        self.catalog
290            .save_derived_blob(&derived_rec)
291            .map_err(|e| K3Error::CapabilityMismatch {
292                architecture: "derived_catalog".to_owned(),
293                reason: format!("failed to save derived association: {e}"),
294            })?;
295
296        // Add reference linking derived blob to source
297        let ref_rec = BlobReferenceRecord {
298            ref_id: format!("derived:{derived_digest}"),
299            blob_digest: derived_digest.clone(),
300            ref_type: "derived".to_owned(),
301            ref_source: source_clean.clone(),
302            created_at: now,
303        };
304        let _ = self.catalog.add_blob_reference(&ref_rec);
305
306        // 8. Apply operator retention policy (Scenario 2)
307        match retention {
308            RetentionPolicy::Keep => {
309                // Keep source blob indefinitely; no action needed
310            }
311            RetentionPolicy::PurgeImmediate => {
312                // Mark source blob for immediate GC reclamation if unreferenced by other manifests
313                let purge_key = format!("purge_immediate:{source_clean}");
314                let _ = self.catalog.set_storage_config(&purge_key, "true");
315
316                // Release manifest and alias references from blob_references
317                if let Ok(refs) = self.catalog.list_blob_references(&source_clean) {
318                    for r in refs {
319                        if r.ref_type == "manifest" || r.ref_type == "alias" {
320                            let _ = self.catalog.remove_blob_reference(&r.ref_id);
321                        }
322                    }
323                }
324                let _ = self.catalog.decrement_blob_ref(&source_clean);
325            }
326            RetentionPolicy::PurgeAfterDelay { delay_secs } => {
327                let purge_timestamp = now.saturating_add(delay_secs);
328                let purge_key = format!("purge_after:{source_clean}");
329                let _ = self
330                    .catalog
331                    .set_storage_config(&purge_key, &purge_timestamp.to_string());
332            }
333        }
334
335        Ok(DerivedResolution::Created {
336            derived_digest,
337            derived_path: final_derived_path,
338        })
339    }
340
341    /// Evaluates existing derived weight layouts against the active runtime recipe version (US-21 Scenario 3).
342    ///
343    /// Upon breaking recipe upgrades:
344    /// - Flags outdated derivations as `status = 'stale'`.
345    /// - If original source weights are still present, automatically rebuilds the layout.
346    /// - If original source weights were purged, returns [`StaleRecipeOutcome::SourcePurgedAdvisory`]
347    ///   with an actionable operator advisory notice.
348    ///
349    /// # Errors
350    /// Returns [`K3Error`] on rebuild error or catalog failure.
351    pub fn detect_and_handle_stale_recipe<F>(
352        &self,
353        source_digest: &str,
354        active_recipe_version: &str,
355        target_layout: &str,
356        model_name: &str,
357        rebuilder: F,
358    ) -> Result<StaleRecipeOutcome, K3Error>
359    where
360        F: FnOnce(&Path, &Path) -> Result<u64, K3Error>,
361    {
362        super::cas::validate_sha256_digest(source_digest)?;
363        let source_clean = source_digest.to_owned();
364
365        // Query existing derived blobs for this source and layout
366        let all_derived = self
367            .catalog
368            .list_derived_for_source(source_digest)
369            .map_err(|e| K3Error::CapabilityMismatch {
370                architecture: "derived_stale".to_owned(),
371                reason: format!("failed to list derived blobs for source: {e}"),
372            })?;
373
374        let matching_layout: Vec<DerivedBlobRecord> = all_derived
375            .into_iter()
376            .filter(|d| d.target_layout == target_layout)
377            .collect();
378
379        // If a valid derivation exists for the active recipe version, we are up to date
380        if let Some(up_to_date) = matching_layout
381            .iter()
382            .find(|d| d.recipe_version == active_recipe_version && d.status == "valid")
383        {
384            return Ok(StaleRecipeOutcome::UpToDate {
385                derived_digest: up_to_date.derived_digest.clone(),
386            });
387        }
388
389        // Check if there is an outdated derivation
390        let Some(stale_rec) = matching_layout
391            .iter()
392            .find(|d| d.recipe_version != active_recipe_version)
393        else {
394            // No prior derivation at all: not stale, simply unbuilt
395            return Ok(StaleRecipeOutcome::UpToDate {
396                derived_digest: String::new(),
397            });
398        };
399
400        // Flag existing derived record as stale in DuckDB
401        let _ = self
402            .catalog
403            .update_derived_status(&stale_rec.derived_digest, "stale");
404
405        // Check whether the original source weights are still present on disk
406        let source_path = self.store.blobs_dir().join(&source_clean);
407        if !source_path.starts_with(self.store.blobs_dir()) {
408            return Err(K3Error::CapabilityMismatch {
409                architecture: "derived_stale".to_owned(),
410                reason: format!("path traversal detected in source_digest '{source_digest}'"),
411            });
412        }
413        if source_path.exists() {
414            // Auto-rebuild the derived blob for the new recipe version (US-21 Scenario 3)
415            let resolution = self.get_or_create_derived_layout(
416                &source_clean,
417                active_recipe_version,
418                target_layout,
419                RetentionPolicy::Keep,
420                rebuilder,
421            )?;
422
423            match resolution {
424                DerivedResolution::Cached {
425                    derived_digest,
426                    derived_path,
427                }
428                | DerivedResolution::Created {
429                    derived_digest,
430                    derived_path,
431                } => Ok(StaleRecipeOutcome::Rebuilt {
432                    new_derived_digest: derived_digest,
433                    new_derived_path: derived_path,
434                }),
435            }
436        } else {
437            // Original source weights were purged; prompt operator to re-acquire (US-21 Scenario 3)
438            let advisory = format!(
439                "Derived weight layout for model '{model_name}' (source: {source_clean}) is stale \
440                 (built for recipe version {}, current runtime is {active_recipe_version}), \
441                 and original source weights have been purged. Please re-acquire the source model.",
442                stale_rec.recipe_version
443            );
444
445            Ok(StaleRecipeOutcome::SourcePurgedAdvisory {
446                source_digest: source_clean,
447                stale_recipe_version: stale_rec.recipe_version.clone(),
448                active_recipe_version: active_recipe_version.to_owned(),
449                advisory_message: advisory,
450            })
451        }
452    }
