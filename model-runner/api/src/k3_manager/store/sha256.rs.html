23/// Streaming SHA-256 state context.
24#[derive(Debug, Clone)]
25pub struct Sha256 {
26    state: [u32; 8],
27    buffer: [u8; 64],
28    buf_len: usize,
29    total_len: u64,
30}
31
32impl Default for Sha256 {
33    fn default() -> Self {
34        Self::new()
35    }
36}
37
38impl Sha256 {
39    /// Constructs a new SHA-256 hashing context.
40    #[must_use]
41    pub const fn new() -> Self {
42        Self {
43            state: [
44                0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
45                0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19,
46            ],
47            buffer: [0u8; 64],
48            buf_len: 0,
49            total_len: 0,
50        }
51    }
52
53    /// Feeds data bytes into the SHA-256 state context.
54    pub fn update(&mut self, mut data: &[u8]) {
55        self.total_len = self.total_len.saturating_add(data.len() as u64);
56
57        if self.buf_len > 0 {
58            let needed = 64 - self.buf_len;
59            if data.len() < needed {
60                self.buffer[self.buf_len..self.buf_len + data.len()].copy_from_slice(data);
61                self.buf_len += data.len();
62                return;
63            }
64            self.buffer[self.buf_len..64].copy_from_slice(&data[..needed]);
65            let block = self.buffer;
66            self.process_block(&block);
67            self.buf_len = 0;
68            data = &data[needed..];
69        }
70
71        while data.len() >= 64 {
72            let mut block = [0u8; 64];
73            block.copy_from_slice(&data[..64]);
74            self.process_block(&block);
75            data = &data[64..];
76        }
77
78        if !data.is_empty() {
79            self.buffer[..data.len()].copy_from_slice(data);
80            self.buf_len = data.len();
81        }
82    }
83
84    /// Finalizes the hash calculation and returns the 32-byte digest array.
85    #[must_use]
86    pub fn finalize(mut self) -> [u8; 32] {
87        let bit_len = self.total_len.saturating_mul(8);
88        self.buffer[self.buf_len] = 0x80;
89        self.buf_len += 1;
90
91        if self.buf_len > 56 {
92            self.buffer[self.buf_len..64].fill(0);
93            let block = self.buffer;
94            self.process_block(&block);
95            self.buf_len = 0;
96        }
97
98        self.buffer[self.buf_len..56].fill(0);
99        self.buffer[56..64].copy_from_slice(&bit_len.to_be_bytes());
100        let block = self.buffer;
101        self.process_block(&block);
102
103        let mut out = [0u8; 32];
104        for (i, word) in self.state.iter().enumerate() {
105            out[i * 4..(i + 1) * 4].copy_from_slice(&word.to_be_bytes());
106        }
107        out
108    }

110    /// Finalizes the hash calculation and formats as a 64-character lowercase hex string.
111    #[must_use]
112    pub fn finalize_hex(self) -> String {
113        let digest = self.finalize();
114        hex_encode(&digest)
115    }
116
117    /// Computes the SHA-256 lowercase hex digest for the given byte slice.
118    #[must_use]
119    pub fn digest(data: &[u8]) -> String {
120        let mut hasher = Self::new();
121        hasher.update(data);
122        hasher.finalize_hex()
123    }
124
125    /// Computes the SHA-256 lowercase hex digest by reading a stream to completion.
126    ///
127    /// # Errors
128    /// Returns [`io::Error`] if reading from `reader` fails.
129    pub fn digest_reader<R: Read>(reader: &mut R) -> io::Result<String> {
130        let mut hasher = Self::new();
131        let mut buf = [0u8; 16384];
132        loop {
133            let n = reader.read(&mut buf)?;
134            if n == 0 {
135                break;
136            }
137            hasher.update(&buf[..n]);
138        }
139        Ok(hasher.finalize_hex())
140    }
141
142    /// Computes the SHA-256 lowercase hex digest of the file at `path`.
143    ///
144    /// # Errors
145    /// Returns [`io::Error`] if opening or reading the file fails.
146    pub fn digest_file(path: impl AsRef<Path>) -> io::Result<String> {
147        let mut file = File::open(path)?;
148        Self::digest_reader(&mut file)
149    }

151    fn process_block(&mut self, block: &[u8; 64]) {
152        let mut w = [0u32; 64];
153        for i in 0..16 {
154            w[i] = u32::from_be_bytes(
155                block[i * 4..(i + 1) * 4]
156                    .try_into()
157                    .expect("slice has 4 bytes"),
158            );
159        }
160        for i in 16..64 {
161            let s0 = w[i - 15].rotate_right(7) ^ w[i - 15].rotate_right(18) ^ (w[i - 15] >> 3);
162            let s1 = w[i - 2].rotate_right(17) ^ w[i - 2].rotate_right(19) ^ (w[i - 2] >> 10);
163            w[i] = w[i - 16]
164                .wrapping_add(s0)
165                .wrapping_add(w[i - 7])
166                .wrapping_add(s1);
167        }
168
169        let mut a = self.state[0];
170        let mut b = self.state[1];
171        let mut c = self.state[2];
172        let mut d = self.state[3];
173        let mut e = self.state[4];
174        let mut f = self.state[5];
175        let mut g = self.state[6];
176        let mut h = self.state[7];
177
178        for i in 0..64 {
179            let s1 = e.rotate_right(6) ^ e.rotate_right(11) ^ e.rotate_right(25);
180            let ch = (e & f) ^ ((!e) & g);
181            let temp1 = h
182                .wrapping_add(s1)
183                .wrapping_add(ch)
184                .wrapping_add(K[i])
185                .wrapping_add(w[i]);
186            let s0 = a.rotate_right(2) ^ a.rotate_right(13) ^ a.rotate_right(22);
187            let maj = (a & b) ^ (a & c) ^ (b & c);
188            let temp2 = s0.wrapping_add(maj);
189
190            h = g;
191            g = f;
192            f = e;
193            e = d.wrapping_add(temp1);
194            d = c;
195            c = b;
196            b = a;
197            a = temp1.wrapping_add(temp2);
198        }
199
200        self.state[0] = self.state[0].wrapping_add(a);
201        self.state[1] = self.state[1].wrapping_add(b);
202        self.state[2] = self.state[2].wrapping_add(c);
203        self.state[3] = self.state[3].wrapping_add(d);
204        self.state[4] = self.state[4].wrapping_add(e);
205        self.state[5] = self.state[5].wrapping_add(f);
206        self.state[6] = self.state[6].wrapping_add(g);
207        self.state[7] = self.state[7].wrapping_add(h);
208    }

211/// Formats a byte array as a lowercase hex string.
212#[must_use]
213pub fn hex_encode(bytes: &[u8]) -> String {
214    const HEX_CHARS: &[u8; 16] = b"0123456789abcdef";
215    let mut s = String::with_capacity(bytes.len() * 2);
216    for &b in bytes {
217        s.push(HEX_CHARS[(b >> 4) as usize] as char);
218        s.push(HEX_CHARS[(b & 0x0f) as usize] as char);
219    }
220    s
221}
