1//! Multi-model compute scheduler enforcing pass-boundary time-slicing,
2//! priority preemption, cost-weighted fair sharing, and dynamic idle core spillover.
3//!
4//! # Architecture & Guarantees
5//! 1. **Operator Tuning Immutability (US-08, ADR-010, ADR-032)**:
6//!    The operator's configured [`TuningProfile`] is the sole authority governing
7//!    cluster reservations, priority class, context cap, and core lending/borrowing flags.
8//!    Live pass costs are observed strictly to refine internal fair-sharing weights
9//!    and generate on-demand advisory trade-off recommendations without auto-mutation.
10//! 2. **Multi-Model Coexistence & Concurrency Cap (US-09, ADR-043)**:
11//!    Enforces `max_active_models = 3` cap at load time, rejecting excess with
12//!    [`ManagerError::ConcurrencyLimitExceeded`].
13//!    Disjoint core reservations execute in parallel (`(mask_a & mask_b) == 0`).
14//!    Overlapping reservations time-slice strictly at forward-pass boundaries.
15//!    Single-prompt exclusivity is guaranteed per model forward pass (no cross-prompt batching).
16//! 3. **Priority Preemption & Cost-Weighted Fair Queuing (US-10, ADR-010, ADR-014)**:
17//!    [`PriorityClass::Latency`] preempts [`PriorityClass::Throughput`] at the next forward-pass boundary.
18//!    Passes in the same priority class interleave using Weighted Fair Queuing (WFQ)
19//!    virtual runtime tracking (`V_i += cost_i / w_i`) driven by live measured pass costs.
20//!    Client-supplied priority overrides are rejected or stripped.
21//! 4. **Opt-In Dynamic Idle Core Spillover & Reclaim (US-11, ADR-030, ADR-041)**:
22//!    Requires dual opt-in (`lend_idle_cores == true` and `borrow_idle_cores == true`).
23//!    Idle models lend cores for a single pass at a time.
24//!    Cores are reclaimed instantly at the pass boundary when the owner receives traffic.
25//!    Pre-pinned worker threads execute tasks without runtime cpuset rewrites.
26//!    Dedicated control cores (`0x00C0`) and spare recovery core (`0x0020`) are strictly non-lendable.
27

47/// Operating state of an active model in the multi-model scheduler.
48#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
49pub enum ModelSchedulingState {
50    /// Model has no active pass and no pending requests; cores are lendable if opted in.
51    Idle,
52    /// Model has pending forward pass requests queued and is awaiting physical core availability.
53    Waiting,
54    /// Model is currently executing a forward pass on physical cores.
55    Running,
56    /// Model was idle with cores lent to a peer, but received incoming traffic; awaiting pass boundary to reclaim.
57    ReclaimPending,
58}

60/// Online execution cost statistics tracked per active model to drive WFQ and advisories.
61#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
62pub struct ModelExecutionMetrics {
63    /// Logical model identifier.
64    pub model_id: String,
65    /// Total forward passes completed.
66    pub total_passes: u32,
67    /// Total tokens processed across all forward passes.
68    pub total_tokens: u32,
69    /// Total prompt prefill tokens processed.
70    pub total_prefill_tokens: u32,
71    /// Total autoregressive decode tokens generated.
72    pub total_decode_tokens: u32,
73    /// Measured cost of the most recent forward pass in milliseconds.
74    pub last_pass_cost_ms: f64,
75    /// Running average forward pass execution cost in milliseconds.
76    pub avg_pass_cost_ms: f64,
77    /// Exponential moving average of prefill cost in ms per token.
78    pub avg_prefill_ms_per_tok: f64,
79    /// Exponential moving average of decode cost in ms per token.
80    pub avg_decode_ms_per_tok: f64,
81    /// Dynamic scheduling weight used for weighted fair queue interleaving.
82    pub dynamic_scheduling_weight: f64,
83    /// Cumulative virtual runtime in milliseconds for weighted fair queuing (`V_i += cost_i / w_i`).
84    pub virtual_runtime_ms: f64,
85}

87const EMA_ALPHA: f64 = 0.2;

89impl ModelExecutionMetrics {
90    /// Constructs a new [`ModelExecutionMetrics`] instance for `model_id`.
91    #[must_use]
92    pub fn new(model_id: impl Into<String>) -> Self {
93        Self {
94            model_id: model_id.into(),
95            total_passes: 0,
96            total_tokens: 0,
97            total_prefill_tokens: 0,
98            total_decode_tokens: 0,
99            last_pass_cost_ms: 0.0,
100            avg_pass_cost_ms: 0.0,
101            avg_prefill_ms_per_tok: 0.0,
102            avg_decode_ms_per_tok: 0.0,
103            dynamic_scheduling_weight: 1.0,
104            virtual_runtime_ms: 0.0,
105        }
106    }
107
108    /// Records completion of a forward pass, updating rolling statistics and advancing virtual runtime.
109    ///
110    /// Preserves operator profile immutability per ADR-010 and ADR-032.
111    pub fn record_pass(&mut self, is_prefill: bool, elapsed_ms: f64, token_count: u32) {
112        self.total_passes = self.total_passes.saturating_add(1);
113        self.total_tokens = self.total_tokens.saturating_add(token_count);
114        self.last_pass_cost_ms = elapsed_ms;
115
116        // Cumulative running average of pass latency
117        let n = f64::from(self.total_passes);
118        self.avg_pass_cost_ms = self.avg_pass_cost_ms * ((n - 1.0) / n) + (elapsed_ms / n);
119
120        if token_count > 0 && elapsed_ms > 0.0 {
121            let ms_per_tok = elapsed_ms / f64::from(token_count);
122            if is_prefill {
123                self.total_prefill_tokens = self.total_prefill_tokens.saturating_add(token_count);
124                if self.avg_prefill_ms_per_tok == 0.0 {
125                    self.avg_prefill_ms_per_tok = ms_per_tok;
126                } else {
127                    self.avg_prefill_ms_per_tok =
128                        EMA_ALPHA * ms_per_tok + (1.0 - EMA_ALPHA) * self.avg_prefill_ms_per_tok;
129                }
130            } else {
131                self.total_decode_tokens = self.total_decode_tokens.saturating_add(token_count);
132                if self.avg_decode_ms_per_tok == 0.0 {
133                    self.avg_decode_ms_per_tok = ms_per_tok;
134                } else {
135                    self.avg_decode_ms_per_tok =
136                        EMA_ALPHA * ms_per_tok + (1.0 - EMA_ALPHA) * self.avg_decode_ms_per_tok;
137                }
138            }
139        }
140
141        // Refine dynamic scheduling weight (cost-weighted fair sharing per ADR-010)
142        self.dynamic_scheduling_weight = if self.avg_pass_cost_ms > 0.0 {
143            self.avg_pass_cost_ms
144        } else {
145            1.0
146        };
147
148        // Advance virtual runtime proportional to measured pass cost and fair weight (default w_i = 1.0):
149        // V_i += cost_i / w_i
150        self.virtual_runtime_ms += elapsed_ms;
151    }
152
153    /// Records completion of a pass where both prefill and decode metrics are measured.
154    pub fn record_combined_pass(&mut self, prefill_ms: f64, decode_ms: f64, token_count: u32) {
155        let total_ms = prefill_ms + decode_ms;
156        self.total_passes = self.total_passes.saturating_add(1);
157        self.total_tokens = self.total_tokens.saturating_add(token_count);
158        self.last_pass_cost_ms = total_ms;
159
160        let n = f64::from(self.total_passes);
161        self.avg_pass_cost_ms = self.avg_pass_cost_ms * ((n - 1.0) / n) + (total_ms / n);
162
163        if token_count > 0 {
164            if prefill_ms > 0.0 {
165                let ms_per_tok = prefill_ms / f64::from(token_count);
166                self.total_prefill_tokens = self.total_prefill_tokens.saturating_add(token_count);
167                if self.avg_prefill_ms_per_tok == 0.0 {
168                    self.avg_prefill_ms_per_tok = ms_per_tok;
169                } else {
170                    self.avg_prefill_ms_per_tok =
171                        EMA_ALPHA * ms_per_tok + (1.0 - EMA_ALPHA) * self.avg_prefill_ms_per_tok;
172                }
173            }
174            if decode_ms > 0.0 {
175                let ms_per_tok = decode_ms / f64::from(token_count);
176                self.total_decode_tokens = self.total_decode_tokens.saturating_add(token_count);
177                if self.avg_decode_ms_per_tok == 0.0 {
178                    self.avg_decode_ms_per_tok = ms_per_tok;
179                } else {
180                    self.avg_decode_ms_per_tok =
181                        EMA_ALPHA * ms_per_tok + (1.0 - EMA_ALPHA) * self.avg_decode_ms_per_tok;
182                }
183            }
184        }
185
186        self.dynamic_scheduling_weight = if self.avg_pass_cost_ms > 0.0 {
187            self.avg_pass_cost_ms
188        } else {
189            1.0
190        };
191
192        // Advance virtual runtime proportional to measured pass cost:
193        // V_i += cost_i / w_i
194        self.virtual_runtime_ms += total_ms;
195    }
196}

279/// Multi-model compute scheduler arbitrating overlapping reservations at forward-pass boundaries.
280///
281/// Implements:
282/// - Operator Tuning Immutability (US-08)
283/// - Multi-Model Coexistence & Overlapping Time-Slicing (US-09)
284/// - Priority Preemption & Cost-Weighted Fair Sharing (US-10)
285/// - Opt-In Dynamic Idle Core Spillover & Reclaim (US-11)
286#[derive(Debug)]
287pub struct MultiModelScheduler {
288    state: Mutex<SchedulerState>,
289    /// Condvar signaled at every forward-pass boundary when cores unlock or requests update.

377    /// Increments the pending request counter for `model_id`.
378    ///
379    /// # Errors
380    /// Returns [`ManagerError::ModelNotFound`] if model is not registered.
381    pub fn inc_pending_requests(&self, model_id: &str) -> Result<u32, ManagerError> {
382        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
383        let entry = state
384            .active_models
385            .get_mut(model_id)
386            .ok_or_else(|| ManagerError::ModelNotFound(model_id.to_owned()))?;
387        entry.pending_requests_count = entry.pending_requests_count.saturating_add(1);
388        entry.state = ModelSchedulingState::Waiting;
389        entry.idle_since = None;
390        self.condvar.notify_all();
391        Ok(entry.pending_requests_count)
392    }

396    /// # Errors

398    pub fn dec_pending_requests(&self, model_id: &str) -> Result<u32, ManagerError> {
399        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
400        let entry = state
401            .active_models
402            .get_mut(model_id)
403            .ok_or_else(|| ManagerError::ModelNotFound(model_id.to_owned()))?;
404        entry.pending_requests_count = entry.pending_requests_count.saturating_sub(1);
405        if entry.is_idle() {
406            entry.state = ModelSchedulingState::Idle;
407            entry.idle_since = Some(Instant::now());
408        }
409        self.condvar.notify_all();
410        Ok(entry.pending_requests_count)
411    }

413    /// Admits or registers an active model placement into the scheduler.
414    ///
415    /// If `model_id` is already registered, updates its placement and tuning configuration.
416    /// If `model_id` is new and active models count has reached `max_active_models`,
417    /// rejects upfront with [`ManagerError::ConcurrencyLimitExceeded`] per ADR-043 and US-09 Scenario 3.
418    ///
419    /// # Errors
420    /// Returns [`ManagerError::ConcurrencyLimitExceeded`] if capacity is full,
421    /// [`ManagerError::Affinity`] on core mask violation, or
422    /// [`ManagerError::WorkerSpawn`] if unprofiled models attempt to enable lending/borrowing.
423    pub fn admit_or_register_model(
424        &self,
425        model_id: &str,
426        cluster: String,
427        reservation_mask: u64,
428        tuning: Option<&TuningProfile>,
429    ) -> Result<(), ManagerError> {
430        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
431
432        let is_existing = state.active_models.contains_key(model_id);
433
434        if !is_existing && state.active_models.len() >= state.max_active_models {
435            return Err(ManagerError::ConcurrencyLimitExceeded {
436                active: state.active_models.len(),
437                max: state.max_active_models,
438            });
439        }
440
441        // Validate compute mask against SoC isolation boundaries (ADR-007, ADR-041)
442        if (reservation_mask & self.capacity.active_control_mask) != 0 {
443            return Err(ManagerError::Affinity(k3_core::K3Error::AffinityError {
444                requested_mask: reservation_mask,
445                requested_threads: reservation_mask.count_ones() as usize,
446                reason: format!(
447                    "model '{model_id}' reservation 0x{reservation_mask:04X} overlaps dedicated active control cores 0x{:04X}",
448                    self.capacity.active_control_mask
449                ),
450            }));
451        }
452
453        let is_unprofiled = tuning.is_none_or(|t| t.unprofiled);
454
455        // Unprofiled models cannot enable dynamic idle core lending or borrowing (ADR-030, ADR-031)
456        if is_unprofiled {
457            if let Some(t) = tuning {
458                if t.lend_idle_cores || t.borrow_idle_cores {
459                    return Err(ManagerError::WorkerSpawn(format!(
460                        "unprofiled model '{model_id}' cannot enable dynamic idle core lending or borrowing (ADR-030)"
461                    )));
462                }
463            }
464        }
465
466        let priority_class = tuning.map_or(PriorityClass::Throughput, |t| t.priority_class);
467        let lend_idle_cores = tuning.is_some_and(|t| t.lend_idle_cores);
468        let borrow_idle_cores = tuning.is_some_and(|t| t.borrow_idle_cores);
469        let n_ctx = tuning.map_or(2048, |t| t.n_ctx);
470
471        let compute_cap_mask = match cluster.to_ascii_lowercase().as_str() {
472            "x100" => DEFAULT_X100_COMPUTE_CAP_MASK,
473            "a100" => A100_CLUSTER_MASK,
474            _ => DEFAULT_X100_COMPUTE_CAP_MASK | A100_CLUSTER_MASK,
475        };
476
477        if let Some(existing) = state.active_models.get_mut(model_id) {
478            existing.cluster = cluster;
479            existing.reservation_mask = reservation_mask;
480            existing.compute_cap_mask = compute_cap_mask;
481            existing.priority_class = priority_class;
482            existing.lend_idle_cores = lend_idle_cores;
483            existing.borrow_idle_cores = borrow_idle_cores;
484            existing.unprofiled = is_unprofiled;
485            existing.n_ctx = n_ctx;
486            existing.tuning_profile = tuning.cloned();
487        } else {
488            let mut metrics = ModelExecutionMetrics::new(model_id);
489            // Catch up virtual runtime to current system minimum to prevent unfair starvation
490            metrics.virtual_runtime_ms = state.min_virtual_runtime_ms;
491
492            state.active_models.insert(
493                model_id.to_owned(),
494                ModelSchedulerEntry {
495                    model_id: model_id.to_owned(),
496                    cluster,
497                    reservation_mask,
498                    compute_cap_mask,
499                    priority_class,
500                    lend_idle_cores,
501                    borrow_idle_cores,
502                    unprofiled: is_unprofiled,
503                    n_ctx,
504                    state: ModelSchedulingState::Idle,
505                    active_pass_count: 0,
506                    active_running_mask: 0,
507                    active_borrowed_mask: 0,
508                    active_lent_mask: 0,
509                    borrowed_from: Vec::new(),
510                    pending_requests_count: 0,
511                    idle_since: Some(Instant::now()),
512                    metrics,
513                    tuning_profile: tuning.cloned(),
514                },
515            );
516        }
517
518        self.condvar.notify_all();
519        Ok(())
520    }

522    /// Unregisters an active model from the scheduler, releasing any held core locks and waking waiting peers.
523    pub fn unregister_model(&self, model_id: &str) -> bool {
524        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
525        if let Some(removed) = state.active_models.remove(model_id) {
526            state.locked_cores_mask &= !removed.reservation_mask;
527            state.locked_cores_mask &= !removed.active_borrowed_mask;
528
529            // If this model lent cores to peers, clean up lender tracking
530            for peer in state.active_models.values_mut() {
531                if peer.borrowed_from.iter().any(|id| id == model_id) {
532                    peer.borrowed_from.retain(|id| id != model_id);
533                }
534            }
535
536            self.condvar.notify_all();
537            true
538        } else {
539            false
540        }
541    }

543    /// Evaluates whether a candidate model can be scheduled given the current scheduler state.
544    ///
545    /// Returns `Some((allocated_mask, borrowed_mask, borrowed_from))` if eligible, or `None`.
546    fn try_select_model(
547        state: &SchedulerState,
548        candidate_id: &str,
549    ) -> Option<(u64, u64, Vec<String>)> {
550        let candidate = state.active_models.get(candidate_id)?;
551
552        // 1. Single-Prompt Exclusivity (ADR-005): model must not be running a pass
553        if candidate.active_pass_count > 0 {
554            return None;
555        }
556
557        // 2. Physical core availability: candidate base reservation must not overlap locked cores
558        let my_mask = candidate.reservation_mask;
559        if (my_mask & state.locked_cores_mask) != 0 {
560            return None;
561        }
562
563        // 3. Contention & Priority Arbitration (ADR-014, ADR-010)
564        // Check if any other model with pending requests contends for overlapping cores
565        // and has higher priority or lower virtual runtime
566        for (peer_id, peer) in &state.active_models {
567            if peer_id == candidate_id || peer.pending_requests_count == 0 {
568                continue;
569            }
570
571            // Do we contend for overlapping cores with this peer?
572            let overlaps = (peer.reservation_mask & my_mask) != 0;
573            if !overlaps {
574                continue;
575            }
576
577            // Priority Preemption: Latency preempts Throughput
578            if candidate.priority_class == PriorityClass::Throughput
579                && peer.priority_class == PriorityClass::Latency
580            {
581                return None;
582            }
583
584            // Intra-Class Fairness: Compare virtual runtime
585            if candidate.priority_class == peer.priority_class {
586                if peer.metrics.virtual_runtime_ms < candidate.metrics.virtual_runtime_ms {
587                    return None;
588                } else if (peer.metrics.virtual_runtime_ms - candidate.metrics.virtual_runtime_ms)
589                    .abs()
590                    < f64::EPSILON
591                    && peer_id.as_str() < candidate_id
592                {
593                    // Deterministic tie-breaker
594                    return None;
595                }
596            }
597        }
598
599        // Model is selected! Calculate dynamic idle core borrowing if opted in (US-11 Scenario 2)
600        let allocated_mask = candidate.reservation_mask;
601        let mut borrowed_mask = 0_u64;
602        let mut borrowed_from = Vec::new();
603
604        if candidate.borrow_idle_cores && !candidate.unprofiled {
605            for (peer_id, peer) in &state.active_models {
606                if peer_id == candidate_id {
607                    continue;
608                }
609
610                // Peer must opt into lending, be completely idle, have zero pending requests, and not be unprofiled
611                if peer.lend_idle_cores
612                    && peer.is_idle()
613                    && peer.active_lent_mask == 0
614                    && !peer.unprofiled
615                {
616                    // Lendable cores: peer's reservation that does not overlap currently locked cores
617                    let lendable = peer.reservation_mask & !state.locked_cores_mask;
618                    // Guard: strictly bound within candidate's compute cap and cluster
619                    let candidate_cap = candidate.compute_cap_mask;
620                    let valid_lendable = lendable
621                        & candidate_cap
622                        & (DEFAULT_X100_COMPUTE_CAP_MASK | A100_CLUSTER_MASK);
623
624                    // Guard: never touch dedicated active control cores or spare recovery core
625                    let sanitized = valid_lendable
626                        & !(DEFAULT_ACTIVE_CONTROL_MASK | DEFAULT_SPARE_RECOVERY_MASK);
627
628                    if sanitized != 0 {
629                        borrowed_mask |= sanitized;
630                        borrowed_from.push(peer_id.clone());
631                    }
632                }
633            }
634        }
635
636        Some((allocated_mask, borrowed_mask, borrowed_from))
637    }

639    /// Schedules an exclusive forward pass for `model_id`, blocking until required cores are available.
640    ///
641    /// Yields a [`ScheduledPass`] specifying base allocated cores and optional borrowed cores.
642    ///
643    /// # Errors
644    /// Returns [`ManagerError::ModelNotFound`] if model is not registered,
645    /// or [`ManagerError::WorkerUnresponsive`] if the timeout expires.
646    pub fn schedule_forward_pass(
647        &self,
648        model_id: &str,
649        timeout: Option<Duration>,
650    ) -> Result<ScheduledPass, ManagerError> {
651        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
652
653        // 1. Verify model presence and increment pending requests count
654        {
655            let entry = state
656                .active_models
657                .get_mut(model_id)
658                .ok_or_else(|| ManagerError::ModelNotFound(model_id.to_owned()))?;
659            entry.pending_requests_count += 1;
660            entry.state = ModelSchedulingState::Waiting;
661            entry.idle_since = None;
662        }
663
664        let start_time = Instant::now();
665
666        // 2. Loop until scheduling conditions are satisfied or timeout occurs
667        loop {
668            if let Some((allocated_mask, borrowed_mask, borrowed_from)) =
669                Self::try_select_model(&state, model_id)
670            {
671                let effective_mask = allocated_mask | borrowed_mask;
672                state.locked_cores_mask |= effective_mask;
673
674                if let Some(entry) = state.active_models.get_mut(model_id) {
675                    entry.pending_requests_count = entry.pending_requests_count.saturating_sub(1);
676                    entry.active_pass_count = 1;
677                    entry.active_running_mask = effective_mask;
678                    entry.active_borrowed_mask = borrowed_mask;
679                    entry.borrowed_from.clone_from(&borrowed_from);
680                    entry.state = ModelSchedulingState::Running;
681                }
682
683                // Mark lenders with lent mask
684                for lender_id in &borrowed_from {
685                    if let Some(lender) = state.active_models.get_mut(lender_id) {
686                        lender.active_lent_mask |= lender.reservation_mask & borrowed_mask;
687                    }
688                }
689
690                return Ok(ScheduledPass {
691                    model_id: model_id.to_owned(),
692                    allocated_mask,
693                    borrowed_mask,
694                    effective_mask,
695                    borrowed_from,
696                    scheduled_at: Instant::now(),
697                });
698            }
699
700            // Wait on condvar for next pass boundary or request update
701            if let Some(t) = timeout {
702                let elapsed = start_time.elapsed();
703                if elapsed >= t {
704                    if let Some(e) = state.active_models.get_mut(model_id) {
705                        e.pending_requests_count = e.pending_requests_count.saturating_sub(1);
706                        if e.is_idle() {
707                            e.state = ModelSchedulingState::Idle;
708                            e.idle_since = Some(Instant::now());
709                        }
710                    }
711                    let elapsed_u64 = u64::try_from(elapsed.as_millis()).unwrap_or(u64::MAX);
712                    return Err(ManagerError::WorkerUnresponsive {
713                        model_id: model_id.to_owned(),
714                        pid: None,
715                        elapsed_ms: elapsed_u64,
716                    });
717                }
718                let remaining = t.saturating_sub(elapsed);
719                let (new_state, timeout_res) = self
720                    .condvar
721                    .wait_timeout(state, remaining)
722                    .unwrap_or_else(PoisonError::into_inner);
723                state = new_state;
724                if timeout_res.timed_out() {
725                    if let Some(e) = state.active_models.get_mut(model_id) {
726                        e.pending_requests_count = e.pending_requests_count.saturating_sub(1);
727                        if e.is_idle() {
728                            e.state = ModelSchedulingState::Idle;
729                            e.idle_since = Some(Instant::now());
730                        }
731                    }
732                    let elapsed_u64 =
733                        u64::try_from(start_time.elapsed().as_millis()).unwrap_or(u64::MAX);
734                    return Err(ManagerError::WorkerUnresponsive {
735                        model_id: model_id.to_owned(),
736                        pid: None,
737                        elapsed_ms: elapsed_u64,
738                    });
739                }
740            } else {
741                state = self
742                    .condvar
743                    .wait(state)
744                    .unwrap_or_else(PoisonError::into_inner);
745            }
746        }
747    }

749    /// Attempts to schedule a forward pass immediately without blocking.
750    ///
751    /// Returns `Ok(Some(permit))` if cores are available and the model is selected,
752    /// or `Ok(None)` if cores are contested or a higher-priority model is waiting.
753    ///
754    /// # Errors
755    /// Returns [`ManagerError::ModelNotFound`] if model is not registered.
756    pub fn try_schedule_forward_pass(
757        &self,
758        model_id: &str,
759    ) -> Result<Option<ScheduledPass>, ManagerError> {
760        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
761
762        if !state.active_models.contains_key(model_id) {
763            return Err(ManagerError::ModelNotFound(model_id.to_owned()));
764        }
765
766        if let Some((allocated_mask, borrowed_mask, borrowed_from)) =
767            Self::try_select_model(&state, model_id)
768        {
769            let effective_mask = allocated_mask | borrowed_mask;
770            state.locked_cores_mask |= effective_mask;
771
772            if let Some(entry) = state.active_models.get_mut(model_id) {
773                entry.pending_requests_count = entry.pending_requests_count.saturating_sub(1);
774                entry.active_pass_count = 1;
775                entry.active_running_mask = effective_mask;
776                entry.active_borrowed_mask = borrowed_mask;
777                entry.borrowed_from.clone_from(&borrowed_from);
778                entry.state = ModelSchedulingState::Running;
779            }
780
781            for lender_id in &borrowed_from {
782                if let Some(lender) = state.active_models.get_mut(lender_id) {
783                    lender.active_lent_mask |= lender.reservation_mask & borrowed_mask;
784                }
785            }
786
787            Ok(Some(ScheduledPass {
788                model_id: model_id.to_owned(),
789                allocated_mask,
790                borrowed_mask,
791                effective_mask,
792                borrowed_from,
793                scheduled_at: Instant::now(),
794            }))
795        } else {
796            Ok(None)
797        }
798    }

800    /// Records completion of a forward pass, updating internal metrics and releasing core locks.
801    ///
802    /// Implements instant pass-boundary reclaim (US-11 Scenario 3) and preserves operator tuning
803    /// profile immutability (US-08 Scenario 1).
804    pub fn record_pass_completion(
805        &self,
806        pass: &ScheduledPass,
807        prefill_latency_ms: f64,
808        decode_latency_ms: f64,
809        token_count: u32,
810    ) {
811        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
812
813        // 1. Release running cores (both base reservation and borrowed cores)
814        state.locked_cores_mask &= !pass.effective_mask;
815
816        // 2. Clear borrowed cores and lenders' lent cores (Instant pass-boundary reclaim)
817        for lender_id in &pass.borrowed_from {
818            if let Some(lender) = state.active_models.get_mut(lender_id) {
819                lender.active_lent_mask &= !pass.borrowed_mask;
820                if lender.pending_requests_count > 0 {
821                    lender.state = ModelSchedulingState::Waiting;
822                } else {
823                    lender.state = ModelSchedulingState::Idle;
824                    lender.idle_since = Some(Instant::now());
825                }
826            }
827        }
828
829        // 3. Update borrower model entry
830        if let Some(entry) = state.active_models.get_mut(&pass.model_id) {
831            entry.active_pass_count = 0;
832            entry.active_running_mask = 0;
833            entry.active_borrowed_mask = 0;
834            entry.borrowed_from.clear();
835
836            if entry.pending_requests_count == 0 {
837                entry.state = ModelSchedulingState::Idle;
838                entry.idle_since = Some(Instant::now());
839            } else {
840                entry.state = ModelSchedulingState::Waiting;
841            }
842
843            let has_prefill = prefill_latency_ms > 0.0;
844            let has_decode = decode_latency_ms > 0.0;
845            if has_prefill && !has_decode {
846                entry
847                    .metrics
848                    .record_pass(true, prefill_latency_ms, token_count);
849            } else if has_decode && !has_prefill {
850                entry
851                    .metrics
852                    .record_pass(false, decode_latency_ms, token_count);
853            } else if has_prefill && has_decode {
854                entry.metrics.record_combined_pass(
855                    prefill_latency_ms,
856                    decode_latency_ms,
857                    token_count,
858                );
859            } else {
860                entry.metrics.record_pass(false, 0.0, token_count);
861            }
862        }
863
864        // 4. Update system minimum virtual runtime
865        state.min_virtual_runtime_ms = state
866            .active_models
867            .values()
868            .map(|e| e.metrics.virtual_runtime_ms)
869            .fold(f64::INFINITY, f64::min);
870        if state.min_virtual_runtime_ms.is_infinite() {
871            state.min_virtual_runtime_ms = 0.0;
872        }
873
874        // 5. Notify all waiting threads at this pass boundary
875        self.condvar.notify_all();
876    }

878    /// Records failure, timeout, or abnormal termination of a forward pass, unlocking cores.
879    pub fn record_pass_failure(&self, pass: &ScheduledPass) {
880        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
881
882        state.locked_cores_mask &= !pass.effective_mask;
883
884        for lender_id in &pass.borrowed_from {
885            if let Some(lender) = state.active_models.get_mut(lender_id) {
886                lender.active_lent_mask &= !pass.borrowed_mask;
887                if lender.pending_requests_count > 0 {
888                    lender.state = ModelSchedulingState::Waiting;
889                } else {
890                    lender.state = ModelSchedulingState::Idle;
891                    lender.idle_since = Some(Instant::now());
892                }
893            }
894        }
895
896        if let Some(entry) = state.active_models.get_mut(&pass.model_id) {
897            entry.active_pass_count = 0;
898            entry.active_running_mask = 0;
899            entry.active_borrowed_mask = 0;
900            entry.borrowed_from.clear();
901
902            if entry.pending_requests_count == 0 {
903                entry.state = ModelSchedulingState::Idle;
904                entry.idle_since = Some(Instant::now());
905            } else {
906                entry.state = ModelSchedulingState::Waiting;
907            }
908        }
909
910        self.condvar.notify_all();
911    }

913    /// Handles an unexpected worker crash, clearing all locked cores and waking waiting peers.
914    pub fn on_worker_crashed(&self, model_id: &str) {
915        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
916
917        let cleanup = if let Some(entry) = state.active_models.get_mut(model_id) {
918            let running = entry.active_running_mask;
919            let borrowed = entry.active_borrowed_mask;
920            let from = entry.borrowed_from.clone();
921            entry.active_pass_count = 0;
922            entry.active_running_mask = 0;
923            entry.active_borrowed_mask = 0;
924            entry.borrowed_from.clear();
925            entry.state = ModelSchedulingState::Idle;
926            Some((running, borrowed, from))
927        } else {
928            None
929        };
930
931        if let Some((running, borrowed, from)) = cleanup {
932            state.locked_cores_mask &= !running;
933            for lender_id in &from {
934                if let Some(lender) = state.active_models.get_mut(lender_id) {
935                    lender.active_lent_mask &= !borrowed;
936                    if lender.pending_requests_count > 0 {
937                        lender.state = ModelSchedulingState::Waiting;
938                    } else {
939                        lender.state = ModelSchedulingState::Idle;
940                    }
941                }
942            }
943        }
944
945        self.condvar.notify_all();
946    }

992    /// Generates an on-demand evidence-based advisory tuning recommendation (US-08 Scenario 2, ADR-032).
993    ///
994    /// Compares online execution metrics against static hardware ceilings without mutating
995    /// the active operator tuning profile.
996    ///
997    /// # Errors
998    /// Returns [`ManagerError::ModelNotFound`] if `model_id` is unknown.
999    pub fn suggest_tuning(
1000        &self,
1001        model_id: &str,
1002        baseline: Option<&CostProfile>,
1003    ) -> Result<TuningAdvisory, ManagerError> {
1004        let state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
1005        let entry = state
1006            .active_models
1007            .get(model_id)
1008            .ok_or_else(|| ManagerError::ModelNotFound(model_id.to_owned()))?;
1009
1010        let m = &entry.metrics;
1011        let mut recommendations = Vec::new();
1012
1013        // 1. Bottleneck classification based on decode vs prefill latency ratio and baseline ceilings
1014        let is_bandwidth_bound = (m.avg_prefill_ms_per_tok == 0.0
1015            || m.avg_decode_ms_per_tok > m.avg_prefill_ms_per_tok * 1.5)
1016            && m.avg_decode_ms_per_tok > 0.0;
1017        let bottleneck = if is_bandwidth_bound {
1018            recommendations.push(
1019                "Autoregressive decode is memory-bandwidth bound. Consider increasing A100 core allocation or reducing KV cache precision.".to_owned(),
1020            );
1021            "bandwidth-bound"
1022        } else {
1023            recommendations.push(
1024                "Prompt prefill is compute bound. Consider enabling SpacemiT IME matrix acceleration or expanding X100 compute cap.".to_owned(),
1025            );
1026            "compute-bound"
1027        };
1028
1029        // 2. Cluster targeting recommendation
1030        let recommended_cluster = if bottleneck == "compute-bound" && entry.cluster == "x100" {
1031            recommendations.push(
1032                "Model exhibits high prefill latency on Cluster 0 (X100). Offloading prefill to Cluster 1 (A100) will yield ~3.2x MAC acceleration.".to_owned(),
1033            );
1034            "both".to_owned()
1035        } else {
1036            entry.cluster.clone()
1037        };
1038
1039        // 3. Priority recommendation
1040        if entry.priority_class == PriorityClass::Throughput
1041            && (model_id.contains("router") || model_id.contains("interactive"))
1042        {
1043            recommendations.push(
1044                "Interactive/decision model configured with Throughput priority. Transitioning to Latency priority class will enable pass-boundary preemption.".to_owned(),
1045            );
1046        }
1047
1048        // 4. Dynamic core lending/borrowing recommendation
1049        if !entry.lend_idle_cores && !entry.borrow_idle_cores {
1050            recommendations.push(
1051                "Dynamic core lending and borrowing are disabled. Enabling both flags will permit single-pass idle core work-stealing.".to_owned(),
1052            );
1053        }
1054
1055        // 5. Unprofiled status warning
1056        if entry.unprofiled {
1057            recommendations.push(
1058                "Model is currently running conservative defaults (X100, throughput, n_ctx=2048). Run 'k3-diag' to evaluate empirical hardware ceiling baselines.".to_owned(),
1059            );
1060        }
1061
1062        let rationale = format!(
1063            "Observed {} passes (avg pass: {:.2}ms, prefill: {:.2}ms/tok, decode: {:.2}ms/tok). Baseline static weight: {:.2}, dynamic weight: {:.2}.",
1064            m.total_passes,
1065            m.avg_pass_cost_ms,
1066            m.avg_prefill_ms_per_tok,
1067            m.avg_decode_ms_per_tok,
1068            baseline.map_or(1.0, |b| b.static_scheduling_weight),
1069            m.dynamic_scheduling_weight
1070        );
1071
1072        let recommended_ime = recommended_cluster.eq_ignore_ascii_case("a100")
1073            || recommended_cluster.eq_ignore_ascii_case("both");
1074
1075        Ok(TuningAdvisory {
1076            bottleneck_classification: bottleneck.to_owned(),
1077            recommended_cluster,
1078            recommended_split: None,
1079            recommended_ime,
1080            rationale,
1081            recommendations,
1082            operator_authority_notice:
1083                "Advisory recommendation only. Runtime preserves operator tuning authority per ADR-032; changes will not take effect until explicitly applied."
1084                    .to_owned(),
1085        })
1086    }

1088    /// Explicitly applies an operator tuning profile, transitioning unprofiled models
1089    /// and updating scheduling parameters immediately without process restart.
1090    ///
1091    /// # Errors
1092    /// Returns [`ManagerError::ModelNotFound`] if model is not registered,
1093    /// or [`ManagerError::Affinity`] if the new reservation mask violates topology invariants.
1094    pub fn apply_tuning_profile(
1095        &self,
1096        model_id: &str,
1097        mut profile: TuningProfile,
1098    ) -> Result<(), ManagerError> {

1100        let entry = state
1101            .active_models
1102            .get_mut(model_id)
1103            .ok_or_else(|| ManagerError::ModelNotFound(model_id.to_owned()))?;
1104

1106        profile.profile_status = ProfileStatus::Profiled;
1107        profile.unprofiled = false;

1114        entry.borrow_idle_cores = profile.borrow_idle_cores;
1115        entry.unprofiled = false;
1116        entry.tuning_profile = Some(profile);

1118        self.condvar.notify_all();
1119        Ok(())
