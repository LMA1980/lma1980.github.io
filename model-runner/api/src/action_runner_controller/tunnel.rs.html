33pub async fn handle_cluster_tunnel(
34    ws: WebSocketUpgrade,
35    State(state): State<Arc<AppState>>,
36    headers: HeaderMap,
37    Query(query): Query<TunnelQuery>,
38) -> Result<impl IntoResponse, StatusCode> {
39    // Validate cluster authentication token
40    let auth_header = headers
41        .get("x-cluster-token")
42        .and_then(|h| h.to_str().ok())
43        .map(|s| s.to_string());
44
45    let provided_token = auth_header.or(query.token).unwrap_or_default();
46    if provided_token != state.config.cluster_token {
47        warn!("Unauthorized worker connection attempt to /api/cluster/tunnel");
48        return Err(StatusCode::UNAUTHORIZED);
49    }
50
51    Ok(ws.on_upgrade(move |socket| handle_socket(socket, state)))
52}

54async fn handle_socket(socket: WebSocket, state: Arc<AppState>) {
55    let (mut ws_sender, mut ws_receiver) = socket.split();
56    let fory = match create_fory_runtime() {
57        Ok(f) => f,
58        Err(e) => {
59            error!("Failed to create Fory runtime: {}", e);
60            return;
61        }
62    };
63
64    let (cmd_tx, mut cmd_rx) = mpsc::channel::<WorkerCommand>(32);
65    let mut registered_node_id: Option<String> = None;
66    let mut seq = 0u64;
67
68    // First message must be WorkerRegistration
69    if let Some(Ok(Message::Binary(bin))) = ws_receiver.next().await {
70        if let Ok((hdr, payload)) = IpcHeader::parse(&bin) {
71            if hdr.msg_type == MSG_TYPE_WORKER_REGISTRATION {
72                if let Ok(reg) = fory.deserialize::<WorkerRegistration>(payload) {
73                    let node_id = reg.node_id.clone();
74                    state.cluster.register(reg, cmd_tx).await;
75                    registered_node_id = Some(node_id);
76                }
77            }
78        }
79    }
80
81    let node_id = match registered_node_id {
82        Some(id) => id,
83        None => {
84            warn!("Worker failed to register within initial frame; terminating tunnel");
85            return;
86        }
87    };
88
89    info!(node = %node_id, "Cluster tunnel session active");
90
91    loop {
92        tokio::select! {
93            // Outbound commands dispatched to worker
94            Some(cmd) = cmd_rx.recv() => {
95                seq += 1;
96                match cmd {
97                    WorkerCommand::Dispatch(task) => {
98                        match fory.serialize(&task) {
99                            Ok(payload) => {
100                                let header = IpcHeader::new(MSG_TYPE_STAGE_DISPATCH, payload.len() as u32, 0, seq);
101                                let mut frame = Vec::with_capacity(action_runner_ipc::HEADER_SIZE + payload.len());
102                                frame.extend_from_slice(zerocopy::IntoBytes::as_bytes(&header));
103                                frame.extend_from_slice(&payload);
104                                if let Err(e) = ws_sender.send(Message::Binary(frame.into())).await {
105                                    error!("Failed to send dispatch frame over ws: {}", e);
106                                } else {
107                                    info!(stage = %task.stage_id, "Dispatched stage frame successfully sent to ws_sender");
108                                }
109                            }
110                            Err(e) => {
111                                error!(stage = %task.stage_id, error = %e, "Failed to serialize StageDispatchTask with Fory");
112                            }
113                        }
114                    }
115                    WorkerCommand::Cancel(cancel) => {
116                        if let Ok(payload) = fory.serialize(&cancel) {
117                            let header = IpcHeader::new(MSG_TYPE_STAGE_CANCEL, payload.len() as u32, 0, seq);
118                            let mut frame = Vec::with_capacity(action_runner_ipc::HEADER_SIZE + payload.len());
119                            frame.extend_from_slice(zerocopy::IntoBytes::as_bytes(&header));
120                            frame.extend_from_slice(&payload);
121                            let _ = ws_sender.send(Message::Binary(frame.into())).await;
122                        }
123                    }
124                }
125            }
126
127            // Inbound messages received from worker
128            msg_res = ws_receiver.next() => {
129                let msg = match msg_res {
130                    Some(Ok(m)) => m,
131                    _ => break, // Disconnected
132                };
133
134                if let Message::Binary(bin) = msg {
135                    if let Ok((hdr, payload)) = IpcHeader::parse(&bin) {
136                        match hdr.msg_type {
137                            MSG_TYPE_WORKER_HEARTBEAT => {
138                                if let Ok(hb) = fory.deserialize::<WorkerHeartbeat>(payload) {
139                                    state.cluster.heartbeat(hb).await;
140                                }
141                            }
142                            MSG_TYPE_STAGE_COMPLETION => {
143                                if let Ok(report) = fory.deserialize::<StageCompletionReport>(payload) {
144                                    info!(
145                                        node = %node_id,
146                                        job_id = %report.job_id,
147                                        stage = %report.stage_id,
148                                        exit_code = report.exit_code,
149                                        duration = report.duration_secs,
150                                        "Stage completed by worker"
151                                    );
152
153                                    let status = if report.exit_code == 0 {
154                                        StageStatus::Completed {
155                                            duration_secs: report.duration_secs,
156                                            exit_code: report.exit_code,
157                                        }
158                                    } else {
159                                        StageStatus::Failed {
160                                            error: report.error_message.unwrap_or_else(|| {
161                                                format!("Process exited with code {}", report.exit_code)
162                                            }),
163                                        }
164                                    };
165
166                                    // Release the stage slot BEFORE updating stage status so any job woken by
167                                    // `update_stage_status` (which calls `notify_waiters()`) sees the freed node slot immediately.
168                                    state
169                                        .cluster
170                                        .release_stage_slot(
171                                            &node_id,
172                                            &report.job_id,
173                                            &report.stage_id,
174                                        )
175                                        .await;
176                                    state
177                                        .job_store
178                                        .update_stage_status(
179                                            &report.job_id,
180                                            &report.stage_id,
181                                            status,
182                                        )
183                                        .await;
184                                    state.job_store.notify_event();
185                                }
186                            }
187                            MSG_TYPE_WORKER_SHUTDOWN => {
188                                if let Ok(shutdown) = fory.deserialize::<WorkerShutdown>(payload) {
189                                    info!(
190                                        node = %node_id,
191                                        reason = %shutdown.reason,
192                                        interrupted = ?shutdown.interrupted_stages,
193                                        "Worker signaled shutdown/restart"
194                                    );
195                                    state.cluster.release_slot(&node_id).await;
196                                    let recovered = state
197                                        .job_store
198                                        .recover_stages_for_node(
199                                            &node_id,
200                                            Some(&shutdown.interrupted_stages),
201                                        )
202                                        .await;
203                                    for (job_id, stage_id) in &recovered {
204                                        if let Some(record) = state.job_store.get_job(job_id).await
205                                        {
206                                            let full_path =
207                                                state.config.workspace_dir.join(&record.log_file);
208                                            let note = format!(
209                                                "{} [{}] [{}] [stderr] Worker signaled restart; stage execution interrupted and returned to queue for re-dispatch\n",
210                                                chrono::Utc::now().to_rfc3339(),
211                                                node_id,
212                                                stage_id
213                                            );
214                                            if let Ok(mut f) = tokio::fs::OpenOptions::new()
215                                                .create(true)
216                                                .append(true)
217                                                .open(&full_path)
218                                                .await
219                                            {
220                                                use tokio::io::AsyncWriteExt;
221                                                let _ = f.write_all(note.as_bytes()).await;
222                                            }
223                                            state.broadcast_log(job_id, &note);
224                                        }
225                                    }
226                                    state.job_store.notify_event();
227                                }
228                            }
229                            MSG_TYPE_STAGE_STARTED => {
230                                if let Ok(started) = fory.deserialize::<StageStarted>(payload) {
231                                    info!(
232                                        job_id = %started.job_id,
233                                        stage_id = %started.stage_id,
234                                        node_id = %started.node_id,
235                                        pid = started.pid,
236                                        "Worker reported stage process spawned with PID"
237                                    );
238                                    state
239                                        .job_store
240                                        .update_stage_running(
241                                            &started.job_id,
242                                            &started.stage_id,
243                                            &started.node_id,
244                                            started.pid,
245                                        )
246                                        .await;
247                                }
248                            }
249                            MSG_TYPE_ARROW_LOG => {
250                                if let Ok(entries) = decode_log_batch(payload) {
251                                    if let Some(first) = entries.first() {
252                                        if let Some(record) = state.job_store.get_job(&first.job_id).await {
253                                            let full_path = state.config.workspace_dir.join(&record.log_file);
254                                            let _ = append_logs(&full_path, &entries);
255                                            let chunk = format_log_entries(&entries);
256                                            state.broadcast_log(&first.job_id, &chunk);
257                                        }
258                                    }
259                                }
260                            }
261                            MSG_TYPE_ARROW_METRIC => {
262                                if let Ok(metric_entries) = decode_metric_batch(payload) {
263                                    if let Some(first) = metric_entries.first() {
264                                        let aggregated = aggregate_metrics(&metric_entries, true);
265                                        state.job_store.update_metrics(&first.job_id, aggregated).await;
266                                    }
267                                }
268                            }
269                            _ => {}
270                        }
271                    }
272                }
273            }
274        }
275    }
276
277    state.cluster.unregister(&node_id).await;
278    // Auto-recover any remaining stages assigned to this worker node
279    let recovered = state
280        .job_store
281        .recover_stages_for_node(&node_id, None)
282        .await;
283    if !recovered.is_empty() {
284        warn!(
285            node = %node_id,
286            count = recovered.len(),
287            "Worker disconnected; recovered orphan stages back to Pending"
288        );
289        for (job_id, stage_id) in &recovered {
290            if let Some(record) = state.job_store.get_job(job_id).await {
291                let full_path = state.config.workspace_dir.join(&record.log_file);
292                let note = format!(
293                    "{} [controller] [{}] [stderr] Worker '{}' disconnected unexpectedly; stage returned to queue for recovery\n",
294                    chrono::Utc::now().to_rfc3339(),
295                    stage_id,
296                    node_id
297                );
298                if let Ok(mut f) = tokio::fs::OpenOptions::new()
299                    .create(true)
300                    .append(true)
301                    .open(&full_path)
302                    .await
303                {
304                    use tokio::io::AsyncWriteExt;
305                    let _ = f.write_all(note.as_bytes()).await;
306                }
307                state.broadcast_log(job_id, &note);
308            }
309        }
310        state.job_store.notify_event();
311    }
312    info!(node = %node_id, "Cluster tunnel session terminated");
313}
