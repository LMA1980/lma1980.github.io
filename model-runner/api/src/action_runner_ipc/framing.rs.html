5pub const MAGIC: [u8; 4] = *b"K3CI";
6pub const CURRENT_VERSION: u16 = 1;

24pub const HEADER_SIZE: usize = std::mem::size_of::<IpcHeader>();
25
26#[derive(IntoBytes, FromBytes, KnownLayout, Immutable, Debug, PartialEq, Eq, Clone, Copy)]
27#[repr(C)]
28pub struct IpcHeader {
29    pub magic: [u8; 4],
30    pub version: u16,
31    pub msg_type: u16,
32    pub payload_len: u32,
33    pub flags: u32,
34    pub sequence: u64,
35}
36
37impl IpcHeader {
38    pub fn new(msg_type: u16, payload_len: u32, flags: u32, sequence: u64) -> Self {
39        Self {
40            magic: MAGIC,
41            version: CURRENT_VERSION,
42            msg_type,
43            payload_len,
44            flags,
45            sequence,
46        }
47    }
48
49    pub fn parse(buf: &[u8]) -> Result<(Self, &[u8]), IpcError> {
50        if buf.len() < HEADER_SIZE {
51            return Err(IpcError::BufferTooSmall {
52                expected: HEADER_SIZE,
53                actual: buf.len(),
54            });
55        }
56
57        let (hdr, rest) = Self::read_from_prefix(buf)
58            .map_err(|e| IpcError::MalformedHeader(format!("{:?}", e)))?;
59
60        if hdr.magic != MAGIC {
61            return Err(IpcError::InvalidMagic);
62        }
63
64        if hdr.version != CURRENT_VERSION {
65            return Err(IpcError::UnsupportedVersion {
66                expected: CURRENT_VERSION,
67                actual: hdr.version,
68            });
69        }
70
71        Ok((hdr, rest))
72    }
73}

75pub async fn read_frame<R: AsyncRead + Unpin>(
76    reader: &mut R,
77) -> Result<(IpcHeader, Vec<u8>), IpcError> {
78    let mut header_buf = [0u8; HEADER_SIZE];
79    reader.read_exact(&mut header_buf).await?;
80
81    let (header, _) = IpcHeader::parse(&header_buf)?;
82
83    let payload_len = header.payload_len as usize;
84    if payload_len > MAX_FRAME_SIZE {
85        return Err(IpcError::FrameTooLarge {
86            max: MAX_FRAME_SIZE,
87            actual: payload_len,
88        });
89    }
90
91    let mut payload = vec![0u8; payload_len];
92    if payload_len > 0 {
93        reader.read_exact(&mut payload).await?;
94    }
95
96    Ok((header, payload))
97}

99pub async fn write_frame<W: AsyncWrite + Unpin>(
100    writer: &mut W,
101    header: &IpcHeader,
102    payload: &[u8],
103) -> Result<(), IpcError> {
104    writer.write_all(header.as_bytes()).await?;
105    if !payload.is_empty() {
106        writer.write_all(payload).await?;
107    }
108    writer.flush().await?;
109    Ok(())
