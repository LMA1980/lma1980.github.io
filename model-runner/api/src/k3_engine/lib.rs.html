5//! with RVV 1.0 (`+v`) per ADR-029.

58#[cfg(test)]
59mod tests {
60    use super::{HAS_ARCH_SYNTHETIC, HAS_KERNELS_RVV, QuantType};
61
62    #[test]
63    fn feature_gates_match_cfg() {
64        assert_eq!(HAS_ARCH_SYNTHETIC, cfg!(feature = "arch-synthetic"));
65        assert_eq!(HAS_KERNELS_RVV, cfg!(feature = "kernels-rvv"));
66        assert_eq!(QuantType::Q4_K_M.as_str(), "Q4_K_M");
67    }
68
69    #[cfg(feature = "arch-synthetic")]
70    #[test]
71    fn arch_synthetic_recipe_wired() {
72        use super::ArchRecipe;
73        use super::arch_synthetic::SyntheticRecipe;
74
75        let recipe = SyntheticRecipe::new();
76        assert_eq!(recipe.architecture_id(), "synthetic");
77    }
78
79    #[cfg(feature = "kernels-rvv")]
80    #[test]
81    fn kernels_rvv_wired() {
82        use super::kernels_rvv::{BaselineKernel, VectorKernel};
83
84        let kernel = BaselineKernel::new();
85        assert!((kernel.dot_product_f32(&[2.0], &[3.0]) - 6.0).abs() < f32::EPSILON);
86    }
