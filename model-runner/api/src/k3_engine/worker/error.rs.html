1//! Standardized error types and process exit code mappings for `k3-worker`.
2
3use std::fmt;
4use std::io;
5
6use k3_core::error::K3Error;
7use k3_core::ipc::protocol::StatusCode;
8use k3_core::ipc::shm::IpcError;
9
10/// Process exit codes per K3 architecture specifications.
11pub const EXIT_SUCCESS: i32 = 0;
12pub const EXIT_INVALID_ARGS: i32 = 2;
13pub const EXIT_CAPABILITY_MISMATCH: i32 = 10;
14pub const EXIT_INSUFFICIENT_MEMLOCK: i32 = 11;
15pub const EXIT_RESOURCE_EXHAUSTED: i32 = 12;
16pub const EXIT_PROTOCOL_ERROR: i32 = 13;
17
18/// Errors produced during worker initialization and execution.
19#[derive(Debug)]
20pub enum WorkerError {
21    /// Command line arguments invalid or missing (exit code 2).
22    InvalidArguments(String),
23    /// Host capabilities or quantization incompatible (exit code 10).
24    CapabilityMismatch(String),
25    /// Memory footprint exceeds RLIMIT_MEMLOCK bounds or mlock failure (exit code 11).
26    InsufficientLockableMemory(String),
27    /// Model file missing, empty, or resource exhausted (exit code 12).
28    ResourceExhausted(String),
29    /// UDS framing, magic, or sequence protocol violation (exit code 13).
30    ProtocolError(String),
31    /// Internal unexpected execution failure (exit code 13).
32    Internal(String),
33}
34
35impl WorkerError {
36    /// Returns the standardized process exit code for this error.
37    #[must_use]
38    pub const fn exit_code(&self) -> i32 {
39        match self {
40            Self::InvalidArguments(_) => EXIT_INVALID_ARGS,
41            Self::CapabilityMismatch(_) => EXIT_CAPABILITY_MISMATCH,
42            Self::InsufficientLockableMemory(_) => EXIT_INSUFFICIENT_MEMLOCK,
43            Self::ResourceExhausted(_) => EXIT_RESOURCE_EXHAUSTED,
44            Self::ProtocolError(_) | Self::Internal(_) => EXIT_PROTOCOL_ERROR,
45        }
46    }
47
48    /// Returns corresponding [`StatusCode`] for IPC responses.
49    #[must_use]
50    pub const fn status_code(&self) -> StatusCode {
51        match self {
52            Self::InvalidArguments(_) => StatusCode::InternalError,
53            Self::CapabilityMismatch(_) => StatusCode::CapabilityMismatch,
54            Self::InsufficientLockableMemory(_) => StatusCode::InsufficientLockableMemory,
55            Self::ResourceExhausted(_) => StatusCode::InternalError,
56            Self::ProtocolError(_) => StatusCode::ProtocolError,
57            Self::Internal(_) => StatusCode::InternalError,
58        }
59    }
60}
61
62impl fmt::Display for WorkerError {
63    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
64        match self {
65            Self::InvalidArguments(msg) => write!(f, "invalid arguments: {msg}"),
66            Self::CapabilityMismatch(msg) => write!(f, "capability mismatch: {msg}"),
67            Self::InsufficientLockableMemory(msg) => {
68                write!(f, "insufficient lockable memory: {msg}")
69            }
70            Self::ResourceExhausted(msg) => write!(f, "resource exhausted: {msg}"),
71            Self::ProtocolError(msg) => write!(f, "protocol error: {msg}"),
72            Self::Internal(msg) => write!(f, "internal error: {msg}"),
73        }
74    }
75}
76
77impl std::error::Error for WorkerError {}
78
79impl From<io::Error> for WorkerError {
80    fn from(err: io::Error) -> Self {
81        if err.kind() == io::ErrorKind::InvalidData || err.kind() == io::ErrorKind::UnexpectedEof {
82            Self::ProtocolError(err.to_string())
83        } else {
84            Self::Internal(err.to_string())
85        }
86    }
87}
88
89impl From<K3Error> for WorkerError {
90    fn from(err: K3Error) -> Self {
91        match err {
92            K3Error::CapabilityMismatch { reason, .. } => Self::CapabilityMismatch(reason),
93            K3Error::InsufficientLockableMemory { .. } => {
94                Self::InsufficientLockableMemory(err.to_string())
95            }
96            K3Error::ContextLengthExceeded { .. } => Self::ResourceExhausted(err.to_string()),
97            K3Error::AffinityError { reason, .. } => Self::CapabilityMismatch(reason),
98            K3Error::TensorBoundsError { reason, .. } => Self::ProtocolError(reason),
99            K3Error::RoutingUnavailable { reason } => Self::ProtocolError(reason),
100            K3Error::QueueSaturated { .. } => Self::ResourceExhausted(err.to_string()),
101        }
102    }
103}
104
105impl From<IpcError> for WorkerError {
106    fn from(err: IpcError) -> Self {
107        match err {
108            IpcError::InsufficientLockableMemory(k3) => Self::from(k3),
109            IpcError::TensorBoundsError(k3) => Self::from(k3),
110            IpcError::ProtocolError(msg) => Self::ProtocolError(msg),
111            IpcError::IoError { reason, .. } => Self::Internal(reason),
112        }
113    }
114}
