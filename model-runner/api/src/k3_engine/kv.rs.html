1//! Standalone in-process locked KV cache arena.
2//!
3//! Organizes KV cache memory into fixed 16-token blocks sized strictly for 1 active slot × `n_ctx`
4//! tokens, locked in physical memory via `mlock()` and indexed via block tables (ADR-027, ADR-037, `kv-allocator.md`).
5
6use std::ptr;
7
8use k3_core::{K3Error, ModelMetadata, align_up};

10use crate::memory::{KV_BLOCK_SIZE, system_page_size};

12/// In-process single-slot KV cache arena organized in 16-token blocks and locked in RAM.
13#[derive(Debug)]
14pub struct SingleSlotKvArena {
15    ptr: *mut f32,
16    total_elements: usize,
17    aligned_bytes: usize,
18    n_ctx: usize,
19    num_blocks: usize,
20    layer_count: usize,
21    head_count_kv: usize,
22    head_dim: usize,
23    elements_per_token: usize,
24    elements_per_block: usize,
25}
26
27// SAFETY: SingleSlotKvArena owns an mmap/mlocked buffer of floats. Concurrent immutable access is safe.
28unsafe impl Send for SingleSlotKvArena {}
29// SAFETY: SingleSlotKvArena access adheres to Rust borrow rules (& vs &mut).
30unsafe impl Sync for SingleSlotKvArena {}
31
32#[derive(Debug, Clone, Copy)]
33struct ArenaLayout {
34    num_blocks: usize,
35    elements_per_token: usize,
36    elements_per_block: usize,
37    total_elements: usize,
38    aligned_bytes: usize,
39}
40
41fn calculate_arena_layout(
42    layer_count: usize,
43    head_count_kv: usize,
44    head_dim: usize,
45    n_ctx: usize,
46) -> Result<ArenaLayout, K3Error> {
47    let num_blocks = n_ctx.div_ceil(KV_BLOCK_SIZE);
48    let allocated_tokens =
49        num_blocks
50            .checked_mul(KV_BLOCK_SIZE)
51            .ok_or_else(|| K3Error::TensorBoundsError {
52                tensor_name: "kv_arena".to_owned(),
53                offset: 0,
54                byte_len: 0,
55                buffer_len: usize::MAX,
56                reason: "allocated tokens overflowed usize".to_owned(),
57            })?;
58
59    // 2 vectors (Key and Value) per layer * head_count_kv * head_dim
60    let elements_per_token = 2usize
61        .checked_mul(layer_count)
62        .and_then(|v| v.checked_mul(head_count_kv))
63        .and_then(|v| v.checked_mul(head_dim))
64        .ok_or_else(|| K3Error::TensorBoundsError {
65            tensor_name: "kv_arena".to_owned(),
66            offset: 0,
67            byte_len: 0,
68            buffer_len: usize::MAX,
69            reason: "elements per token overflowed usize".to_owned(),
70        })?;
71
72    let elements_per_block = KV_BLOCK_SIZE
73        .checked_mul(elements_per_token)
74        .ok_or_else(|| K3Error::TensorBoundsError {
75            tensor_name: "kv_arena".to_owned(),
76            offset: 0,
77            byte_len: 0,
78            buffer_len: usize::MAX,
79            reason: "elements per block overflowed usize".to_owned(),
80        })?;
81
82    let total_elements = allocated_tokens
83        .checked_mul(elements_per_token)
84        .ok_or_else(|| K3Error::TensorBoundsError {
85            tensor_name: "kv_arena".to_owned(),
86            offset: 0,
87            byte_len: 0,
88            buffer_len: usize::MAX,
89            reason: "total elements overflowed usize".to_owned(),
90        })?;
91
92    let raw_bytes = total_elements
93        .checked_mul(size_of::<f32>())
94        .ok_or_else(|| K3Error::TensorBoundsError {
95            tensor_name: "kv_arena".to_owned(),
96            offset: 0,
97            byte_len: 0,
98            buffer_len: usize::MAX,
99            reason: "raw bytes overflowed usize".to_owned(),
100        })?;
101
102    let page_size = system_page_size();
103    let aligned_bytes = align_up(raw_bytes, page_size);
104
105    Ok(ArenaLayout {
106        num_blocks,
107        elements_per_token,
108        elements_per_block,
109        total_elements,
110        aligned_bytes,
111    })
112}

114impl SingleSlotKvArena {
115    /// Allocates and locks an in-process single-slot KV cache arena sized for `n_ctx` tokens
116    /// matching the given model metadata parameters (ADR-027, ADR-037, `kv-allocator.md`).
117    ///
118    /// # Errors
119    ///
120    /// - Returns [`K3Error::CapabilityMismatch`] if `n_ctx` is zero.
121    /// - Returns [`K3Error::ContextLengthExceeded`] if `n_ctx` exceeds `metadata.context_length()`.
122    /// - Returns [`K3Error::InsufficientLockableMemory`] if `mmap` or `mlock` allocation fails.
123    pub fn new(metadata: &ModelMetadata, n_ctx: usize) -> Result<Self, K3Error> {
124        if n_ctx == 0 {
125            return Err(K3Error::CapabilityMismatch {
126                architecture: metadata.architecture().to_owned(),
127                reason: "KV cache context length n_ctx must be greater than zero".to_owned(),
128            });
129        }
130
131        if metadata.context_length() > 0 && n_ctx > metadata.context_length() {
132            return Err(K3Error::ContextLengthExceeded {
133                requested_tokens: n_ctx,
134                max_context_tokens: metadata.context_length(),
135            });
136        }
137
138        let layer_count = metadata.layer_count().max(1);
139        let head_count_kv = metadata.head_count_kv().max(1);
140        let head_dim = if metadata.head_count() > 0 {
141            metadata.embedding_dim() / metadata.head_count()
142        } else {
143            metadata.embedding_dim()
144        }
145        .max(1);
146
147        Self::with_dimensions(layer_count, head_count_kv, head_dim, n_ctx)
148    }
149
150    /// Allocates and locks an arena with explicitly specified model dimensions.
151    ///
152    /// # Errors
153    ///
154    /// Returns [`K3Error::InsufficientLockableMemory`] if allocation or `mlock` fails.
155    pub fn with_dimensions(
156        layer_count: usize,
157        head_count_kv: usize,
158        head_dim: usize,
159        n_ctx: usize,
160    ) -> Result<Self, K3Error> {
161        let layout = calculate_arena_layout(layer_count, head_count_kv, head_dim, n_ctx)?;
162
163        // 1. Allocate anonymous memory pages
164        // SAFETY: mmap is called with valid length, anonymous mapping, and read/write protection.
165        let raw_ptr = unsafe {
166            libc::mmap(
167                ptr::null_mut(),
168                layout.aligned_bytes,
169                libc::PROT_READ | libc::PROT_WRITE,
170                libc::MAP_PRIVATE | libc::MAP_ANONYMOUS,
171                -1,
172                0,
173            )
174        };
175
176        if raw_ptr == libc::MAP_FAILED {
177            return Err(K3Error::InsufficientLockableMemory {
178                required_bytes: u64::try_from(layout.aligned_bytes).unwrap_or(u64::MAX),
179                weight_bytes: 0,
180                kv_bytes: u64::try_from(layout.aligned_bytes).unwrap_or(u64::MAX),
181                available_bytes: 0,
182                rlimit_memlock_bytes: 0,
183            });
184        }
185
186        // 2. Best-effort huge page hint (ADR-037)
187        // SAFETY: madvise is called on the valid anonymous memory mapping.
188        #[cfg(target_os = "linux")]
189        unsafe {
190            libc::madvise(raw_ptr, layout.aligned_bytes, libc::MADV_HUGEPAGE);
191        }
192
193        // 3. Mandatory mlock() (ADR-037)
194        // SAFETY: mlock is called on the valid anonymous memory mapping.
195        let mlock_res = unsafe { libc::mlock(raw_ptr, layout.aligned_bytes) };
196        if mlock_res != 0 {
197            // SAFETY: Clean up mmap on mlock failure.
198            unsafe {
199                libc::munmap(raw_ptr, layout.aligned_bytes);
200            }
201            return Err(K3Error::InsufficientLockableMemory {
202                required_bytes: u64::try_from(layout.aligned_bytes).unwrap_or(u64::MAX),
203                weight_bytes: 0,
204                kv_bytes: u64::try_from(layout.aligned_bytes).unwrap_or(u64::MAX),
205                available_bytes: 0,
206                rlimit_memlock_bytes: 0,
207            });
208        }
209
210        let ptr = raw_ptr.cast::<f32>();
211
212        // Zero-initialize memory
213        // SAFETY: ptr is valid for writes of total_elements f32 elements.
214        unsafe {
215            ptr::write_bytes(ptr, 0, layout.total_elements);
216        }
217
218        Ok(Self {
219            ptr,
220            total_elements: layout.total_elements,
221            aligned_bytes: layout.aligned_bytes,
222            n_ctx,
223            num_blocks: layout.num_blocks,
224            layer_count,
225            head_count_kv,
226            head_dim,
227            elements_per_token: layout.elements_per_token,
228            elements_per_block: layout.elements_per_block,
229        })
230    }

274    /// Returns the full KV cache buffer as a flat immutable slice.
275    #[must_use]
276    pub fn as_slice(&self) -> &[f32] {
277        // SAFETY: self.ptr is valid for reads of self.total_elements f32 elements.
278        unsafe { std::slice::from_raw_parts(self.ptr, self.total_elements) }
279    }
280
281    /// Returns the full KV cache buffer as a flat mutable slice.
282    pub fn as_mut_slice(&mut self) -> &mut [f32] {
283        // SAFETY: self.ptr is valid for reads and writes of self.total_elements f32 elements.
284        unsafe { std::slice::from_raw_parts_mut(self.ptr, self.total_elements) }
285    }

287    /// Returns an immutable slice for a specific 16-token block.
288    ///
289    /// # Errors
290    ///
291    /// Returns [`K3Error::TensorBoundsError`] if `block_idx >= self.num_blocks`.
292    pub fn block_slice(&self, block_idx: usize) -> Result<&[f32], K3Error> {
293        if block_idx >= self.num_blocks {
294            return Err(K3Error::TensorBoundsError {
295                tensor_name: "kv_block".to_owned(),
296                offset: block_idx,
297                byte_len: self.elements_per_block * size_of::<f32>(),
298                buffer_len: self.total_elements * size_of::<f32>(),
299                reason: format!(
300                    "block index {block_idx} exceeds allocated block count {}",
301                    self.num_blocks
302                ),
303            });
304        }
305
306        let start = block_idx * self.elements_per_block;
307        let end = start + self.elements_per_block;
308        Ok(&self.as_slice()[start..end])
309    }

335    /// Returns `(key_slice, value_slice)` views for a specific logical token position and layer.
336    ///
337    /// # Errors
338    ///
339    /// Returns [`K3Error::ContextLengthExceeded`] if `token_pos >= self.n_ctx`.
340    /// Returns [`K3Error::TensorBoundsError`] if `layer >= self.layer_count`.
341    pub fn token_kv_slice(
342        &self,
343        token_pos: usize,
344        layer: usize,
345    ) -> Result<(&[f32], &[f32]), K3Error> {
346        if token_pos >= self.n_ctx {
347            return Err(K3Error::ContextLengthExceeded {
348                requested_tokens: token_pos + 1,
349                max_context_tokens: self.n_ctx,
350            });
351        }
352
353        if layer >= self.layer_count {
354            return Err(K3Error::TensorBoundsError {
355                tensor_name: "kv_layer".to_owned(),
356                offset: layer,
357                byte_len: 0,
358                buffer_len: self.layer_count,
359                reason: format!(
360                    "layer index {layer} exceeds model layer count {}",
361                    self.layer_count
362                ),
363            });
364        }
365
366        let block_idx = token_pos / KV_BLOCK_SIZE;
367        let token_in_block = token_pos % KV_BLOCK_SIZE;
368        let head_kv_elements = self.head_count_kv * self.head_dim;
369
370        let token_start =
371            (block_idx * self.elements_per_block) + (token_in_block * self.elements_per_token);
372        let layer_start = token_start + (layer * 2 * head_kv_elements);
373        let key_start = layer_start;
374        let key_end = key_start + head_kv_elements;
375        let val_start = key_end;
376        let val_end = val_start + head_kv_elements;
377
378        let full = self.as_slice();
379        Ok((&full[key_start..key_end], &full[val_start..val_end]))
380    }
381
382    /// Returns mutable `(key_slice, value_slice)` views for a specific logical token position and layer.
383    ///
384    /// # Errors
385    ///
386    /// Returns [`K3Error::ContextLengthExceeded`] if `token_pos >= self.n_ctx`.
387    /// Returns [`K3Error::TensorBoundsError`] if `layer >= self.layer_count`.
388    pub fn token_kv_slice_mut(
389        &mut self,
390        token_pos: usize,
391        layer: usize,
392    ) -> Result<(&mut [f32], &mut [f32]), K3Error> {
393        if token_pos >= self.n_ctx {
394            return Err(K3Error::ContextLengthExceeded {
395                requested_tokens: token_pos + 1,
396                max_context_tokens: self.n_ctx,
397            });
398        }
399
400        if layer >= self.layer_count {
401            return Err(K3Error::TensorBoundsError {
402                tensor_name: "kv_layer".to_owned(),
403                offset: layer,
404                byte_len: 0,
405                buffer_len: self.layer_count,
406                reason: format!(
407                    "layer index {layer} exceeds model layer count {}",
408                    self.layer_count
409                ),
410            });
411        }
412
413        let block_idx = token_pos / KV_BLOCK_SIZE;
414        let token_in_block = token_pos % KV_BLOCK_SIZE;
415        let head_kv_elements = self.head_count_kv * self.head_dim;
416
417        let token_start =
418            (block_idx * self.elements_per_block) + (token_in_block * self.elements_per_token);
419        let layer_start = token_start + (layer * 2 * head_kv_elements);
420        let key_start = layer_start;
421        let key_end = key_start + head_kv_elements;
422        let val_start = key_end;
423
424        let full = self.as_mut_slice();
425        let (left, right) = full.split_at_mut(val_start);
426        let key_slice = &mut left[key_start..key_end];
427        let val_slice = &mut right[..head_kv_elements];
428
429        Ok((key_slice, val_slice))
430    }
431}

433impl Drop for SingleSlotKvArena {
434    fn drop(&mut self) {
435        let ptr = self.ptr.cast::<libc::c_void>();
436        let bytes = self.aligned_bytes;
437
438        // SAFETY: self.ptr was mapped by mmap and locked by mlock with size aligned_bytes.
439        unsafe {
440            libc::munlock(ptr, bytes);
441            libc::munmap(ptr, bytes);
442        }
443    }
444}

446#[cfg(test)]
447mod tests {
448    use super::*;
449    use k3_core::GgufFile;
450    use k3_core::QuantType;
451    use k3_core::test_helpers::create_synthetic_gguf_v3;
452
453    #[test]
454    fn test_kv_arena_allocation_and_indexing() {
455        let tensor_specs = [("weight", QuantType::F32, &[16][..])];
456        let bytes = create_synthetic_gguf_v3("synthetic", 32, &tensor_specs);
457        let model = GgufFile::parse(&bytes).expect("parse model");
458        let meta = model.metadata();
459
460        // 32 tokens -> 2 blocks of 16 tokens
461        let mut arena = SingleSlotKvArena::new(meta, 32).expect("allocate kv arena");
462        assert_eq!(arena.n_ctx(), 32);
463        assert_eq!(arena.num_blocks(), 2);
464        assert_eq!(arena.allocated_tokens(), 32);
465        assert!(arena.total_elements() > 0);
466        assert!(arena.aligned_bytes() > 0);
467
468        // Test block slicing
469        let block0 = arena.block_slice(0).expect("block 0 slice");
470        assert_eq!(block0.len(), arena.elements_per_block());
471
472        let block1 = arena.block_slice(1).expect("block 1 slice");
473        assert_eq!(block1.len(), arena.elements_per_block());
474
475        let block2_err = arena.block_slice(2).expect_err("block 2 out of bounds");
476        assert!(matches!(block2_err, K3Error::TensorBoundsError { .. }));
477
478        // Test token KV slice access
479        {
480            let (k_mut, v_mut) = arena
481                .token_kv_slice_mut(5, 0)
482                .expect("token 5 layer 0 mutable slice");
483            assert_eq!(k_mut.len(), v_mut.len());
484            k_mut[0] = 3.5;
485            v_mut[0] = 2.5;
486        }
487
488        let (k_ref, v_ref) = arena
489            .token_kv_slice(5, 0)
490            .expect("token 5 layer 0 immutable slice");
491        assert!((k_ref[0] - 3.5).abs() < f32::EPSILON);
492        assert!((v_ref[0] - 2.5).abs() < f32::EPSILON);
493
494        // Token position out of n_ctx bound returns ContextLengthExceeded (ADR-027)
495        let oob_err = arena
496            .token_kv_slice(32, 0)
497            .expect_err("token 32 exceeds n_ctx=32");
498        assert_eq!(
499            oob_err,
500            K3Error::ContextLengthExceeded {
501                requested_tokens: 33,
502                max_context_tokens: 32,
503            }
504        );
505    }
