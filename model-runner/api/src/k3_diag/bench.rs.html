1//! US-03 matrix-forward benchmark: cross-cluster workload splitting and IME isolation.
2//!
3//! Runs the engine's own integer transformer workload ([`k3_engine::forward`]) under a chosen
4//! placement (`--cluster x100|a100`, or `--cluster both --split phase|op|row`) and matrix kernel
5//! (`--ime on|off`), reporting prefill and decode separately against measured hardware ceilings
6//! (ADR-034). No external baseline is used (ADR-033) and no pass/fail speed threshold is applied.

48fn measure_peak_resident_bytes() -> u64 {
49    if let Ok(content) = std::fs::read_to_string("/proc/self/status") {
50        for line in content.lines() {
51            if let Some(rest) = line.strip_prefix("VmHWM:") {
52                if let Some(kb_str) = rest.split_whitespace().next() {
53                    if let Ok(kb) = kb_str.parse::<u64>() {
54                        return kb.saturating_mul(1024);
55                    }
56                }
57            }
58        }
59    }
60    // SAFETY: zeroed rusage struct is safely initialized for C getrusage.
61    let mut usage: libc::rusage = unsafe { std::mem::zeroed() };
62    // SAFETY: standard getrusage query
63    let ret = unsafe { libc::getrusage(libc::RUSAGE_SELF, &raw mut usage) };
64    if ret == 0 {
65        #[cfg(target_os = "macos")]
66        return usage.ru_maxrss as u64;
67        #[cfg(not(target_os = "macos"))]
68        return (usage.ru_maxrss as u64).saturating_mul(1024);
69    }
70    0
71}

245/// Builds the placement plan for a selection, calibrating the row split from measurement.
246fn build_plan(
247    pool: &WorkerPool,
248    affinity: &ResolvedAffinity,
249    kernel: MatKernel,
250) -> Result<(PlacementPlan, Option<RowSplitReport>), K3Error> {
251    match affinity.split_strategy() {
252        None => {
253            let lane = match affinity.cluster {
254                Some(ClusterPreset::A100) => Lane::A100,
255                _ => Lane::X100,
256            };
257            Ok((PlacementPlan::single(lane), None))
258        }
259        Some(SplitStrategy::Phase) => Ok((PlacementPlan::phase(Lane::A100, Lane::X100), None)),
260        Some(SplitStrategy::Op) => Ok((PlacementPlan::op(Lane::A100, Lane::X100), None)),
261        Some(SplitStrategy::Row) => {
262            let calibrate = |shape| -> Result<RowCalibration, K3Error> {
263                let x = calibrate_matmul_rate(pool, Lane::X100, kernel, shape)?;
264                let a = calibrate_matmul_rate(pool, Lane::A100, kernel, shape)?;
265                Ok(RowCalibration {
266                    x100_macs_per_sec: x,
267                    a100_macs_per_sec: a,
268                    x100_share: row_split_share(x, a)?,
269                })
270            };
271            let prefill = calibrate(CalibrationShape::Prefill)?;
272            let decode = calibrate(CalibrationShape::Decode)?;
273            Ok((
274                PlacementPlan::row(prefill.x100_share, decode.x100_share),
275                Some(RowSplitReport { prefill, decode }),
276            ))
277        }
278    }
279}

288fn mean(values: impl Iterator<Item = f64>) -> f64 {
289    let (sum, n) = values.fold((0.0, 0_u32), |(s, n), v| (s + v, n + 1));
290    if n == 0 { 0.0 } else { sum / f64::from(n) }
291}

293fn prefill_tps(s: &ForwardSummary) -> f64 {
294    if s.timing.prefill_ms > 0.0 {
295        s.prompt_tokens as f64 / (s.timing.prefill_ms / 1000.0)
296    } else {
297        0.0
298    }
299}

301fn digest(tokens: &[u32]) -> String {
302    let mut h: u64 = 0xCBF2_9CE4_8422_2325;
303    for t in tokens {
304        for b in t.to_le_bytes() {
305            h ^= u64::from(b);
306            h = h.wrapping_mul(0x0100_0000_01B3);
307        }
308    }
309    format!("{h:016x}")
310}

312fn lanes_of(layout: &PhaseLayout) -> Vec<Lane> {
313    layout
314        .matmul
315        .iter()
316        .filter(|(_, share)| *share > 0.0)
317        .map(|&(lane, _)| lane)
318        .collect()
319}

321#[allow(clippy::too_many_arguments)]
322fn run_arm(
323    config: &DiagConfig,
324    pool: &WorkerPool,
325    affinity: &ResolvedAffinity,
326    model: &ForwardModel,
327    prompt_tokens: &[u32],
328    max_tokens: usize,
329    vocab: usize,
330    kernel: MatKernel,
331    inject_fault: Option<usize>,
332) -> Result<ArmResult, K3Error> {
333    let (plan, row_split) = build_plan(pool, affinity, kernel)?;
334    let repeat = config.repeat.unwrap_or(1);
335    let mut summaries = Vec::with_capacity(repeat);
336    for run in 0..repeat {
337        if is_abort_requested() {
338            break;
339        }
340        let mut s = model.evaluate(pool, kernel, &plan, prompt_tokens, max_tokens)?;
341        if inject_fault == Some(run) {
342            if let Some(first) = s.generated_tokens.first_mut() {
343                *first = (*first + 1) % u32::try_from(vocab.max(1)).unwrap_or(1);
344            }
345        }
346        summaries.push(s);
347    }
348    Ok(ArmResult {
349        kernel,
350        plan,
351        row_split,
352        summaries,
353    })
354}

356/// Executes the US-03 benchmark and writes a table or a `--json` report to `out`.
357///
358/// # Errors
359///
360/// Returns a [`K3Error`] for invalid placement, unsupported IME, cluster-isolation violations or
361/// (after the report is written) a determinism violation.
362#[allow(clippy::too_many_lines)]
363pub(crate) fn run_matrix_benchmark<W: Write>(
364    config: &DiagConfig,
365    mut out: W,
366) -> Result<(), K3Error> {
367    let model_path = config
368        .model_path
369        .as_deref()
370        .ok_or_else(|| cap_err("cli", "missing required --model <path>".to_owned()))?;
371    let prompt_text = config.prompt.as_deref().unwrap_or("hello world");
372    let n_ctx = config.n_ctx.unwrap_or(64);
373    let profile = config.coord.unwrap_or(k3_core::CoordinationProfile::Spin);
374    let repeat = config.repeat.unwrap_or(1);
375
376    // 1. Resolve placement; invalid cluster/split/mask combinations fail here.
377    let affinity = k3_engine::affinity::resolve_affinity_split(
378        config.cluster.as_deref(),
379        config.split.as_ref().map(SplitStrategy::as_str),
380        config.core_mask,
381        config.threads,
382    )?;
383    let pool = WorkerPool::with_coordination(affinity.clone(), profile)?;
384    let before = pool.verify_live_affinities()?;
385
386    // 2. Probe IME on every cluster the pool owns, on that cluster's real cores.
387    let mut ime_supported = HashMap::new();
388    if let Some(&core) = affinity.x100_cores.first() {
389        ime_supported.insert("x100".to_owned(), probe_ime_on_core(core));
390    }
391    if let Some(&core) = affinity.a100_cores.first() {
392        ime_supported.insert("a100".to_owned(), probe_ime_on_core(core));
393    }
394    let ime_on = config.ime == Some(true);
395    if ime_on {
396        let any_ime_cluster = ime_supported.values().any(|&ok| ok);
397        if !any_ime_cluster {
398            let cluster_name = config.cluster.as_deref().unwrap_or("x100");
399            return Err(cap_err(
400                cluster_name,
401                format!(
402                    "cluster '{cluster_name}' does not support SpacemiT IME (XSMTVDotII / smt.vmadot) matrix acceleration; use '--cluster a100' or '--cluster both'"
403                ),
404            ));
405        }
406    }
407
408    // 3. Model residency, tokenization.
409    let inspector = SystemMemoryInspector::new();
410    let (locked_weights, _recipe, _footprint) =
411        LockedModelWeights::open_with_admission(model_path, n_ctx, &inspector)?;
412    let gguf_view = GgufFile::parse(locked_weights.as_bytes())?;
413    let tokenizer = GgufTokenizer::from_gguf(&gguf_view)?;
414    let prompt_tokens = tokenizer.tokenize_prompt(prompt_text, true, n_ctx)?;
415    let vocab = tokenizer.vocab_size();
416    let max_tokens = config
417        .max_tokens
418        .unwrap_or(128)
419        .min(n_ctx.saturating_sub(prompt_tokens.len()));
420    let seed = locked_weights
421        .as_bytes()
422        .iter()
423        .take(1 << 20)
424        .fold(0xCBF2_9CE4_8422_2325_u64, |h, &b| {
425            (h ^ u64::from(b)).wrapping_mul(0x0100_0000_01B3)
426        });
427    let model = ForwardModel::new(seed, vocab);
428    let fault = std::env::var(FAULT_ENV)
429        .ok()
430        .and_then(|v| v.parse::<usize>().ok());
431
432    // 4. Measure the primary arm, plus the IME-off baseline when IME is on.
433    let primary_kernel = if ime_on {
434        MatKernel::Ime
435    } else {
436        MatKernel::Vector
437    };
438    let primary = run_arm(
439        config,
440        &pool,
441        &affinity,
442        &model,
443        &prompt_tokens,
444        max_tokens,
445        vocab,
446        primary_kernel,
447        fault,
448    )?;
449    let baseline = if ime_on {
450        Some(run_arm(
451            config,
452            &pool,
453            &affinity,
454            &model,
455            &prompt_tokens,
456            max_tokens,
457            vocab,
458            MatKernel::Vector,
459            None,
460        )?)
461    } else {
462        None
463    };
464
465    // 5. Cluster isolation: observed execution CPUs and live affinity must stay in assigned lanes.
466    let after = pool.verify_live_affinities()?;
467    let expected_matmul: u64 = primary
468        .plan
469        .prefill
470        .matmul
471        .iter()
472        .chain(&primary.plan.decode.matmul)
473        .filter(|(_, s)| *s > 0.0)
474        .fold(0, |m, &(lane, _)| m | lane_mask(lane));
475    let expected_vector =
476        lane_mask(primary.plan.prefill.vector) | lane_mask(primary.plan.decode.vector);
477    let observed_matmul = primary
478        .summaries
479        .iter()
480        .fold(0, |m, s| m | s.matmul_cpu_mask);
481    let observed_vector = primary
482        .summaries
483        .iter()
484        .fold(0, |m, s| m | s.vector_cpu_mask);
485    if observed_matmul & !expected_matmul != 0 || observed_vector & !expected_vector != 0 {
486        return Err(K3Error::AffinityError {
487            requested_mask: affinity.effective_mask,
488            requested_threads: affinity.threads,
489            reason: format!(
490                "work executed outside its assigned cluster (matmul cpus 0x{observed_matmul:04X} vs 0x{expected_matmul:04X}, vector cpus 0x{observed_vector:04X} vs 0x{expected_vector:04X})"
491            ),
492        });
493    }
494
495    // 6. Determinism across repeats at temperature 0.
496    let tokens0 = primary
497        .summaries
498        .first()
499        .map(|s| s.generated_tokens.clone())
500        .unwrap_or_default();
501    let divergent = primary
502        .summaries
503        .iter()
504        .position(|s| s.generated_tokens != tokens0);
505    let determinism = DeterminismReport {
506        verified: divergent.is_none(),
507        repeat_count: primary.summaries.len(),
508        token_digest: digest(&tokens0),
509        first_divergent_run: divergent,
510    };
511
512    // 7. Ceilings (ADR-034): peak MAC/s for prefill, streaming bandwidth for decode.
513    let first = primary.summaries.first().ok_or_else(|| {
514        cap_err(
515            "cli",
516            "benchmark aborted before any run completed".to_owned(),
517        )
518    })?;
519    let mut peak_cache: HashMap<&'static str, f64> = HashMap::new();
520    let mut bw_cache: HashMap<&'static str, f64> = HashMap::new();
521    let mut ceilings = |layout: &PhaseLayout, kernel: MatKernel| -> Result<(f64, f64), K3Error> {
522        let (mut peak, mut bw) = (0.0, 0.0);
523        for lane in lanes_of(layout) {
524            if !peak_cache.contains_key(lane.as_str()) {
525                peak_cache.insert(lane.as_str(), measure_peak_macs(&pool, lane, kernel)?);
526                bw_cache.insert(lane.as_str(), measure_memory_bandwidth(&pool, lane)?);
527            }
528            peak += peak_cache[lane.as_str()];
529            bw += bw_cache[lane.as_str()];
530        }
531        Ok((peak, bw))
532    };
533    let (prefill_peak, _) = ceilings(&primary.plan.prefill, primary.kernel)?;
534    let (_, decode_bw) = ceilings(&primary.plan.decode, primary.kernel)?;
535
536    let prefill_tps_mean = mean(primary.summaries.iter().map(prefill_tps));
537    let decode_tps_mean = mean(
538        primary
539            .summaries
540            .iter()
541            .map(|s| s.timing.decode_tokens_per_sec),
542    );
543    let prefill_ms = mean(primary.summaries.iter().map(|s| s.timing.prefill_ms));
544    let decode_ms = mean(primary.summaries.iter().map(|s| s.timing.decode_ms));
545    let prefill_macs_per_sec =
546        first.prefill_macs as f64 * prefill_tps_mean / first.prompt_tokens as f64;
547    let decode_bytes_per_sec = first.decode_weight_bytes_per_token as f64 * decode_tps_mean;
548    let pct = |achieved: f64, ceiling: f64| {
549        if ceiling > 0.0 {
550            achieved / ceiling * 100.0
551        } else {
552            0.0
553        }
554    };
555    let prefill_cost_per_token_ms = if prompt_tokens.is_empty() {
556        0.0
557    } else {
558        prefill_ms / prompt_tokens.len() as f64
559    };
560    let mean_decode_steps = mean(
561        primary
562            .summaries
563            .iter()
564            .map(|s| s.timing.decode_steps as f64),
565    );
566    let decode_cost_per_token_ms = if mean_decode_steps > 0.0 {
567        decode_ms / mean_decode_steps
568    } else {
569        0.0
570    };
571    let prefill = PhaseReport {
572        latency_ms: prefill_ms,
573        tokens_per_sec: prefill_tps_mean,
574        cost_per_token_ms: prefill_cost_per_token_ms,
575        ceiling_pct: pct(prefill_macs_per_sec, prefill_peak),
576        ceiling_metric: "peak_macs".to_owned(),
577        ceiling_value: prefill_peak,
578    };
579    let decode = PhaseReport {
580        latency_ms: decode_ms,
581        tokens_per_sec: decode_tps_mean,
582        cost_per_token_ms: decode_cost_per_token_ms,
583        ceiling_pct: pct(decode_bytes_per_sec, decode_bw),
584        ceiling_metric: "memory_bandwidth".to_owned(),
585        ceiling_value: decode_bw,
586    };
587
588    // 8. IME comparison (Scenario 2): the delta is computed from the reported arm throughputs.
589    let ime_comparison = baseline.as_ref().map(|base| {
590        let arm = |r: &ArmResult| ArmThroughput {
591            kernel_path: r.kernel.path_name().to_owned(),
592            prefill_tps: mean(r.summaries.iter().map(prefill_tps)),
593            decode_tps: mean(r.summaries.iter().map(|s| s.timing.decode_tokens_per_sec)),
594            vmadot_issued: r.summaries.iter().map(|s| s.vmadot_issued).sum(),
595        };
596        let (on, off) = (arm(&primary), arm(base));
597        let pct_of = |d: f64, b: f64| if b > 0.0 { d / b * 100.0 } else { 0.0 };
598        let (dp, dd) = (
599            on.prefill_tps - off.prefill_tps,
600            on.decode_tps - off.decode_tps,
601        );
602        ImeComparison {
603            delta_prefill_pct: pct_of(dp, off.prefill_tps),
604            delta_decode_pct: pct_of(dd, off.decode_tps),
605            delta_prefill_tps: dp,
606            delta_decode_tps: dd,
607            numerically_identical: base.summaries.iter().all(|s| s.generated_tokens == tokens0),
608            on,
609            off,
610        }
611    });
612
613    let run_metrics: Vec<RunMetric> = primary
614        .summaries
615        .iter()
616        .enumerate()
617        .map(|(i, s)| RunMetric {
618            run_index: i,
619            ttft_ms: s.timing.ttft_ms,
620            prefill_ms: s.timing.prefill_ms,
621            decode_ms: s.timing.decode_ms,
622            decode_tokens_per_sec: s.timing.decode_tokens_per_sec,
623            total_wall_time_ms: s.timing.total_wall_time_ms,
624            cumulative_barrier_wait_ms: s.timing.cumulative_barrier_wait_ms,
625            barrier_wait_ratio: s.timing.barrier_wait_ratio,
626            compute_utilization_ratio: s.timing.compute_utilization_ratio,
627            tokens: Some(s.generated_tokens.clone()),
628        })
629        .collect();
630    let col = |f: fn(&RunMetric) -> f64| {
631        MetricStats::calculate(&run_metrics.iter().map(f).collect::<Vec<_>>())
632    };
633    let statistics = StatisticsSummary {
634        ttft_ms: col(|r| r.ttft_ms),
635        decode_tokens_per_sec: col(|r| r.decode_tokens_per_sec),
636        cumulative_barrier_wait_ms: col(|r| r.cumulative_barrier_wait_ms),
637        total_wall_time_ms: col(|r| r.total_wall_time_ms),
638        compute_utilization_ratio: col(|r| r.compute_utilization_ratio),
639    };
640
641    let cluster = config
642        .cluster
643        .clone()
644        .unwrap_or_else(|| affinity.cluster.map_or("x100", |c| c.as_str()).to_owned());
645    let split_name = config.split.map(|s| s.as_str().to_owned());
646
647    let footprint = calculate_footprint(locked_weights.metadata(), n_ctx)?;
648    let measured_rss = measure_peak_resident_bytes();
649    let peak_resident_bytes = measured_rss.max(footprint.total_required_bytes);
650
651    let model_name = gguf_view
652        .metadata_value("general.name")
653        .and_then(|v| v.as_str())
654        .map(ToOwned::to_owned)
655        .unwrap_or_else(|| {
656            std::path::Path::new(model_path)
657                .file_stem()
658                .and_then(|s| s.to_str())
659                .unwrap_or("unknown")
660                .to_owned()
661        });
662
663    let model_identity = ModelIdentity {
664        name: model_name,
665        architecture: gguf_view.architecture().to_owned(),
666        file_path: model_path.to_owned(),
667        file_digest: format!("{seed:016x}"),
668    };
669
670    let quantization_level =
671        QuantizationInfo::from_descriptors(locked_weights.metadata().tensors());
672
673    let split_strategy_str = split_name.clone().unwrap_or_else(|| "none".to_owned());
674
675    let cluster_placement = ClusterPlacement {
676        cluster: cluster.clone(),
677        effective_core_mask: format!("0x{:04X}", affinity.effective_mask),
678        allocated_cores: affinity.threads,
679        x100_cores: affinity.x100_cores.len(),
680        a100_cores: affinity.a100_cores.len(),
681        coordination_profile: profile.as_str().to_owned(),
682        split_strategy: split_strategy_str.clone(),
683        ime: if ime_on { "on" } else { "off" }.to_owned(),
684    };
685
686    let allocated_cores = CoreAllocation {
687        total: affinity.threads,
688        total_cores: affinity.threads,
689        x100_cores: affinity.x100_cores.len(),
690        a100_cores: affinity.a100_cores.len(),
691        effective_mask: format!("0x{:04X}", affinity.effective_mask),
692    };
693
694    let static_scheduling_weight = if decode_cost_per_token_ms > 0.0 {
695        decode_cost_per_token_ms
696    } else {
697        prefill_cost_per_token_ms.max(1.0)
698    };
699
700    let execution_cost = ExecutionCost {
701        prefill_cost_per_token_ms,
702        prefill_tokens_per_sec: prefill_tps_mean,
703        prefill_hardware_ceiling_pct: prefill.ceiling_pct,
704        decode_cost_per_token_ms,
705        decode_tokens_per_sec: decode_tps_mean,
706        decode_hardware_ceiling_pct: decode.ceiling_pct,
707        static_scheduling_weight,
708    };
709
710    let memory_footprint = MemoryFootprintProfile {
711        weight_bytes: footprint.weight_bytes,
712        kv_bytes: footprint.kv_bytes,
713        total_required_bytes: footprint.total_required_bytes,
714        peak_resident_bytes,
715    };
716
717    let (
718        bottleneck_classification,
719        recommended_cluster,
720        recommended_split,
721        recommended_ime,
722        rationale,
723        recommendations,
724    ) = {
725        let bottleneck = if decode.ceiling_pct >= prefill.ceiling_pct {
726            "bandwidth-bound"
727        } else {
728            "compute-bound"
729        };
730
731        let (rec_cluster, rec_split, rec_ime) = match cluster.as_str() {
732            "x100" => {
733                if quantization_level.primary == "Q4_K_M" || quantization_level.primary == "Q8_0" {
734                    ("both".to_owned(), Some("phase".to_owned()), true)
735                } else {
736                    ("a100".to_owned(), None, true)
737                }
738            }
739            "a100" => ("a100".to_owned(), None, true),
740            "both" => (
741                "both".to_owned(),
742                config
743                    .split
744                    .map(|s| s.as_str().to_owned())
745                    .or_else(|| Some("phase".to_owned())),
746                true,
747            ),
748            _ => ("both".to_owned(), Some("phase".to_owned()), true),
749        };
750
751        let rat = format!(
752            "Evaluation indicates {bottleneck} execution: prefill achieved {:.1}% of theoretical compute capacity (peak_macs), while decode achieved {:.1}% of memory bandwidth capacity.",
753            prefill.ceiling_pct, decode.ceiling_pct
754        );
755
756        let mut recs = Vec::new();
757        if bottleneck == "bandwidth-bound" {
758            recs.push("Autoregressive token decode is memory bandwidth-bound; consider low-bit quantization (e.g. Q4_K_M) to reduce per-token memory traffic.".to_owned());
759        } else {
760            recs.push("Prompt prefill is compute-bound; offload prefill matrix multiplies to Cluster 1 (A100) with SpacemiT IME matrix acceleration enabled.".to_owned());
761        }
762        if !ime_on {
763            recs.push("SpacemiT IME (XSMTVDotII / smt.vmadot) matrix acceleration is currently disabled; enable '--ime on' to accelerate matrix operations.".to_owned());
764        }
765        if cluster == "x100" {
766            recs.push("Evaluating '--cluster both --split phase' leverages A100 matrix acceleration for prefill while reserving X100 for sequential decode.".to_owned());
767        }
768        recs.push("Operator tuning authority (ADR-032): These recommendations are advisory only; operator configuration remains authoritative.".to_owned());
769
770        (
771            bottleneck.to_owned(),
772            rec_cluster,
773            rec_split,
774            rec_ime,
775            rat,
776            recs,
777        )
778    };
779
780    let tuning_advisory = TuningAdvisory {
781        bottleneck_classification,
782        recommended_cluster,
783        recommended_split,
784        recommended_ime,
785        rationale,
786        recommendations,
787        operator_authority_notice: "Advisory recommendation only per ADR-032. Operator configuration and tuning-profile remain authoritative.".to_owned(),
788    };
789
790    let cost_profile = CostProfile {
791        model_identity,
792        quantization_level,
793        cluster_placement,
794        allocated_cores,
795        split_strategy: split_strategy_str,
796        execution_cost,
797        prefill_cost_per_token_ms,
798        prefill_tokens_per_sec: prefill_tps_mean,
799        prefill_hardware_ceiling_pct: prefill.ceiling_pct,
800        decode_tokens_per_sec: decode_tps_mean,
801        decode_cost_per_token_ms,
802        decode_hardware_ceiling_pct: decode.ceiling_pct,
803        static_scheduling_weight,
804        scheduling_weight: static_scheduling_weight,
805        memory_footprint,
806        weight_bytes: footprint.weight_bytes,
807        kv_bytes: footprint.kv_bytes,
808        total_required_bytes: footprint.total_required_bytes,
809        peak_resident_bytes,
810        tuning_advisory,
811    };
812
813    if let Some(export_path) = &config.export_profile {
814        let json_str = serde_json::to_string_pretty(&cost_profile)
815            .map_err(|e| cap_err("json", format!("failed to serialize cost profile: {e}")))?;
816        if let Some(parent) = std::path::Path::new(export_path).parent() {
817            if !parent.as_os_str().is_empty() && !parent.exists() {
818                let _ = std::fs::create_dir_all(parent);
819            }
820        }
821        std::fs::write(export_path, json_str).map_err(|e| {
822            cap_err(
823                "io",
824                format!("failed to write cost profile to '{export_path}': {e}"),
825            )
826        })?;
827    }
828
829    let report = MatrixBenchReport {
830        mode: "matrix-benchmark".to_owned(),
831        cluster,
832        split: split_name.clone(),
833        split_strategy: split_name,
834        ime: if ime_on { "on" } else { "off" }.to_owned(),
835        ime_enabled: ime_on,
836        kernel_path: primary.kernel.path_name().to_owned(),
837        ime_supported,
838        coordination_profile: profile.as_str().to_owned(),
839        repeat_count: primary.summaries.len(),
840        threads: affinity.threads,
841        effective_core_mask: format!("0x{:04X}", affinity.effective_mask),
842        prompt: prompt_text.to_owned(),
843        prompt_tokens: prompt_tokens.len(),
844        generated_tokens: tokens0.len(),
845        prefill_latency_ms: prefill.latency_ms,
846        decode_latency_ms: decode.latency_ms,
847        prefill_cost_per_token_ms,
848        decode_cost_per_token_ms,
849        prefill_tps: prefill.tokens_per_sec,
850        decode_tps: decode.tokens_per_sec,
851        prefill_hardware_ceiling_pct: prefill.ceiling_pct,
852        decode_hardware_ceiling_pct: decode.ceiling_pct,
853        total_wall_time_ms: statistics.total_wall_time_ms.mean,
854        cost_profile,
855        prefill,
856        decode,
857        ime_vmadot_issued: primary.summaries.iter().map(|s| s.vmadot_issued).sum(),
858        ime_delta_prefill_tps: ime_comparison.as_ref().map(|c| c.delta_prefill_tps),
859        ime_delta_decode_tps: ime_comparison.as_ref().map(|c| c.delta_decode_tps),
860        ime_comparison,
861        row_split_partition_ratio: primary.row_split.as_ref().map(|r| r.prefill.x100_share),
862        row_split: primary.row_split.clone(),
863        placement_prefill: placement_report(&primary.plan.prefill),
864        placement_decode: placement_report(&primary.plan.decode),
865        matmul_cpu_mask: observed_matmul,
866        vector_cpu_mask: observed_vector,
867        cluster_isolation_verified: true,
868        workers_before: worker_reports(&before),
869        workers_after: worker_reports(&after),
870        determinism_verified: determinism.verified,
871        determinism: determinism.clone(),
872        tokens: tokens0,
873        runs: run_metrics,
874        statistics,
875    };
876
877    if config.json {
878        let json =
879            serde_json::to_string_pretty(&report).map_err(|e| cap_err("json", e.to_string()))?;
880        let _ = writeln!(out, "{json}");
881    } else {
882        write_table(&mut out, &report);
883    }
884    let _ = out.flush();
885
886    if let Some(run) = determinism.first_divergent_run {
887        return Err(cap_err(
888            "determinism",
889            format!(
890                "determinism violation: run {} produced a different token sequence than run 0 at temperature 0 (repeat {repeat})",
891                run + 1
892            ),
893        ));
894    }
895    Ok(())
896}
897
898fn write_table<W: Write>(out: &mut W, r: &MatrixBenchReport) {
899    let rule = "==========================================================";
900    let _ = writeln!(
901        out,
902        "{rule}\n k3-diag: Cluster Placement & IME Benchmark\n{rule}"
903    );
904    let _ = writeln!(out, "Cluster              : {}", r.cluster);
905    if let Some(split) = &r.split {
906        let _ = writeln!(out, "Split Policy         : {split}");
907    }
908    let _ = writeln!(out, "IME                  : {} ({})", r.ime, r.kernel_path);
909    let _ = writeln!(out, "Worker Threads       : {}", r.threads);
910    let _ = writeln!(out, "Effective Core Mask  : {}", r.effective_core_mask);
911    let _ = writeln!(
912        out,
913        "Prompt / Generated   : {} / {} tokens",
914        r.prompt_tokens, r.generated_tokens
915    );
916    let _ = writeln!(
917        out,
918        "Determinism          : {}",
919        if r.determinism_verified {
920            "verified (bit-identical across all runs at temp 0)"
921        } else {
922            "VIOLATION"
923        }
924    );
925    let _ = writeln!(
926        out,
927        "Prefill              : {:.3} ms, {:.1} tok/s, {:.3} ms/token, {:.1}% of {} ceiling",
928        r.prefill.latency_ms,
929        r.prefill.tokens_per_sec,
930        r.prefill.cost_per_token_ms,
931        r.prefill.ceiling_pct,
932        r.prefill.ceiling_metric
933    );
934    let _ = writeln!(
935        out,
936        "Decode               : {:.3} ms, {:.1} tok/s, {:.3} ms/token, {:.1}% of {} ceiling",
937        r.decode.latency_ms,
938        r.decode.tokens_per_sec,
939        r.decode.cost_per_token_ms,
940        r.decode.ceiling_pct,
941        r.decode.ceiling_metric
942    );
943    if let Some(rs) = &r.row_split {
944        let _ = writeln!(
945            out,
946            "Row Split (X100 share): prefill {:.3}, decode {:.3} (from measured throughput)",
947            rs.prefill.x100_share, rs.decode.x100_share
948        );
949    }
950    if let Some(c) = &r.ime_comparison {
951        let _ = writeln!(
952            out,
953            "IME delta (on - off) : prefill {:+.1} tok/s ({:+.1}%), decode {:+.1} tok/s ({:+.1}%)",
954            c.delta_prefill_tps, c.delta_prefill_pct, c.delta_decode_tps, c.delta_decode_pct
955        );
956    }
957    let _ = writeln!(
958        out,
959        "----------------------------------------------------------"
960    );
961    let _ = writeln!(out, "Cost Profile Baseline (US-04):");
962    let _ = writeln!(
963        out,
964        "  Prefill Cost       : {:.3} ms/token ({:.1}% of peak_macs)",
965        r.cost_profile.prefill_cost_per_token_ms, r.cost_profile.prefill_hardware_ceiling_pct
966    );
967    let _ = writeln!(
968        out,
969        "  Decode Cost        : {:.3} ms/token ({:.1}% of memory_bandwidth)",
970        r.cost_profile.decode_cost_per_token_ms, r.cost_profile.decode_hardware_ceiling_pct
971    );
972    let _ = writeln!(
973        out,
974        "  Static Sched Weight: {:.3}",
975        r.cost_profile.static_scheduling_weight
976    );
977    let _ = writeln!(
978        out,
979        "  Peak Resident RAM  : {} bytes (weights: {}, kv: {})",
980        r.cost_profile.peak_resident_bytes, r.cost_profile.weight_bytes, r.cost_profile.kv_bytes
981    );
982    let _ = writeln!(
983        out,
984        "  Tuning Advisory    : {} (Cluster: {}, IME: {})",
985        r.cost_profile.tuning_advisory.bottleneck_classification,
986        r.cost_profile.tuning_advisory.recommended_cluster,
987        if r.cost_profile.tuning_advisory.recommended_ime {
988            "on"
989        } else {
990            "off"
991        }
992    );
993    let _ = writeln!(out, "{rule}");
