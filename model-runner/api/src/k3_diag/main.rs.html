34/// Global atomic flag set when `SIGINT` or `SIGTERM` is received so the decode loop
35/// exits cleanly and triggers RAII `Drop` guards (`munlock`, `munmap`, `flock(LOCK_UN)`).
36static ABORT_REQUESTED: AtomicBool = AtomicBool::new(false);
37
38extern "C" fn handle_termination_signal(_sig: libc::c_int) {
39    ABORT_REQUESTED.store(true, Ordering::SeqCst);
40}
41
42fn install_signal_handlers() {
43    // SAFETY: handle_termination_signal is an async-signal-safe extern "C" function
44    // that only performs an atomic relaxed/seqcst store on a static AtomicBool.
45    unsafe {
46        libc::signal(
47            libc::SIGINT,
48            handle_termination_signal as *const () as libc::sighandler_t,
49        );
50        libc::signal(
51            libc::SIGTERM,
52            handle_termination_signal as *const () as libc::sighandler_t,
53        );
54    }
55}

57/// Returns `true` if an operator `SIGINT` or `SIGTERM` abort was requested.
58#[must_use]
59pub fn is_abort_requested() -> bool {
60    ABORT_REQUESTED.load(Ordering::SeqCst)
61}

117impl DiagConfig {
118    /// Parses CLI arguments from an iterator of string arguments.
119    ///
120    /// # Errors
121    ///
122    /// Returns a descriptive error message if arguments are malformed or missing values.
123    pub fn parse_args<I>(mut args: I) -> Result<Option<Self>, String>
124    where
125        I: Iterator<Item = String>,
126    {
127        let mut config = Self::default();
128
129        // Skip binary name if present
130        let _ = args.next();
131
132        while let Some(arg) = args.next() {
133            match arg.as_str() {
134                "--help" | "-h" => {
135                    print_usage();
136                    return Ok(None);
137                }
138                "--version" | "-v" => {
139                    println!("k3-diag {}", env!("CARGO_PKG_VERSION"));
140                    return Ok(None);
141                }
142                "--model" | "-m" => {
143                    config.model_path = Some(
144                        args.next()
145                            .ok_or_else(|| "--model requires a file path".to_owned())?,
146                    );
147                }
148                "--prompt" | "-p" => {
149                    config.prompt = Some(
150                        args.next()
151                            .ok_or_else(|| "--prompt requires an input string".to_owned())?,
152                    );
153                }
154                "--cluster" => {
155                    let cluster = args
156                        .next()
157                        .ok_or_else(|| "--cluster requires 'both', 'x100' or 'a100'".to_owned())?;
158                    if cluster != "x100" && cluster != "a100" && cluster != "both" {
159                        return Err(format!(
160                            "invalid cluster '{cluster}': expected 'both', 'x100' or 'a100'"
161                        ));
162                    }
163                    config.cluster = Some(cluster);
164                }
165                "--split" => {
166                    let val = args
167                        .next()
168                        .ok_or_else(|| "--split requires 'phase', 'op' or 'row'".to_owned())?;
169                    config.split = Some(match val.as_str() {
170                        "phase" => SplitStrategy::Phase,
171                        "op" => SplitStrategy::Op,
172                        "row" => SplitStrategy::Row,
173                        _ => {
174                            return Err(format!(
175                                "invalid split strategy '{val}': expected 'phase', 'op' or 'row'"
176                            ));
177                        }
178                    });
179                }
180                "--ime" => {
181                    let val = args
182                        .next()
183                        .ok_or_else(|| "--ime requires 'on' or 'off'".to_owned())?;
184                    config.ime = Some(match val.as_str() {
185                        "on" => true,
186                        "off" => false,
187                        _ => {
188                            return Err(format!(
189                                "invalid IME toggle '{val}': expected 'on' or 'off'"
190                            ));
191                        }
192                    });
193                }
194                "--threads" | "-t" => {
195                    let val = args
196                        .next()
197                        .ok_or_else(|| "--threads requires an integer count".to_owned())?;
198                    let n = val
199                        .parse::<usize>()
200                        .map_err(|_| format!("invalid thread count '{val}'"))?;
201                    if n == 0 {
202                        return Err("thread count must be at least 1".to_owned());
203                    }
204                    config.threads = Some(n);
205                }
206                "--cores" => {
207                    let val = args
208                        .next()
209                        .ok_or_else(|| "--cores requires a hex or decimal mask".to_owned())?;
210                    let mask = if let Some(hex) =
211                        val.strip_prefix("0x").or_else(|| val.strip_prefix("0X"))
212                    {
213                        u64::from_str_radix(hex, 16)
214                    } else {
215                        val.parse::<u64>()
216                    }
217                    .map_err(|_| format!("invalid core mask '{val}'"))?;
218                    config.core_mask = Some(mask);
219                }
220                "--n-ctx" => {
221                    let val = args
222                        .next()
223                        .ok_or_else(|| "--n-ctx requires a token limit integer".to_owned())?;
224                    let n = val
225                        .parse::<usize>()
226                        .map_err(|_| format!("invalid n-ctx '{val}'"))?;
227                    config.n_ctx = Some(n);
228                }
229                "--max-tokens" => {
230                    let val = args
231                        .next()
232                        .ok_or_else(|| "--max-tokens requires an integer".to_owned())?;
233                    let n = val
234                        .parse::<usize>()
235                        .map_err(|_| format!("invalid max-tokens '{val}'"))?;
236                    config.max_tokens = Some(n);
237                }
238                "--coord" => {
239                    let val = args
240                        .next()
241                        .ok_or_else(|| "--coord requires 'spin', 'futex', or 'rayon'".to_owned())?;
242                    let profile = val.parse::<CoordinationProfile>().map_err(|_| {
243                        format!(
244                            "invalid coordination profile '{val}': expected 'spin', 'futex', or 'rayon'"
245                        )
246                    })?;
247                    config.coord = Some(profile);
248                }
249                "--repeat" => {
250                    let val = args.next().ok_or_else(|| {
251                        "--repeat requires an integer count of at least 3".to_owned()
252                    })?;
253                    let n = val.parse::<isize>().map_err(|_| {
254                        format!("invalid repeat count '{val}': expected integer of at least 3")
255                    })?;
256                    if n < 3 {
257                        return Err(format!(
258                            "--repeat must be an integer of at least 3 (got {val})"
259                        ));
260                    }
261                    config.repeat = Some(n as usize);
262                }
263                "--json" => {
264                    config.json = true;
265                }
266                "--export-profile" => {
267                    let path = args
268                        .next()
269                        .ok_or_else(|| "--export-profile requires a file path".to_owned())?;
270                    if path.trim().is_empty() {
271                        return Err("--export-profile requires a non-empty file path".to_owned());
272                    }
273                    config.export_profile = Some(path);
274                }
275                "--summary" | "--bench" => {
276                    config.summary = true;
277                }
278                unknown => {
279                    return Err(format!("unknown option '{unknown}'"));
280                }
281            }
282        }
283
284        // Cluster-boundary rules (US-03): a mixed-cluster mask is allowed only through an
285        // explicit split policy, and a split policy only makes sense across both clusters.
286        if config.split.is_some() && config.cluster.as_deref() != Some("both") {
287            return Err("--split requires --cluster both".to_owned());
288        }
289        if config.cluster.as_deref() == Some("both") && config.split.is_none() {
290            return Err(
291                "--cluster both requires an explicit --split <phase|op|row> policy".to_owned(),
292            );
293        }
294
295        Ok(Some(config))
296    }
297
298    /// Returns `true` when the matrix-forward benchmark / hardware ceiling profiling should run.
299    #[must_use]
300    pub fn is_matrix_benchmark(&self) -> bool {
301        self.split.is_some()
302            || self.ime.is_some()
303            || self.cluster.as_deref() == Some("both")
304            || self.export_profile.is_some()
305            || self.summary
306            || (self.cluster.is_some() && self.json)
307            || (self.cluster.is_some() && self.core_mask.is_none())
308    }
309}

311/// Descriptive statistics for a series of continuous numerical samples ($N \ge 1$).
312#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
313pub struct MetricStats {
314    pub mean: f64,
315    pub min: f64,
316    pub max: f64,
317    pub stddev: f64,
318}
319
320impl MetricStats {
321    /// Computes sample mean, min, max, and sample standard deviation with Bessel's correction ($N-1$).
322    #[must_use]
323    #[allow(clippy::cast_precision_loss)]
324    pub fn calculate(samples: &[f64]) -> Self {
325        if samples.is_empty() {
326            return Self {
327                mean: 0.0,
328                min: 0.0,
329                max: 0.0,
330                stddev: 0.0,
331            };
332        }
333        let n = samples.len() as f64;
334        let mean = samples.iter().sum::<f64>() / n;
335        let min = samples.iter().copied().fold(f64::INFINITY, f64::min);
336        let max = samples.iter().copied().fold(f64::NEG_INFINITY, f64::max);
337
338        let stddev = if samples.len() < 2 {
339            0.0
340        } else {
341            let variance = samples.iter().map(|&x| (x - mean).powi(2)).sum::<f64>() / (n - 1.0);
342            let s = variance.sqrt();
343            if s.is_nan() || s < 1e-12 { 0.0 } else { s }
344        };
345
346        Self {
347            mean,
348            min,
349            max,
350            stddev,
351        }
352    }
353}

394/// Maps domain [`K3Error`] variants to distinct process exit codes per US-01.7.1 and US-02.
395#[must_use]
396pub const fn exit_code_for_error(err: &K3Error) -> u8 {
397    match err {
398        K3Error::CapabilityMismatch { .. } => 10,
399        K3Error::InsufficientLockableMemory { .. } => 11,
400        K3Error::ContextLengthExceeded { .. } => 12,
401        K3Error::AffinityError { .. } => 13,
402        K3Error::TensorBoundsError { .. } => 14,
403        K3Error::RoutingUnavailable { .. } => 15,
404        K3Error::QueueSaturated { .. } => 23,
405    }
406}

432/// Executes multi-iteration evaluation pipeline for multi-core coordination & jitter measurement (US-02).
433pub fn run_multi_run_benchmark<W: Write>(config: &DiagConfig, mut out: W) -> Result<(), K3Error> {
434    let model_path = config
435        .model_path
436        .as_deref()
437        .ok_or_else(|| K3Error::CapabilityMismatch {
438            architecture: "cli".to_owned(),
439            reason: "missing required --model <path>".to_owned(),
440        })?;
441    let prompt_text = config.prompt.as_deref().unwrap_or("hello world");
442    let n_ctx = config.n_ctx.unwrap_or(64);
443    let max_tokens = config.max_tokens.unwrap_or(128);
444    let repeat_count = config.repeat.unwrap_or(3);
445    let profile = config.coord.unwrap_or(CoordinationProfile::Spin);
446
447    // 1. Resolve cluster & core affinity and spawn pinned worker pool with selected coordination profile
448    let resolved_affinity =
449        resolve_affinity(config.cluster.as_deref(), config.core_mask, config.threads)?;
450    let effective_core_mask = format!("0x{:04X}", resolved_affinity.effective_mask);
451    let thread_count = resolved_affinity.threads;
452
453    let worker_pool = WorkerPool::with_coordination(resolved_affinity, profile)?;
454    worker_pool.verify_live_affinities()?;
455
456    // 2. Open model with flock(LOCK_SH), GGUF parse, capability filter, pre-flight RAM admission, and mlock
457    let inspector = SystemMemoryInspector::new();
458    let (locked_weights, recipe, _footprint) =
459        LockedModelWeights::open_with_admission(model_path, n_ctx, &inspector)?;
460
461    // 3. Extract GGUF vocabulary/template and tokenize prompt in-process
462    let gguf_view = GgufFile::parse(locked_weights.as_bytes())?;
463    let tokenizer = GgufTokenizer::from_gguf(&gguf_view)?;
464    let prompt_tokens = tokenizer.tokenize_prompt(prompt_text, true, n_ctx)?;
465
466    // 4. Multi-Run Execution Loop
467    let mut run_metrics: Vec<RunMetric> = Vec::with_capacity(repeat_count);
468    let mut all_tokens: Vec<Vec<u32>> = Vec::with_capacity(repeat_count);
469    let mut engine = SinglePromptEngine::new();
470
471    for run_idx in 0..repeat_count {
472        if is_abort_requested() {
473            break;
474        }
475
476        let mut kv_arena = SingleSlotKvArena::new(locked_weights.metadata(), n_ctx)?;
477        let summary = engine.evaluate_prompt_with_pool(
478            recipe,
479            &locked_weights,
480            &mut kv_arena,
481            Some(&worker_pool),
482            &prompt_tokens,
483            tokenizer.vocab_size(),
484            max_tokens,
485            |_tok| {
486                if is_abort_requested() {
487                    Ok(StreamAction::Stop)
488                } else {
489                    Ok(StreamAction::Continue)
490                }
491            },
492        )?;
493
494        let run_metric = RunMetric {
495            run_index: run_idx,
496            ttft_ms: summary.timing.ttft_ms,
497            prefill_ms: summary.timing.prefill_ms,
498            decode_ms: summary.timing.decode_ms,
499            decode_tokens_per_sec: summary.timing.decode_tokens_per_sec,
500            total_wall_time_ms: summary.timing.total_wall_time_ms,
501            cumulative_barrier_wait_ms: summary.timing.cumulative_barrier_wait_ms,
502            barrier_wait_ratio: summary.timing.barrier_wait_ratio,
503            compute_utilization_ratio: summary.timing.compute_utilization_ratio,
504            tokens: Some(summary.generated_tokens.clone()),
505        };
506
507        all_tokens.push(summary.generated_tokens);
508        run_metrics.push(run_metric);
509    }
510
511    // 5. Re-verify worker affinities post-execution to guarantee zero cross-cluster migration
512    worker_pool.verify_live_affinities()?;
513
514    // 6. Assert Output Determinism: Bit-identical tokens across repeat runs at temp 0
515    let mut determinism_verified = true;
516    if all_tokens.len() >= 2 {
517        for i in 1..all_tokens.len() {
518            if all_tokens[i] != all_tokens[0] {
519                determinism_verified = false;
520                break;
521            }
522        }
523    }
524
525    // 7. Calculate Descriptive Statistics
526    let ttft_samples: Vec<f64> = run_metrics.iter().map(|r| r.ttft_ms).collect();
527    let throughput_samples: Vec<f64> = run_metrics
528        .iter()
529        .map(|r| r.decode_tokens_per_sec)
530        .collect();
531    let barrier_samples: Vec<f64> = run_metrics
532        .iter()
533        .map(|r| r.cumulative_barrier_wait_ms)
534        .collect();
535    let wall_time_samples: Vec<f64> = run_metrics.iter().map(|r| r.total_wall_time_ms).collect();
536    let util_samples: Vec<f64> = run_metrics
537        .iter()
538        .map(|r| r.compute_utilization_ratio)
539        .collect();
540
541    let statistics = StatisticsSummary {
542        ttft_ms: MetricStats::calculate(&ttft_samples),
543        decode_tokens_per_sec: MetricStats::calculate(&throughput_samples),
544        cumulative_barrier_wait_ms: MetricStats::calculate(&barrier_samples),
545        total_wall_time_ms: MetricStats::calculate(&wall_time_samples),
546        compute_utilization_ratio: MetricStats::calculate(&util_samples),
547    };
548
549    // 8. Output Formatting: JSON or Table
550    if config.json {
551        let report = MultiRunReport {
552            coordination_profile: profile.as_str().to_owned(),
553            repeat_count: run_metrics.len(),
554            threads: thread_count,
555            effective_core_mask,
556            determinism_verified,
557            prompt: prompt_text.to_owned(),
558            runs: run_metrics,
559            statistics,
560        };
561        let json_str =
562            serde_json::to_string_pretty(&report).map_err(|e| K3Error::CapabilityMismatch {
563                architecture: "json".to_owned(),
564                reason: e.to_string(),
565            })?;
566        let _ = writeln!(out, "{json_str}");
567    } else {
568        let _ = writeln!(
569            out,
570            "=========================================================="
571        );
572        let _ = writeln!(
573            out,
574            " k3-diag: Multi-Core Synchronization & Jitter Benchmark"
575        );
576        let _ = writeln!(
577            out,
578            "=========================================================="
579        );
580        let _ = writeln!(out, "Coordination Profile : {}", profile.as_str());
581        let _ = writeln!(out, "Repeat Iterations    : {}", run_metrics.len());
582        let _ = writeln!(out, "Worker Threads       : {thread_count}");
583        let _ = writeln!(out, "Effective Core Mask  : {effective_core_mask}");
584        let _ = writeln!(
585            out,
586            "Output Determinism   : {}",
587            if determinism_verified {
588                "verified (bit-identical across all runs at temp 0)"
589            } else {
590                "MISMATCH"
591            }
592        );
593        let _ = writeln!(
594            out,
595            "----------------------------------------------------------"
596        );
597        let _ = writeln!(out, "Iteration Breakdown:");
598        let _ = writeln!(
599            out,
600            "  Run | TTFT (ms) | Decode (tokens/s) | Barrier Wait (ms) | Total Wall (ms)"
601        );
602        for r in &run_metrics {
603            let _ = writeln!(
604                out,
605                "  {:3} | {:9.2} | {:17.2} | {:17.2} | {:15.2}",
606                r.run_index + 1,
607                r.ttft_ms,
608                r.decode_tokens_per_sec,
609                r.cumulative_barrier_wait_ms,
610                r.total_wall_time_ms
611            );
612        }
613        let _ = writeln!(
614            out,
615            "----------------------------------------------------------"
616        );
617        let _ = writeln!(out, "Statistical Jitter Summary (Mean, Min, Max, StdDev):");
618        let _ = writeln!(
619            out,
620            "  Metric                      |      Mean |       Min |       Max |    StdDev"
621        );
622        let _ = writeln!(
623            out,

660            statistics.compute_utilization_ratio.stddev

662        let _ = writeln!(
663            out,
664            "=========================================================="
665        );
666    }
667    let _ = out.flush();

669    Ok(())

672/// Executes single-prompt standalone diagnostic evaluation pipeline with real-time streaming (US-01).
673pub fn run_single_eval_session<W: Write>(config: &DiagConfig, mut out: W) -> Result<(), K3Error> {
674    let model_path = config
675        .model_path
676        .as_deref()
677        .ok_or_else(|| K3Error::CapabilityMismatch {
678            architecture: "cli".to_owned(),
679            reason: "missing required --model <path>".to_owned(),
680        })?;
681    let prompt_text = config.prompt.as_deref().unwrap_or("hello world");
682    let n_ctx = config.n_ctx.unwrap_or(64);
683    let max_tokens = config.max_tokens.unwrap_or(128);
684    let profile = config.coord.unwrap_or(CoordinationProfile::Spin);
685
686    // 1. Resolve cluster & core affinity and spawn pinned worker pool (US-01.5, Scenario 2)
687    let resolved_affinity =
688        resolve_affinity(config.cluster.as_deref(), config.core_mask, config.threads)?;
689    let worker_pool = WorkerPool::with_coordination(resolved_affinity, profile)?;
690    worker_pool.verify_live_affinities()?;
691
692    // 2. Open model with flock(LOCK_SH), GGUF parse, capability filter, pre-flight RAM admission, and mlock (US-01.2, US-01.3, Scenario 1 & 3)
693    let inspector = SystemMemoryInspector::new();
694    let (locked_weights, recipe, _footprint) =
695        LockedModelWeights::open_with_admission(model_path, n_ctx, &inspector)?;
696
697    // 3. Allocate and mlock single-slot 16-token paged KV cache arena (US-01.3.3)
698    let mut kv_arena = SingleSlotKvArena::new(locked_weights.metadata(), n_ctx)?;
699
700    // 4. Extract GGUF vocabulary/template and tokenize prompt in-process (US-01.4)
701    let gguf_view = GgufFile::parse(locked_weights.as_bytes())?;
702    let tokenizer = GgufTokenizer::from_gguf(&gguf_view)?;
703    let prompt_tokens = tokenizer.tokenize_prompt(prompt_text, true, n_ctx)?;
704
705    // 5. Execute isolated single-prompt forward pass and stream tokens unbuffered (US-01.6, US-01.7.2)
706    let mut engine = SinglePromptEngine::new();
707    let mut detokenizer = tokenizer.streaming_detokenizer();
708
709    engine.evaluate_prompt_with_pool(
710        recipe,
711        &locked_weights,
712        &mut kv_arena,
713        Some(&worker_pool),
714        &prompt_tokens,
715        tokenizer.vocab_size(),
716        max_tokens,
717        |token_id| {
718            if is_abort_requested() {
719                return Ok(StreamAction::Stop);
720            }
721
722            match detokenizer.push_token(token_id)? {
723                DetokenizeStep::Emit(piece) => {
724                    let _ = out.write_all(piece.as_bytes());
725                    let _ = out.flush();
726                    Ok(StreamAction::Continue)
727                }
728                DetokenizeStep::Hold => Ok(StreamAction::Continue),
729                DetokenizeStep::Stop(prefix) => {
730                    if !prefix.is_empty() {
731                        let _ = out.write_all(prefix.as_bytes());
732                        let _ = out.flush();
733                    }
734                    Ok(StreamAction::Stop)
735                }
736            }
737        },
738    )?;
739
740    if !detokenizer.is_stopped() && !is_abort_requested() {
741        let trailing = detokenizer.finish();
742        if !trailing.is_empty() {
743            let _ = out.write_all(trailing.as_bytes());
744            let _ = out.flush();
745        }
746    }
747
748    let _ = out.write_all(b"\n");
749    let _ = out.flush();
750
751    // Re-verify worker affinities before teardown to guarantee zero cross-cluster drift during evaluation
752    worker_pool.verify_live_affinities()?;
753
754    // RAII Drop guards for `kv_arena`, `locked_weights`, and `worker_pool` execute deterministically here,
755    // releasing mlock, munmap, and flock(LOCK_UN).
756    Ok(())
757}

759/// Executes diagnostic session dispatcher: routes to multi-run benchmark if `--repeat` is specified,
760/// or single-eval streaming pipeline if `--repeat` is absent.
761pub fn run_diagnostic_session<W: Write>(config: &DiagConfig, out: W) -> Result<(), K3Error> {
762    if config.is_matrix_benchmark() {
763        bench::run_matrix_benchmark(config, out)
764    } else if config.repeat.is_some() {
765        run_multi_run_benchmark(config, out)
766    } else {
767        run_single_eval_session(config, out)
768    }
769}
770
771fn main() -> ExitCode {
772    install_signal_handlers();
773
774    let args = env::args();
775    let config = match DiagConfig::parse_args(args) {
776        Ok(Some(cfg)) => cfg,
777        Ok(None) => return ExitCode::SUCCESS,
778        Err(err) => {
779            eprintln!("[error] {err}");
780            eprintln!("Run 'k3-diag --help' for usage instructions.");
781            return ExitCode::from(2);
782        }
783    };
784
785    if config.model_path.is_none() {
786        eprintln!("[error] missing required argument: --model <path>");
787        eprintln!("Run 'k3-diag --help' for usage instructions.");
788        return ExitCode::from(2);
789    }
790
791    let stdout = io::stdout();
792    if let Err(err) = run_diagnostic_session(&config, stdout) {
793        eprintln!("{err}");
794        return ExitCode::from(exit_code_for_error(&err));
795    }
796
797    ExitCode::SUCCESS
798}
