11/// GGUF token type classification (`tokenizer.ggml.token_type`).
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum TokenType {
14    /// Normal vocabulary piece (`1`).
15    Normal,
16    /// Unknown token (`2`).
17    Unknown,
18    /// Control / special token (`3`).
19    Control,
20    /// User-defined token (`4`).
21    UserDefined,
22    /// Unused token (`5`).
23    Unused,
24    /// Raw byte fallback token `<0xXX>` (`6`).
25    Byte,
26}
27
28impl TokenType {
29    /// Maps a GGUF `tokenizer.ggml.token_type` integer to [`TokenType`].
30    #[must_use]
31    pub const fn from_gguf_i32(val: i32) -> Self {
32        match val {
33            2 => Self::Unknown,
34            3 => Self::Control,
35            4 => Self::UserDefined,
36            5 => Self::Unused,
37            6 => Self::Byte,
38            _ => Self::Normal,
39        }
40    }
41}

43/// Special control token IDs and stop sequences extracted from model metadata.
44#[derive(Debug, Clone, PartialEq, Eq, Default)]
45pub struct SpecialTokens {
46    /// Beginning-of-sequence token ID (`tokenizer.ggml.bos_token_id`).
47    pub bos_token_id: Option<u32>,
48    /// End-of-sequence token ID (`tokenizer.ggml.eos_token_id`).
49    pub eos_token_id: Option<u32>,
50    /// End-of-turn token ID (`tokenizer.ggml.eot_token_id`).
51    pub eot_token_id: Option<u32>,
52    /// Padding token ID (`tokenizer.ggml.padding_token_id`).
53    pub padding_token_id: Option<u32>,
54    /// Whether the prompt should automatically prepend `bos_token_id`.
55    pub add_bos_token: bool,
56    /// Explicit stop sequence token IDs.
57    pub stop_tokens: Vec<u32>,
58    /// Text-level stop sequences (e.g., `"<|im_end|>"`, `"</s>"`).
59    pub stop_sequences: Vec<String>,
60}
61
62impl SpecialTokens {
63    /// Checks if a given token ID is considered a sequence stop token.
64    #[must_use]
65    pub fn is_stop(&self, token_id: u32) -> bool {
66        if self.eos_token_id == Some(token_id) || self.eot_token_id == Some(token_id) {
67            return true;
68        }
69        self.stop_tokens.contains(&token_id)
70    }
71}

73/// Architecture-aware chat template formatter (ADR-023, ADR-025).
74#[derive(Debug, Clone, PartialEq, Eq)]
75pub struct ChatTemplate {
76    raw_template: Option<String>,
77}
78
79impl ChatTemplate {
80    /// Creates a new [`ChatTemplate`] from an optional GGUF `tokenizer.chat_template` string.
81    #[must_use]
82    pub const fn new(raw_template: Option<String>) -> Self {
83        Self { raw_template }
84    }
85
86    /// Returns the raw template string from GGUF metadata, if present.
87    #[must_use]
88    pub fn raw_template(&self) -> Option<&str> {
89        self.raw_template.as_deref()
90    }
91
92    /// Formats an operator user prompt according to the model's chat template rules.
93    #[must_use]
94    pub fn format_prompt(&self, user_prompt: &str) -> String {
95        let Some(tmpl) = self.raw_template.as_deref() else {
96            return user_prompt.to_owned();
97        };
98
99        if tmpl.contains("{{prompt}}") {
100            return tmpl.replace("{{prompt}}", user_prompt);
101        }
102        if tmpl.contains("{{content}}") {
103            return tmpl.replace("{{content}}", user_prompt);
104        }
105        if tmpl.contains("<|im_start|>") {
106            return format!("<|im_start|>user\n{user_prompt}<|im_end|>\n<|im_start|>assistant\n");
107        }
108        if tmpl.contains("[INST]") {
109            return format!("[INST] {user_prompt} [/INST]");
110        }
111
112        user_prompt.to_owned()
113    }
114}

116/// Tokenizer abstraction for text encoding and real-time token streaming.
117pub trait Tokenizer: Send + Sync + std::fmt::Debug {
118    /// Encodes raw text into a sequence of token IDs.
119    ///
120    /// # Errors
121    ///
122    /// Returns [`K3Error::CapabilityMismatch`] if characters cannot be encoded.
123    fn encode(&self, text: &str) -> Result<Vec<u32>, K3Error>;
124
125    /// Decodes a sequence of token IDs back into text.
126    ///
127    /// # Errors
128    ///
129    /// Returns [`K3Error::CapabilityMismatch`] if token IDs cannot be resolved.
130    fn decode(&self, tokens: &[u32]) -> Result<String, K3Error>;
131
132    /// Decodes a single token ID into its text fragment for streaming output.
133    ///
134    /// # Errors
135    ///
136    /// Returns [`K3Error::CapabilityMismatch`] if the token ID is not recognized.
137    fn decode_piece(&self, token_id: u32) -> Result<Option<String>, K3Error>;
138
139    /// Returns the active special tokens configuration.
140    fn special_tokens(&self) -> &SpecialTokens;
141}

143/// Baseline in-memory vocabulary tokenizer.
144#[derive(Debug, Clone, Default)]
145pub struct SimpleTokenizer {
146    token_to_id: HashMap<String, u32>,
147    id_to_token: HashMap<u32, String>,
148    special_tokens: SpecialTokens,
149}
150
151impl SimpleTokenizer {
152    /// Creates an empty tokenizer with default special tokens.
153    #[must_use]
154    pub fn new() -> Self {
155        Self::default()
156    }
157
158    /// Sets the special tokens configuration.
159    pub fn set_special_tokens(&mut self, special_tokens: SpecialTokens) {
160        self.special_tokens = special_tokens;
161    }
162
163    /// Registers a token string and corresponding token ID.
164    pub fn add_token(&mut self, token: impl Into<String>, id: u32) {
165        let text = token.into();
166        self.token_to_id.insert(text.clone(), id);
167        self.id_to_token.insert(id, text);
168    }
169
170    /// Number of vocabulary items currently registered.
171    #[must_use]
172    pub fn vocab_size(&self) -> usize {
173        self.token_to_id.len()
174    }
175}

177impl Tokenizer for SimpleTokenizer {
178    fn encode(&self, text: &str) -> Result<Vec<u32>, K3Error> {
179        let mut tokens = Vec::new();
180        if self.special_tokens.add_bos_token {
181            if let Some(bos) = self.special_tokens.bos_token_id {
182                tokens.push(bos);
183            }
184        }
185
186        for word in text.split_whitespace() {
187            if let Some(&id) = self.token_to_id.get(word) {
188                tokens.push(id);
189            } else {
190                return Err(K3Error::CapabilityMismatch {
191                    architecture: "tokenizer".to_owned(),
192                    reason: format!("unknown token '{word}' in input text"),
193                });
194            }
195        }
196
197        Ok(tokens)
198    }
199
200    fn decode(&self, tokens: &[u32]) -> Result<String, K3Error> {
201        let mut result = String::new();
202        for &id in tokens {
203            if let Some(piece) = self.decode_piece(id)? {
204                if !result.is_empty() {
205                    result.push(' ');
206                }
207                result.push_str(&piece);
208            }
209        }
210        Ok(result)
211    }
212
213    fn decode_piece(&self, token_id: u32) -> Result<Option<String>, K3Error> {
214        if self.special_tokens.is_stop(token_id) {
215            return Ok(None);
216        }
217        if let Some(token) = self.id_to_token.get(&token_id) {
218            Ok(Some(token.clone()))
219        } else {
220            Err(K3Error::CapabilityMismatch {
221                architecture: "tokenizer".to_owned(),
222                reason: format!("unknown token ID {token_id} in vocabulary"),
223            })
224        }
225    }
226
227    fn special_tokens(&self) -> &SpecialTokens {
228        &self.special_tokens
229    }
230}

232/// Full GGUF-backed vocabulary, BPE/SentencePiece tokenizer, and chat template engine (US-01.4).
233#[derive(Debug, Clone)]
234pub struct GgufTokenizer {
235    model_type: String,
236    id_to_token: Vec<String>,
237    token_to_id: HashMap<String, u32>,
238    scores: Vec<f32>,
239    token_types: Vec<TokenType>,
240    merges: Vec<(String, String)>,
241    merge_ranks: HashMap<(String, String), usize>,
242    byte_tokens: [Option<u32>; 256],
243    control_tokens: Vec<(String, u32)>,
244    special_tokens: SpecialTokens,
245    chat_template: ChatTemplate,
246}
247
248impl GgufTokenizer {
249    /// Extracts vocabulary tokens, scores, token types, BPE merges, special control tokens,
250    /// and chat template definitions from a parsed [`GgufFile`] (US-01.4.1, ADR-025).
251    ///
252    /// # Errors
253    ///
254    /// Returns [`K3Error::CapabilityMismatch`] if required vocabulary metadata (`tokenizer.ggml.tokens`)
255    /// is missing or malformed.
256    #[allow(clippy::too_many_lines, clippy::similar_names)]
257    pub fn from_gguf(gguf: &GgufFile<'_>) -> Result<Self, K3Error> {
258        let arch = gguf.architecture();
259
260        let model_type = gguf
261            .metadata_value("tokenizer.ggml.model")
262            .and_then(GgufValue::as_str)
263            .unwrap_or("gpt2")
264            .to_owned();
265
266        let tokens_val = gguf
267            .metadata_value("tokenizer.ggml.tokens")
268            .ok_or_else(|| K3Error::CapabilityMismatch {
269                architecture: arch.to_owned(),
270                reason: "missing required 'tokenizer.ggml.tokens' array in GGUF metadata"
271                    .to_owned(),
272            })?;
273
274        let id_to_token: Vec<String> = match tokens_val.as_array() {
275            Some(GgufArray::String(items)) => items.iter().map(|s| (*s).to_owned()).collect(),
276            _ => {
277                return Err(K3Error::CapabilityMismatch {
278                    architecture: arch.to_owned(),
279                    reason: "'tokenizer.ggml.tokens' must be a string array".to_owned(),
280                });
281            }
282        };
283
284        if id_to_token.is_empty() {
285            return Err(K3Error::CapabilityMismatch {
286                architecture: arch.to_owned(),
287                reason: "'tokenizer.ggml.tokens' vocabulary array cannot be empty".to_owned(),
288            });
289        }
290
291        let scores: Vec<f32> = match gguf
292            .metadata_value("tokenizer.ggml.scores")
293            .and_then(GgufValue::as_array)
294        {
295            Some(GgufArray::Float32(vals)) => vals.clone(),
296            _ => vec![0.0_f32; id_to_token.len()],
297        };
298
299        let token_types: Vec<TokenType> = match gguf
300            .metadata_value("tokenizer.ggml.token_type")
301            .and_then(GgufValue::as_array)
302        {
303            Some(GgufArray::Int32(vals)) => {
304                vals.iter().copied().map(TokenType::from_gguf_i32).collect()
305            }
306            Some(GgufArray::Uint32(vals)) => vals
307                .iter()
308                .copied()
309                .map(|u| TokenType::from_gguf_i32(i32::try_from(u).unwrap_or(1)))
310                .collect(),
311            _ => vec![TokenType::Normal; id_to_token.len()],
312        };
313
314        let mut merges = Vec::new();
315        let mut merge_ranks = HashMap::new();
316        if let Some(GgufArray::String(raw_merges)) = gguf
317            .metadata_value("tokenizer.ggml.merges")
318            .and_then(GgufValue::as_array)
319        {
320            for (rank, entry) in raw_merges.iter().enumerate() {
321                if let Some((left, right)) = entry.split_once(' ') {
322                    let pair = (left.to_owned(), right.to_owned());
323                    merge_ranks.entry(pair.clone()).or_insert(rank);
324                    merges.push(pair);
325                }
326            }
327        }
328
329        let mut token_to_id = HashMap::with_capacity(id_to_token.len());
330        let mut byte_tokens = [None; 256];
331        let mut control_tokens = Vec::new();
332
333        for (idx, text) in id_to_token.iter().enumerate() {
334            let Ok(id) = u32::try_from(idx) else {
335                continue;
336            };
337            token_to_id.entry(text.clone()).or_insert(id);
338
339            if let Some(byte_val) = parse_hex_byte_token(text) {
340                byte_tokens[usize::from(byte_val)] = Some(id);
341            }
342
343            let ttype = token_types.get(idx).copied().unwrap_or(TokenType::Normal);
344            if ttype == TokenType::Control && !text.is_empty() {
345                control_tokens.push((text.clone(), id));
346            }
347        }
348
349        // Sort control tokens longest-first so greedy matching prefers longer markers
350        control_tokens.sort_by_key(|b| std::cmp::Reverse(b.0.len()));
351
352        let bos_token_id = gguf
353            .metadata_value("tokenizer.ggml.bos_token_id")
354            .and_then(GgufValue::as_u32);
355        let eos_token_id = gguf
356            .metadata_value("tokenizer.ggml.eos_token_id")
357            .and_then(GgufValue::as_u32);
358        let eot_token_id = gguf
359            .metadata_value("tokenizer.ggml.eot_token_id")
360            .and_then(GgufValue::as_u32);
361        let padding_token_id = gguf
362            .metadata_value("tokenizer.ggml.padding_token_id")
363            .and_then(GgufValue::as_u32);
364        let add_bos_token = gguf
365            .metadata_value("tokenizer.ggml.add_bos_token")
366            .and_then(GgufValue::as_bool)
367            .unwrap_or(false);
368
369        let mut stop_tokens = Vec::new();
370        if let Some(eos) = eos_token_id {
371            stop_tokens.push(eos);
372        }
373        if let Some(eot) = eot_token_id {
374            if !stop_tokens.contains(&eot) {
375                stop_tokens.push(eot);
376            }
377        }
378
379        let mut stop_sequences = Vec::new();
380        for marker in ["<|im_end|>", "<|eot_id|>", "</s>", "<|endoftext|>"] {
381            if let Some(&id) = token_to_id.get(marker) {
382                if !stop_tokens.contains(&id) {
383                    stop_tokens.push(id);
384                }
385            }
386            stop_sequences.push(marker.to_owned());
387        }
388
389        let special_tokens = SpecialTokens {
390            bos_token_id,
391            eos_token_id,
392            eot_token_id,
393            padding_token_id,
394            add_bos_token,
395            stop_tokens,
396            stop_sequences,
397        };
398
399        let chat_template_str = gguf
400            .metadata_value("tokenizer.chat_template")
401            .and_then(GgufValue::as_str)
402            .map(str::to_owned);
403        let chat_template = ChatTemplate::new(chat_template_str);
404
405        Ok(Self {
406            model_type,
407            id_to_token,
408            token_to_id,
409            scores,
410            token_types,
411            merges,
412            merge_ranks,
413            byte_tokens,
414            control_tokens,
415            special_tokens,
416            chat_template,
417        })
418    }

470    /// Formats a user prompt with the model's chat template (if `apply_chat_template` is `true`),
471    /// tokenizes it into token IDs, and validates that `tokens.len() < n_ctx` (US-01.4.2, ADR-027).
472    ///
473    /// # Errors
474    ///
475    /// - Returns [`K3Error::ContextLengthExceeded`] if `n_ctx == 0` or `tokens.len() >= n_ctx`.
476    /// - Returns [`K3Error::CapabilityMismatch`] if input text contains unencodable sequences.
477    pub fn tokenize_prompt(
478        &self,
479        prompt: &str,
480        apply_chat_template: bool,
481        n_ctx: usize,
482    ) -> Result<Vec<u32>, K3Error> {
483        let formatted = if apply_chat_template {
484            self.chat_template.format_prompt(prompt)
485        } else {
486            prompt.to_owned()
487        };
488
489        let tokens = self.encode(&formatted)?;
490        if n_ctx == 0 || tokens.len() >= n_ctx {
491            return Err(K3Error::ContextLengthExceeded {
492                requested_tokens: tokens.len(),
493                max_context_tokens: n_ctx,
494            });
495        }
496
497        Ok(tokens)
498    }

506    fn encode_segment(&self, segment: &str, out: &mut Vec<u32>) -> Result<(), K3Error> {
507        if segment.is_empty() {
508            return Ok(());
509        }
510
511        // Fast path: exact match in vocabulary
512        if let Some(&id) = self.token_to_id.get(segment) {
513            out.push(id);
514            return Ok(());
515        }
516
517        // SentencePiece style space normalization if model_type is "llama" and vocab uses '▁'
518        let normalized;
519        let text_to_process = if self.model_type == "llama"
520            && self.token_to_id.keys().any(|k| k.contains('\u{2581}'))
521        {
522            normalized = segment.replace(' ', "\u{2581}");
523            normalized.as_str()
524        } else {
525            segment
526        };
527
528        // If BPE merges are configured, attempt BPE merge encoding on words/whitespace chunks
529        if !self.merge_ranks.is_empty() && self.try_bpe_encode(text_to_process, out) {
530            return Ok(());
531        }
532
533        // Greedy longest-prefix match in vocabulary with byte-token fallback
534        let mut cursor = 0usize;
535        let bytes = text_to_process.as_bytes();
536        while cursor < bytes.len() {
537            let remaining = &text_to_process[cursor..];
538            let mut matched: Option<(usize, u32)> = None;
539
540            // Find the longest prefix of `remaining` present in `token_to_id`
541            for (char_idx, ch) in remaining.char_indices() {
542                let end = char_idx + ch.len_utf8();
543                let candidate = &remaining[..end];
544                if let Some(&id) = self.token_to_id.get(candidate) {
545                    matched = Some((end, id));
546                }
547            }
548
549            if let Some((len, id)) = matched {
550                out.push(id);
551                cursor += len;
552            } else {
553                // Fallback to `<0xXX>` byte token if available
554                let b = bytes[cursor];
555                if let Some(byte_id) = self.byte_tokens[usize::from(b)] {
556                    out.push(byte_id);
557                    cursor += 1;
558                } else if let Some(pad_id) = self.special_tokens.padding_token_id {
559                    out.push(pad_id);
560                    cursor += 1;
561                } else {
562                    let ch = remaining.chars().next().unwrap_or('\0');
563                    return Err(K3Error::CapabilityMismatch {
564                        architecture: "tokenizer".to_owned(),
565                        reason: format!(
566                            "unable to encode character '{ch}' (byte 0x{b:02X}) in segment '{segment}'"
567                        ),
568                    });
569                }
570            }
571        }
572
573        Ok(())
574    }
575
576    fn try_bpe_encode(&self, text: &str, out: &mut Vec<u32>) -> bool {
577        let mut symbols: Vec<String> = text.chars().map(|c| c.to_string()).collect();
578        if symbols.is_empty() {
579            return true;
580        }
581
582        loop {
583            if symbols.len() < 2 {
584                break;
585            }
586
587            let mut best_pair: Option<(usize, usize)> = None; // (rank, index)
588            for i in 0..(symbols.len() - 1) {
589                let pair = (symbols[i].clone(), symbols[i + 1].clone());
590                if let Some(&rank) = self.merge_ranks.get(&pair) {
591                    if best_pair.is_none_or(|(best_rank, _)| rank < best_rank) {
592                        best_pair = Some((rank, i));
593                    }
594                }
595            }
596
597            let Some((_, idx)) = best_pair else {
598                break;
599            };
600
601            let merged = format!("{}{}", symbols[idx], symbols[idx + 1]);
602            symbols[idx] = merged;
603            symbols.remove(idx + 1);
604        }
605
606        let mut staged = Vec::with_capacity(symbols.len());
607        for sym in &symbols {
608            if let Some(&id) = self.token_to_id.get(sym.as_str()) {
609                staged.push(id);
610            } else {
611                return false;
612            }
613        }
614
615        out.extend(staged);
616        true
617    }

619    fn token_raw_bytes(&self, token_id: u32) -> Result<Option<Vec<u8>>, K3Error> {
620        let idx = usize::try_from(token_id).map_err(|_| K3Error::CapabilityMismatch {
621            architecture: "tokenizer".to_owned(),
622            reason: format!("token ID {token_id} overflows usize"),
623        })?;
624
625        let piece = self
626            .id_to_token
627            .get(idx)
628            .ok_or_else(|| K3Error::CapabilityMismatch {
629                architecture: "tokenizer".to_owned(),
630                reason: format!("unknown token ID {token_id} in vocabulary"),
631            })?;
632
633        let ttype = self
634            .token_types
635            .get(idx)
636            .copied()
637            .unwrap_or(TokenType::Normal);
638        if ttype == TokenType::Control {
639            return Ok(None);
640        }
641
642        if let Some(byte_val) = parse_hex_byte_token(piece) {
643            return Ok(Some(vec![byte_val]));
644        }
645
646        if self.model_type == "llama" && piece.contains('\u{2581}') {
647            return Ok(Some(piece.replace('\u{2581}', " ").into_bytes()));
648        }
649
650        Ok(Some(piece.as_bytes().to_vec()))
651    }

654impl Tokenizer for GgufTokenizer {
655    fn encode(&self, text: &str) -> Result<Vec<u32>, K3Error> {
656        let mut tokens = Vec::new();
657        if self.special_tokens.add_bos_token {
658            if let Some(bos) = self.special_tokens.bos_token_id {
659                tokens.push(bos);
660            }
661        }
662
663        let mut cursor = 0usize;
664        while cursor < text.len() {
665            let remaining = &text[cursor..];
666
667            // 1. Check if `remaining` starts with a control token
668            if let Some((ctrl_str, ctrl_id)) = self
669                .control_tokens
670                .iter()
671                .find(|(ctrl, _)| remaining.starts_with(ctrl.as_str()))
672            {
673                tokens.push(*ctrl_id);
674                cursor += ctrl_str.len();
675                continue;
676            }
677
678            // 2. Find the next control token occurrence so we can encode the normal span before it
679            let next_ctrl_offset = self
680                .control_tokens
681                .iter()
682                .filter_map(|(ctrl, _)| remaining.find(ctrl.as_str()))
683                .min()
684                .unwrap_or(remaining.len());
685
686            let span = &remaining[..next_ctrl_offset];
687            self.encode_segment(span, &mut tokens)?;
688            cursor += next_ctrl_offset;
689        }
690
691        Ok(tokens)
692    }
693
694    fn decode(&self, tokens: &[u32]) -> Result<String, K3Error> {
695        let mut detok = self.streaming_detokenizer();
696        let mut out = String::new();
697        for &id in tokens {
698            match detok.push_token(id)? {
699                DetokenizeStep::Emit(piece) => out.push_str(&piece),
700                DetokenizeStep::Hold => {}
701                DetokenizeStep::Stop(prefix) => {
702                    out.push_str(&prefix);
703                    break;
704                }
705            }
706        }
707        out.push_str(&detok.finish());
708        Ok(out)
709    }
710
711    fn decode_piece(&self, token_id: u32) -> Result<Option<String>, K3Error> {
712        if self.special_tokens.is_stop(token_id) {
713            return Ok(None);
714        }
715        let Some(bytes) = self.token_raw_bytes(token_id)? else {
716            return Ok(None);
717        };
718        Ok(std::str::from_utf8(&bytes).ok().map(str::to_owned))
719    }
720
721    fn special_tokens(&self) -> &SpecialTokens {
722        &self.special_tokens
723    }
724}

726/// Output state of a single step in [`StreamingDetokenizer`].
727#[derive(Debug, Clone, PartialEq, Eq)]
728pub enum DetokenizeStep {
729    /// Emits a complete, valid UTF-8 text fragment ready for immediate `stdout` flushing.
730    Emit(String),
731    /// Holds buffered bytes because a multi-byte UTF-8 sequence or potential stop sequence is incomplete.
732    Hold,
733    /// Generation reached a stop token or text stop sequence; emits any remaining text preceding the stop marker.
734    Stop(String),
735}
736
737/// Stateful incremental token-to-UTF-8 streaming detokenizer with stop-sequence detection (US-01.4.3).
738#[derive(Debug)]
739pub struct StreamingDetokenizer<'a> {
740    tokenizer: &'a GgufTokenizer,
741    pending_utf8_bytes: Vec<u8>,
742    pending_text: String,
743    stopped: bool,
744}
745
746impl<'a> StreamingDetokenizer<'a> {
747    /// Creates a new [`StreamingDetokenizer`] for `tokenizer`.
748    #[must_use]
749    pub const fn new(tokenizer: &'a GgufTokenizer) -> Self {
750        Self {
751            tokenizer,
752            pending_utf8_bytes: Vec::new(),
753            pending_text: String::new(),
754            stopped: false,
755        }
756    }
757
758    /// Returns `true` if a stop token or stop sequence has been triggered.
759    #[must_use]
760    pub const fn is_stopped(&self) -> bool {
761        self.stopped
762    }
763
764    /// Feeds the next generated `token_id` into the incremental detokenizer.
765    ///
766    /// Buffers incomplete UTF-8 multi-byte sequences across token boundaries without emitting
767    /// `U+FFFD` replacement characters, and halts cleanly on token or text stop sequences.
768    ///
769    /// # Errors
770    ///
771    /// Returns [`K3Error::CapabilityMismatch`] if `token_id` is out of vocabulary bounds.
772    pub fn push_token(&mut self, token_id: u32) -> Result<DetokenizeStep, K3Error> {
773        if self.stopped {
774            return Ok(DetokenizeStep::Stop(String::new()));
775        }
776
777        // 1. Check token-ID stop condition (EOS / EOT / explicit stop_tokens)
778        if self.tokenizer.special_tokens().is_stop(token_id) {
779            self.stopped = true;
780            let flushed = core::mem::take(&mut self.pending_text);
781            return Ok(DetokenizeStep::Stop(flushed));
782        }
783
784        // 2. Append raw token bytes to pending UTF-8 buffer
785        if let Some(raw_bytes) = self.tokenizer.token_raw_bytes(token_id)? {
786            self.pending_utf8_bytes.extend_from_slice(&raw_bytes);
787        } else {
788            return Ok(DetokenizeStep::Hold);
789        }
790
791        // 3. Extract complete UTF-8 characters from `pending_utf8_bytes`
792        self.drain_valid_utf8();
793
794        if self.pending_text.is_empty() {
795            return Ok(DetokenizeStep::Hold);
796        }
797
798        // 4. Check text-level stop sequences and partial stop-sequence suffixes
799        let stop_seqs = &self.tokenizer.special_tokens().stop_sequences;
800
801        // Check if any full stop sequence appears in `pending_text`
802        let mut earliest_stop: Option<usize> = None;
803        for stop in stop_seqs {
804            if !stop.is_empty() {
805                if let Some(pos) = self.pending_text.find(stop.as_str()) {
806                    if earliest_stop.is_none_or(|prev| pos < prev) {
807                        earliest_stop = Some(pos);
808                    }
809                }
810            }
811        }
812
813        if let Some(stop_pos) = earliest_stop {
814            self.stopped = true;
815            let prefix = self.pending_text[..stop_pos].to_owned();
816            self.pending_text.clear();
817            return Ok(DetokenizeStep::Stop(prefix));
818        }
819
820        // Check if the tail of `pending_text` is a partial prefix of any stop sequence
821        let holdback_len = longest_stop_prefix_suffix(&self.pending_text, stop_seqs);
822        if holdback_len == self.pending_text.len() {
823            return Ok(DetokenizeStep::Hold);
824        }
825
826        let emit_end = self.pending_text.len() - holdback_len;
827        let emitted = self.pending_text[..emit_end].to_owned();
828        self.pending_text.drain(..emit_end);
829
830        if emitted.is_empty() {
831            Ok(DetokenizeStep::Hold)
832        } else {
833            Ok(DetokenizeStep::Emit(emitted))
834        }
835    }
836
837    /// Flushes any remaining held-back text at the end of generation.
838    #[must_use]
839    pub fn finish(&mut self) -> String {
840        self.drain_valid_utf8();
841        core::mem::take(&mut self.pending_text)
842    }
843
844    fn drain_valid_utf8(&mut self) {
845        loop {
846            if self.pending_utf8_bytes.is_empty() {
847                break;
848            }
849            match std::str::from_utf8(&self.pending_utf8_bytes) {
850                Ok(valid) => {
851                    self.pending_text.push_str(valid);
852                    self.pending_utf8_bytes.clear();
853                    break;
854                }
855                Err(err) => {
856                    let valid_up_to = err.valid_up_to();
857                    if valid_up_to > 0 {
858                        if let Ok(valid_prefix) =
859                            std::str::from_utf8(&self.pending_utf8_bytes[..valid_up_to])
860                        {
861                            self.pending_text.push_str(valid_prefix);
862                        }
863                        self.pending_utf8_bytes.drain(..valid_up_to);
864                    }
865
866                    if let Some(invalid_len) = err.error_len() {
867                        // Skip genuinely invalid UTF-8 byte sequence without emitting U+FFFD
868                        self.pending_utf8_bytes.drain(..invalid_len);
869                    } else {
870                        // Incomplete multi-byte UTF-8 sequence at end of buffer; wait for next token
871                        break;
872                    }
873                }
874            }
875        }
876    }
877}
878
879fn longest_stop_prefix_suffix(text: &str, stop_sequences: &[String]) -> usize {
880    let mut max_holdback = 0usize;
881    for stop in stop_sequences {
882        if stop.is_empty() {
883            continue;
884        }
885        // Check all proper non-empty prefixes of `stop`
886        for (char_idx, _) in stop.char_indices().skip(1) {
887            let prefix = &stop[..char_idx];
888            if text.ends_with(prefix) && prefix.len() > max_holdback {
889                max_holdback = prefix.len();
890            }
891        }
892    }
893    max_holdback
894}

896fn parse_hex_byte_token(token: &str) -> Option<u8> {
897    let inner = token.strip_prefix("<0x")?.strip_suffix('>')?;
898    if inner.len() == 2 {
899        u8::from_str_radix(inner, 16).ok()
900    } else {
901        None
902    }
